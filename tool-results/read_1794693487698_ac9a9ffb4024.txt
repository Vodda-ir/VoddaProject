   101→   101→function sortTeachers(rows: Teacher[], sort: SortKey): Teacher[] {
   102→   102→  const copy = [...rows];
   103→   103→  switch (sort) {
   104→   104→    case "rating":
   105→   105→      return copy.sort((a, b) => b.rating - a.rating);
   106→   106→    case "sessions":
   107→   107→      return copy.sort((a, b) => b.sessions - a.sessions);
   108→   108→    case "newest":
   109→   109→      return copy.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
   110→   110→    default:
   111→   111→      return copy.sort((a, b) => recommendationScore(b) - recommendationScore(a));
   112→   112→  }
   113→   113→}
