    50→function statusLabel(status: string, t: (key: TranslationKey) => string): string {
    51→  switch (status) {
    52→    case "pending": return t("status_pending");
    53→    case "confirmed": return t("status_confirmed");
    54→    case "done": return t("status_done");
    55→    case "canceled": return t("status_canceled");
    56→    case "awaiting_payment": return t("status_awaiting_payment");
    57→    default: return status;
    58→  }
    59→}

    61→function officialTime(detail: SessionDetailData): string {
    62→  if (!detail.localDate || !detail.localStart || !detail.localEnd) {
    63→    return "زمان رسمی جلسه دریافت نشد";
    64→  }
    65→  return `${formatJalaliDate(detail.localDate)} · ${toFaDigits(detail.localStart)} تا ${toFaDigits(detail.localEnd)}`;
    66→}
    67→
    68→function startMs(detail: SessionDetailData): number {
    69→  return Date.parse(detail.scheduledAt);
    70→}
    71→
    72→/** پنجرهٔ ورود به کلاس تصویری: ۱۰ دقیقه قبل از شروع تا بعد از پایان. */
    73→function canJoinVideo(detail: SessionDetailData): boolean {
    74→  const start = startMs(detail);
    75→  if (detail.status !== "confirmed" || Number.isNaN(start)) return false;
    76→  return Date.now() >= start - 10 * 60_000;
    77→}
    78→
    79→/** پایان جلسه از لحظهٔ آغاز رسمی مجاز است. */
    80→function canEndSession(detail: SessionDetailData): boolean {
    81→  const start = startMs(detail);
    82→  if (detail.status !== "confirmed" || Number.isNaN(start)) return false;
    83→  return Date.now() >= start;
    84→}
    85→
    86→function timeHint(detail: SessionDetailData): string {
    87→  const start = startMs(detail);
    88→  if (Number.isNaN(start)) return "";
    89→  const diff = start - Date.now();
    90→  if (diff <= 0) return "";
    91→  const minutes = Math.ceil(diff / 60_000);
    92→  if (minutes <= 60) return `شروع کلاس تا ${toFaDigits(minutes)} دقیقه دیگر`;
    93→  return `شروع کلاس ${formatJalaliDate(detail.localDate)} ساعت ${toFaDigits(detail.localStart)}`;
    94→}

    98→function CoordinationChat({
    99→  detail,
   100→  messages,
   101→  sending,
   102→  onSend,
   103→  peerLabel,
   104→}: {
   105→  detail: SessionDetailData;
   106→  messages: SessionMessageData[];
   107→  sending: boolean;
   108→  onSend: (body: string) => void;
   109→  peerLabel: string;
   110→}) {
   111→  const [draft, setDraft] = useState("");
   112→  const bodyRef = useRef<HTMLDivElement | null>(null);
   113→
   114→  useEffect(() => {
   115→    const body = bodyRef.current;
   116→    if (body) body.scrollTop = body.scrollHeight;
   117→  }, [messages.length]);
   118→
   119→  const submit = () => {
   120→    const text = draft.trim();
   121→    if (!text || sending) return;
   122→    onSend(text);
   123→    setDraft("");
   124→  };
   125→
   126→  return (
   127→    <section aria-label="چت هماهنگی جلسه" style={{ display: "grid", gap: 10 }}>
   128→      <SectionHeader title="چت هماهنگی" description="برای هماهنگی شروع جلسه، پیام‌های شما اینجا رد و بدل می‌شود." />
   129→      <Card style={{ display: "grid", gap: 0, padding: 0, overflow: "hidden" }}>
   130→        <div
   131→          ref={bodyRef}
   132→          style={{
   133→            maxHeight: 300,
   134→            overflowY: "auto",
   135→            display: "grid",
   136→            gap: 8,
   137→            alignContent: "start",
   138→            padding: 14,
   139→            background: colors.sand50,
   140→          }}
   141→        >
   142→          {messages.length === 0 ? (
   143→            <p style={{ margin: "8px 0", textAlign: "center", fontSize: 12, color: colors.inkMuted }}>
   144→              هنوز پیامی رد و بدل نشده؛ اولین پیام را بفرستید.
   145→            </p>
   146→          ) : null}
   147→          {messages.map((message) => {
   148→            const mine = isMyMessage(message, detail);
   149→            return (
   150→              <div
   151→                key={message.id}
   152→                style={{
   153→                  maxWidth: "84%",
   154→                  justifySelf: mine ? "start" : "end",
   155→                  display: "grid",
   156→                  gap: 2,
   157→                }}
   158→              >
   159→                <span
   160→                  style={{
   161→                    display: "block",
   162→                    fontSize: 12.5,
   163→                    lineHeight: 1.9,
   164→                    color: colors.ink,
   165→                    background: mine ? colors.olive50 : colors.surface,
   166→                    border: `1px solid ${mine ? colors.olive100 : colors.sand150}`,
   167→                    borderRadius: 14,
   168→                    padding: "8px 12px",
   169→                    whiteSpace: "pre-wrap",
   170→                    wordBreak: "break-word",
   171→                  }}
   172→                >
   173→                  {message.body}
   174→                </span>
   175→                <span style={{ fontSize: 9.5, color: colors.inkFaint, textAlign: mine ? "right" : "left" }}>
   176→                  {mine ? "شما" : peerLabel}
   177→                </span>
   178→              </div>
   179→            );
   180→          })}
   181→        </div>
   182→        <div style={{ display: "flex", gap: 8, padding: 10, borderTop: `1px solid ${colors.sand150}`, background: colors.surface }}>
   183→          <input
   184→            value={draft}
   185→            onChange={(event) => setDraft(event.target.value)}
   186→            onKeyDown={(event) => {
   187→              if (event.key === "Enter") {
   188→                event.preventDefault();
   189→                submit();
   190→              }
   191→            }}
   192→            maxLength={1000}
   193→            aria-label="متن پیام هماهنگی"
   194→            placeholder="پیام خود را بنویسید…"
   195→            style={{
   196→              flex: 1,
   197→              minWidth: 0,
   198→              minHeight: 44,
   199→              borderRadius: 14,
   200→              border: `1px solid ${colors.sand200}`,
   201→              background: colors.sand50,
   202→              color: colors.ink,
   203→              padding: "0 14px",
   204→              fontFamily: typography.fontFamily,
   205→              fontSize: 13,
   206→            }}
   207→          />
   208→          <Button onClick={submit} loading={sending} style={{ minHeight: 44, padding: "0 18px" }} aria-label="ارسال پیام">
   209→            <Icon name="chevronL" size={18} />
   210→          </Button>
   211→        </div>
   212→      </Card>
   213→    </section>
   214→  );
   215→}

   233→  const [loading, setLoading] = useState(true);
   234→  const [error, setError] = useState<string | null>(null);
   235→  const [busy, setBusy] = useState(false);
   236→  const [actionError, setActionError] = useState<string | null>(null);
   237→  const [liveOpen, setLiveOpen] = useState(false);
   238→  const [rejectOpen, setRejectOpen] = useState(false);
   239→  const [rejectReason, setRejectReason] = useState("");
   240→  const [cancelOpen, setCancelOpen] = useState(false);
   241→  const [sending, setSending] = useState(false);
   242→

   244→  useEffect(() => {
   245→    if (detail) onRoleResolved?.(detail.viewerRole === "admin" ? "teacher" : detail.viewerRole);
   246→  }, [detail, onRoleResolved]);
   247→
   248→  // نظرسنجی شاگرد بعد از جلسه
   249→  const [surveyOpen, setSurveyOpen] = useState(false);
   250→  const [rating, setRating] = useState(5);

   252→  const [reviewSaving, setReviewSaving] = useState(false);
   253→  const [reviewError, setReviewError] = useState<string | null>(null);
   254→
   255→  const load = useCallback(async () => {
   256→    if (!Number.isInteger(id) || id <= 0) {
   257→      setError("شناسه جلسه نامعتبر است.");
   258→      setLoading(false);
   259→      return;
   260→    }
   261→    try {
   262→      const next = await getSessionDetail(runtime, id);
   263→      setDetail(next);
   264→      setError(null);
   265→    } catch (loadError) {
   266→      setError(messageOf(loadError));
   267→    } finally {
   268→      setLoading(false);
   269→    }
   270→  }, [id, runtime]);
   271→
   272→  useEffect(() => {
   273→    void load();
   274→  }, [load]);
   275→
   276→  const loadMessages = useCallback(async () => {
   277→    if (!Number.isInteger(id) || id <= 0) return;
   278→    try {
   279→      setMessages(await getSessionMessages(runtime, id));
   280→    } catch {
   281→      // خطای موقت چت — دور بعد تلاش می‌شود
   282→    }
   283→  }, [id, runtime]);
   284→
   285→  useEffect(() => {
   286→    void loadMessages();
   287→    const timer = window.setInterval(() => void loadMessages(), 10_000);
   288→    return () => window.clearInterval(timer);
   289→  }, [loadMessages]);
   290→
   291→  async function runAction(action: () => Promise<void>) {
   292→    if (busy) return;
   293→    setBusy(true);
   294→    setActionError(null);
   295→    try {
   296→      await action();
   297→      await load();

   299→    } catch (actionFailure) {
   300→      setActionError(messageOf(actionFailure));
   301→    } finally {
   302→      setBusy(false);
   303→    }
   304→  }
   305→
   306→  async function submitMessage(body: string) {
   307→    setSending(true);
   308→    try {
   309→      await sendSessionMessage(runtime, id, body);
   310→      await loadMessages();
   311→    } catch (sendError) {
   312→      setActionError(messageOf(sendError));
   313→    } finally {
   314→      setSending(false);
   315→    }
   316→  }
   317→
   318→  async function submitReview() {
   319→    if (!detail || reviewSaving) return;
   320→    setReviewSaving(true);
   321→    setReviewError(null);
   322→    try {
   323→      await createSessionReview(runtime, { sessionId: detail.id, rating, text: reviewText });
   324→      setSurveyOpen(false);
   325→      await load();
   326→    } catch (reviewFailure) {
   327→      setReviewError(messageOf(reviewFailure));
   328→    } finally {
   329→      setReviewSaving(false);
   330→    }
   331→  }
   332→
   333→  if (loading) {
   334→    return <LoadingState label="در حال دریافت جزئیات جلسه…" style={{ minHeight: 220 }} />;
   335→  }
   336→
   337→  if (error || !detail) {

   345→      />
   346→    );
   347→  }
   348→
   349→  const isTeacher = detail.viewerRole === "teacher" || detail.viewerRole === "admin";
   350→  const isLearner = detail.viewerRole === "learner";
   351→  const peerLabel = isTeacher ? detail.learnerName : detail.teacherName;
   352→  const peerAvatar = isTeacher ? detail.learnerAvatar : detail.teacherAvatar;

   364→        className="vodda-enter"
   365→        style={{
   366→          position: "relative",
   367→          overflow: "hidden",
   368→          display: "grid",
   369→          gap: 14,
   370→          padding: 18,
   371→          borderRadius: 26,

   583→      {liveOpen ? (

   698→      {messages.length === 0 && detail.status === "pending" && isTeacher ? (

   703→        />
   704→      ) : null}
   705→    </section>
   706→  );
   707→}
   708→
   709→function canCancelSession(detail: SessionDetailData): boolean {
   710→  if (detail.status !== "pending" && detail.status !== "confirmed") return false;
   711→  const timestamp = startMs(detail);
   712→  return Number.isNaN(timestamp) || Date.now() < timestamp;
   713→}
