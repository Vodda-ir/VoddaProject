   132→   132→function faDate(iso?: string): string {
   133→   133→  if (!iso) return '—';
   134→   134→  try {
   135→   135→    return new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'short', year: 'numeric' }).format(new Date(iso));
   136→   136→  } catch {
   137→   137→    return '—';
   138→   138→  }
   139→   139→}
   140→   140→
   141→   141→function faDateTime(iso?: string): string {
   142→   142→  if (!iso) return '—';
   143→   143→  try {
   144→   144→    const day = new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'short' }).format(new Date(iso));
   145→   145→    const time = new Intl.DateTimeFormat('fa-IR', { hour: '2-digit', minute: '2-digit' }).format(new Date(iso));
   146→   146→    return `${day} · ${time}`;
   147→   147→  } catch {
   148→   148→    return '—';
   149→   149→  }
   150→   150→}
