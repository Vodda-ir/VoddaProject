    64→    64→export function LiveRoom({

    67→    67→  myUserId,
    68→    68→  peerName,
    69→    69→  peerAvatar,
    70→    70→  startAt,
    71→    71→  durationMinutes,
    72→    72→  onClose,
    73→    73→}: LiveRoomProps) {
    74→    74→  const [phase, setPhase] = useState<"preflight" | "live" | "ended">("preflight");
    75→    75→  const [status, setStatus] = useState<LiveStatus>("connecting");
    76→    76→  const [mediaError, setMediaError] = useState("");
    77→    77→  const [mediaDenied, setMediaDenied] = useState(false);
    78→    78→  const [mediaBusy, setMediaBusy] = useState(false);
    79→    79→  const [hasLiveMedia, setHasLiveMedia] = useState(false);
    80→    80→  const [micLevel, setMicLevel] = useState(0);
    81→    81→  const [micOn, setMicOn] = useState(true);
    82→    82→  const [camOn, setCamOn] = useState(true);
    83→    83→  const [remoteActive, setRemoteActive] = useState(false);
    84→    84→  const [chat, setChat] = useState<ChatLine[]>([]);
    85→    85→  const [chatDraft, setChatDraft] = useState("");
    86→    86→  const [chatOpen, setChatOpen] = useState(false);
    87→    87→  const [notice, setNotice] = useState("");
    88→    88→  const [countdown, setCountdown] = useState("");
    89→    89→
    90→    90→  const clientRef = useRef<LiveClient | null>(null);
    91→    91→  const engineRef = useRef<LiveEngine | null>(null);
    92→    92→  const localVideoRef = useRef<HTMLVideoElement | null>(null);
    93→    93→  const remoteVideoRef = useRef<HTMLVideoElement | null>(null);
    94→    94→  const liveStreamRef = useRef<MediaStream | null>(null);
    95→    95→  const previewStreamRef = useRef<MediaStream | null>(null);
    96→    96→  const mediaBusyRef = useRef(false);
    97→    97→  const previewRunIdRef = useRef(0);
    98→    98→  const stopPreviewRunRef = useRef<() => void>(() => {});
    99→    99→  const chatBodyRef = useRef<HTMLDivElement | null>(null);
   100→   100→  const peerOnlineRef = useRef(false);
   101→   101→  const joinAnnouncedRef = useRef(false);
   102→   102→  const peerSignaledRef = useRef(false);
   103→   103→  const engineRoleRef = useRef<EngineRole>(role);
   104→   104→  const wakeLockRef = useRef<{ release: () => Promise<void>; released: boolean } | null>(null);
   105→   105→  const [needsTapToPlay, setNeedsTapToPlay] = useState(false);
   106→   106→  const { L, lang, dir, N } = useAppearance();
   107→   107→
   108→   108→  const isTeacherSide = role !== "learner";
   109→   109→
   110→   110→  /* ---------- پیش‌آزمون: پیش‌نمایش دوربین و سطح میکروفون ---------- */
   111→   111→  // گرفتن رسانه با helper مشترک و نردبان پلکانی (قید مطلوب ← ساده ← فقط صدا)؛
   112→   112→  // دیالوگ مجوز مرورگر همین‌جا باز می‌شود و هر شکست با پیام راهنما و کلید «تلاش دوباره» قابل جبران است.
   113→   113→  const startPreview = useCallback(() => {
   114→   114→    stopPreviewRunRef.current();
   115→   115→    const runId = ++previewRunIdRef.current;
   116→   116→    mediaBusyRef.current = true;
   117→   117→    setMediaBusy(true);
   118→   118→    let ctx: AudioContext | null = null;
   119→   119→    let raf = 0;
   120→   120→    let cancelled = false;
   121→   121→    const cancelRun = () => {
   122→   122→      cancelled = true;
   123→   123→      cancelAnimationFrame(raf);
   124→   124→      void ctx?.close();
   125→   125→    };
   126→   126→    stopPreviewRunRef.current = cancelRun;
   127→   127→    void (async () => {
   128→   128→      previewStreamRef.current?.getTracks().forEach((track) => track.stop());
   129→   129→      previewStreamRef.current = null;
   130→   130→      const result = await acquireMediaStream("both");
   131→   131→      if (cancelled) {
   132→   132→        if (result.ok) result.stream.getTracks().forEach((track) => track.stop());
   133→   133→        if (runId === previewRunIdRef.current) {
   134→   134→          mediaBusyRef.current = false;
   135→   135→          setMediaBusy(false);
   136→   136→        }
   137→   137→        return;
   138→   138→      }
   139→   139→      mediaBusyRef.current = false;
   140→   140→      setMediaBusy(false);
   141→   141→      if (!result.ok) {
   142→   142→        setMediaError(L(result.message));
   143→   143→        setMediaDenied(result.denied);
   144→   144→        return;
   145→   145→      }
   146→   146→      const stream = result.stream;
   147→   147→      previewStreamRef.current = stream;
   148→   148→      setMediaError("");
   149→   149→      setMediaDenied(false);
   150→   150→      if (localVideoRef.current) localVideoRef.current.srcObject = stream;
   151→   151→      const [audioTrack] = stream.getAudioTracks();
   152→   152→      if (audioTrack) {
   153→   153→        ctx = new AudioContext();
   154→   154→        const source = ctx.createMediaStreamSource(new MediaStream([audioTrack]));
   155→   155→        const analyser = ctx.createAnalyser();
   156→   156→        analyser.fftSize = 256;
   157→   157→        source.connect(analyser);
   158→   158→        const buffer = new Uint8Array(analyser.frequencyBinCount);
   159→   159→        const tick = () => {
   160→   160→          analyser.getByteFrequencyData(buffer);
   161→   161→          const avg = buffer.reduce((sum, value) => sum + value, 0) / buffer.length;
   162→   162→          setMicLevel(Math.min(100, Math.round((avg / 128) * 100)));
   163→   163→          raf = requestAnimationFrame(tick);
   164→   164→        };
   165→   165→        tick();
   166→   166→      }
   167→   167→    })();
   168→   168→  }, [L]);
   169→   169→
   170→   170→  const retryMedia = useCallback(() => {
   171→   171→    if (mediaBusy) return;
   172→   172→    startPreview();
   173→   173→  }, [mediaBusy, startPreview]);
   174→   174→
   175→   175→  useEffect(() => {
   176→   176→    if (phase !== "preflight") return;
   177→   177→    startPreview();
   178→   178→    return () => {
   179→   179→      stopPreviewRunRef.current();
   180→   180→      const preview = previewStreamRef.current;
   181→   181→      preview?.getTracks().forEach((track) => track.stop());
   182→   182→      previewStreamRef.current = null;
   183→   183→    };
   184→   184→    // startPreview عمداً در وابستگی‌ها نمی‌آید؛ اجرای دوبارهٔ افکت جریان تازهٔ پیش‌نمایش را قطع می‌کرد.
   185→   185→  }, [phase]);
   186→   186→
   187→   187→  /* ---------- شمارش معکوس جلسه ---------- */
   188→   188→  useEffect(() => {
   189→   189→    if (!startAt) return;
   190→   190→    const endAt = new Date(startAt).getTime() + Math.max(1, durationMinutes) * 60_000;
   191→   191→    const tick = () => {
   192→   192→      const diff = endAt - Date.now();
   193→   193→      if (diff <= 0) {
   194→   194→        setCountdown(L("زمان جلسه به پایان رسیده است"));
   195→   195→        return;
   196→   196→      }
   197→   197→      const minutes = Math.floor(diff / 60_000);
   198→   198→      const seconds = Math.floor((diff % 60_000) / 1000);
   199→   199→      setCountdown(
   200→   200→        L("{time} باقی‌مانده").replace(
   201→   201→          "{time}",
   202→   202→          `${localizedDigits(minutes, lang)}:${localizedDigits(String(seconds).padStart(2, "0"), lang)}`,
   203→   203→        ),
   204→   204→      );
   205→   205→    };
   206→   206→    tick();
   207→   207→    const timer = window.setInterval(tick, 1000);
   208→   208→    return () => window.clearInterval(timer);
   209→   209→  }, [startAt, durationMinutes, L, lang]);
   210→   210→
   211→   211→  /* ---------- رویدادهای سیگنالینگ ---------- */
   212→   212→  const handleEvent = useCallback(
   213→   213→    (event: LiveEvent) => {
   214→   214→      if (event.from === myUserId && event.type !== "system") return;
   215→   215→      switch (event.type) {
   216→   216→        case "chat": {
   217→   217→          const payload = event.payload as { text?: string } | undefined;
   218→   218→          const text = String(payload?.text ?? "").slice(0, 1000);
   219→   219→          if (!text) return;
   220→   220→          setChat((items) => [...items, { id: uid(), from: event.from, text, at: event.at, mine: false }]);
   221→   221→          break;
   222→   222→        }
   223→   223→        case "control": {
   224→   224→          const payload = event.payload as { action: string; value?: boolean | string } | undefined;
   225→   225→          if (payload?.action === "bye") {
   226→   226→            setNotice(L("طرف مقابل کلاس را ترک کرد."));
   227→   227→            window.setTimeout(() => setNotice(""), 4000);
   228→   228→          }
   229→   229→          if (payload?.action === "end") setPhase("ended");
   230→   230→          break;
   231→   231→        }
   232→   232→        case "offer":
   233→   233→        case "answer":
   234→   234→        case "ice":
   235→   235→          peerSignaledRef.current = true;
   236→   236→          void engineRef.current?.handleSignal(event.type, event.payload);
   237→   237→          break;
   238→   238→        case "system": {
   239→   239→          const payload = event.payload as { action?: string } | undefined;
   240→   240→          if (payload?.action === "join") {
   241→   241→            peerOnlineRef.current = true;
   242→   242→            setChat((items) => [...items, { id: uid(), from: 0, text: L("طرف مقابل وارد کلاس شد."), at: event.at, mine: false }]);
   243→   243→            // هر ورود به کلاس فقط یک‌بار خودکار پیشنهاد می‌فرستد؛ join بازپخش‌شده/اکو‌شده
   244→   244→            // (replay کانال WS یا کانال زنده) نباید دوباره offer بفرستد.
   245→   245→            if (isTeacherSide && !joinAnnouncedRef.current) {
   246→   246→              joinAnnouncedRef.current = true;
   247→   247→              void engineRef.current?.initiate();
   248→   248→            }
   249→   249→          }
   250→   250→          if (payload?.action === "leave") {
   251→   251→            peerOnlineRef.current = false;
   252→   252→            setChat((items) => [...items, { id: uid(), from: 0, text: L("طرف مقابل خارج شد."), at: event.at, mine: false }]);
   253→   253→          }
   254→   254→          break;
   255→   255→        }
   256→   256→      }
   257→   257→    },
   258→   258→    [isTeacherSide, myUserId],
   259→   259→  );
   260→   260→
   261→   261→  useEffect(() => {
   262→   262→    return () => {
   263→   263→      clientRef.current?.close();
   264→   264→      engineRef.current?.close();
   265→   265→      liveStreamRef.current?.getTracks().forEach((track) => track.stop());
   266→   266→    };
   267→   267→  }, []);
   268→   268→
   269→   269→  useEffect(() => {
   270→   270→    const body = chatBodyRef.current;
   271→   271→    if (body) body.scrollTop = body.scrollHeight;
   272→   272→  }, [chat, chatOpen]);
   273→   273→
   274→   274→  /* ---------- موبایل: بیدار نگه‌داشتن صفحه در طول کلاس ---------- */
   275→   275→  useEffect(() => {
   276→   276→    if (phase !== "live") return;
   277→   277→    type WakeLockSentinel = { release: () => Promise<void>; released: boolean };
   278→   278→    const wakeLockApi = (navigator as Navigator & { wakeLock?: { request: (type: "screen") => Promise<WakeLockSentinel> } }).wakeLock;
   279→   279→    if (!wakeLockApi) return;
   280→   280→    let released = false;
   281→   281→    wakeLockApi.request("screen")
   282→   282→      .then((sentinel) => {
   283→   283→        if (released) {
   284→   284→          void sentinel.release().catch(() => {});
   285→   285→          return;
   286→   286→        }
   287→   287→        wakeLockRef.current = sentinel;
   288→   288→      })
   289→   289→      .catch(() => {});
   290→   290→    return () => {
   291→   291→      released = true;
   292→   292→      const sentinel = wakeLockRef.current;
   293→   293→      wakeLockRef.current = null;
   294→   294→      void sentinel?.release().catch(() => {});
   295→   295→    };
   296→   296→  }, [phase]);
   297→   297→
   298→   298→  /* ---------- موبایل: برگشت از پس‌زمینه/تعویض شبکه → بازیابی اتصال ---------- */
   299→   299→  useEffect(() => {
   300→   300→    if (phase !== "live") return;
   301→   301→    const recover = () => {
   302→   302→      void engineRef.current?.reconnect();
   303→   303→      const remote = remoteVideoRef.current;
   304→   304→      if (remote && remote.paused) {
   305→   305→        remote.play().catch(() => setNeedsTapToPlay(true));
   306→   306→      }
   307→   307→    };
   308→   308→    const onVisible = () => {
   309→   309→      if (document.visibilityState === "visible") recover();
   310→   310→    };
   311→   311→    document.addEventListener("visibilitychange", onVisible);
   312→   312→    window.addEventListener("online", recover);
   313→   313→    return () => {
   314→   314→      document.removeEventListener("visibilitychange", onVisible);
   315→   315→      window.removeEventListener("online", recover);
   316→   316→    };
   317→   317→  }, [phase]);
   318→   318→
   319→   319→  /* ---------- معلم: اگر ۹ ثانیه پس از ورود هیچ قاب سیگنالینگ از طرف مقابل نرسید، یک‌بار initiate می‌فرستد ---------- */
   320→   320→  useEffect(() => {
   321→   321→    if (phase !== "live" || !isTeacherSide) return;
   322→   322→    const timer = window.setTimeout(() => {
   323→   323→      if (!peerSignaledRef.current) void engineRef.current?.initiate();
   324→   324→    }, 9_000);
   325→   325→    return () => window.clearTimeout(timer);
   326→   326→  }, [phase, isTeacherSide]);
   327→   327→
   328→   328→  /* ---------- پخش خودکار ویدیوی طرف مقابل (سیاست autoplay موبایل) ---------- */
   329→   329→  useEffect(() => {
   330→   330→    if (!remoteActive) return;
   331→   331→    const remote = remoteVideoRef.current;
   332→   332→    if (!remote) return;
   333→   333→    remote.play().then(() => setNeedsTapToPlay(false)).catch(() => setNeedsTapToPlay(true));
   334→   334→  }, [remoteActive]);
   335→   335→
   336→   336→  /* ---------- ورود به کلاس ---------- */
   337→   337→  // جریان پیش‌نمایش اگر آماده باشد مستقیم برای جلسه به‌کار می‌رود تا دیالوگ مجوز دوباره باز نشود؛
   338→   338→  // اگر نباشد، «ورود با رسانه» همان‌جا درخواست می‌دهد — و شکست هیچ‌وقت ورود به کلاس را блок نمی‌کند.
   339→   339→  async function join(withMedia: boolean) {
   340→   340→    joinAnnouncedRef.current = false;
   341→   341→    stopPreviewRunRef.current();
   342→   342→
   343→   343→    let stream: MediaStream | null = previewStreamRef.current;
   344→   344→    previewStreamRef.current = null;
   345→   345→    if (!withMedia) {
   346→   346→      stream?.getTracks().forEach((track) => track.stop());
   347→   347→      stream = null;
   348→   348→    } else if (!stream) {
   349→   349→      setMediaBusy(true);
   350→   350→      const result = await acquireMediaStream("both");
   351→   351→      setMediaBusy(false);
   352→   352→      if (!result.ok) {
   353→   353→        setMediaError(L(result.message));
   354→   354→        setMediaDenied(result.denied);
   355→   355→        // بدون رسانه هم وارد کلاس می‌شویم — هیچ‌چیز نباید برگزاری کلاس را متوقف کند.
   356→   356→      } else {
   357→   357→        stream = result.stream;
   358→   358→        setMediaError("");
   359→   359→        setMediaDenied(false);
   360→   360→      }
   361→   361→    }
   362→   362→    liveStreamRef.current = stream;
   363→   363→    setHasLiveMedia(Boolean(stream));
   364→   364→    setPhase("live");
   365→   365→
   366→   366→    const client = new LiveClient({
   367→   367→      sessionId,
   368→   368→      onEvent: handleEvent,
   369→   369→      onStatus: (value) => setStatus(value),
   370→   370→    });
   371→   371→    clientRef.current = client;
   372→   372→
   373→   373→    const engine = new LiveEngine({
   374→   374→      role: engineRoleRef.current,
   375→   375→      onRemoteStream: (remote) => {
   376→   376→        setRemoteActive(remote.getTracks().length > 0);
   377→   377→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   378→   378→      },
   379→   379→      onState: () => {},
   380→   380→      send: (type, payload) => client.send(type, payload),
   381→   381→    });
   382→   382→    engineRef.current = engine;
   383→   383→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   384→   384→    await engine.ensureConnection();
   385→   385→    if (stream) await engine.setLocalMedia(stream);
   386→   386→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   387→   387→    if (engine.isInitiator && peerOnlineRef.current) void engine.initiate();
   388→   388→
   389→   389→    await client.connect();
   390→   390→  }
   391→   391→
   392→   392→  function sendChat() {
   393→   393→    const text = chatDraft.trim();
   394→   394→    if (!text || !clientRef.current) return;
   395→   395→    setChat((items) => [...items, { id: uid(), from: myUserId, text, at: Date.now(), mine: true }]);
   396→   396→    clientRef.current.send("chat", { text: text.slice(0, 1000) });
   397→   397→    setChatDraft("");
   398→   398→  }
   399→   399→
   400→   400→  function toggleMic() {
   401→   401→    const next = !micOn;
   402→   402→    setMicOn(next);
   403→   403→    liveStreamRef.current?.getAudioTracks().forEach((track) => (track.enabled = next));
   404→   404→  }
   405→   405→
   406→   406→  function toggleCam() {
   407→   407→    const next = !camOn;
   408→   408→    setCamOn(next);
   409→   409→    liveStreamRef.current?.getVideoTracks().forEach((track) => (track.enabled = next));
   410→   410→  }
   411→   411→
   412→   412→  /* ---------- وصل دوبارهٔ دوربین و میکروفون در میانهٔ کلاس ---------- */
   413→   413→  // وقتی کاربر بدون رسانه وارد شده، این مسیر از نوار کنترل دیالوگ مجوز را باز می‌کند
   414→   414→  // و تراک‌ها را به موتور P2P می‌دهد — خروج از کلاس لازم نیست.
   415→   415→  async function enableLiveMedia() {
