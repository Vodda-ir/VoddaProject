    60→    60→function formatDuration(ms: number, lang: "fa" | "en"): string {

    68→    68→  if (typeof MediaRecorder === "undefined" || !MediaRecorder.isTypeSupported) return "";

    81→    81→async function compressImageFile(file: File): Promise<File> {
    82→    82→  if (!file.type.startsWith("image/") || file.size < 900_000) return file;
    83→    83→  try {
    84→    84→    const bitmap = await createImageBitmap(file);
    85→    85→    const maxSide = 1600;
    86→    86→    const scale = Math.min(1, maxSide / Math.max(bitmap.width, bitmap.height));
    87→    87→    const canvas = document.createElement("canvas");
    88→    88→    canvas.width = Math.round(bitmap.width * scale);
    89→    89→    canvas.height = Math.round(bitmap.height * scale);
    90→    90→    const ctx = canvas.getContext("2d");
    91→    91→    if (!ctx) return file;
    92→    92→    ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
    93→    93→    const blob = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.85));
    94→    94→    if (!blob) return file;
    95→    95→    return new File([blob], `photo_${Date.now()}.jpg`, { type: "image/jpeg" });
    96→    96→  } catch {
    97→    97→    return file;
    98→    98→  }
    99→    99→}

   395→   395→  async function startRecording() {
   396→   396→    setError("");
   397→   397→    try {
   398→   398→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   399→   399→      const mime = pickRecorderMime();
   400→   400→      recordMimeRef.current = mime;
   401→   401→      const recorder = new MediaRecorder(stream, mime ? { mimeType: mime } : undefined);
   402→   402→      recordChunksRef.current = [];
   403→   403→      recorder.ondataavailable = (event) => {
   404→   404→        if (event.data.size > 0) recordChunksRef.current.push(event.data);
   405→   405→      };
   406→   406→      recorder.onstop = () => {
   407→   407→        stream.getTracks().forEach((track) => track.stop());
   408→   408→        if (recordTimerRef.current !== null) {
   409→   409→          window.clearInterval(recordTimerRef.current);
   410→   410→          recordTimerRef.current = null;
   411→   411→        }
   412→   412→        const seconds = recordSecondsRef.current;
   413→   413→        setRecording(false);
   414→   414→        setRecordSeconds(0);
   415→   415→        if (recordChunksRef.current.length === 0) return;
   416→   416→        const blob = new Blob(recordChunksRef.current, { type: mime || "audio/webm" });
   417→   417→        setVoicePreview({
   418→   418→          blob,
   419→   419→          url: URL.createObjectURL(blob),
   420→   420→          ms: Math.max(600, seconds * 1000),
   421→   421→          mime: mime || "audio/webm",
   422→   422→        });
   423→   423→      };
   424→   424→      recorderRef.current = recorder;
   425→   425→      recordSecondsRef.current = 0;
   426→   426→      setRecordSeconds(0);
   427→   427→      setRecording(true);
   428→   428→      recorder.start(250);
   429→   429→      recordTimerRef.current = window.setInterval(() => {
   430→   430→        recordSecondsRef.current += 1;
   431→   431→        setRecordSeconds(recordSecondsRef.current);
   432→   432→        if (recordSecondsRef.current >= MAX_RECORD_SECONDS) {
   433→   433→          recorderRef.current?.stop();
   434→   434→        }
   435→   435→      }, 1000);
   436→   436→    } catch {
   437→   437→      setError(L("دسترسی به میکروفون ممکن نشد؛ اجازهٔ مرورگر را بررسی کنید."));
   438→   438→    }
   439→   439→  }

   441→   441→  function stopRecording(send: boolean) {
   442→   442→    const recorder = recorderRef.current;
