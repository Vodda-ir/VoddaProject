    48→    48→function timeLabelOf(call: { flexible: boolean; scheduled_at?: string | null }): string {
    49→    49→  if (call.flexible || !call.scheduled_at) return "⚡ در اسرع وقت";
    50→    50→  const at = new Date(call.scheduled_at);
    51→    51→  if (Number.isNaN(at.getTime())) return "⚡ در اسرع وقت";
    52→    52→  const now = new Date();
    53→    53→  const diffMin = Math.round((at.getTime() - now.getTime()) / 60000);
    54→    54→  if (diffMin < 60 && diffMin > -60) return `⚡ حدود ${fa(Math.max(0, diffMin))} دقیقه دیگر`;
    55→    55→  return `📅 ${formatJalaliDateTime(call.scheduled_at, { withWeekday: true })}`;
    56→    56→}

    58→    58→function remainingSeconds(expiresAt: string): number {
    59→    59→  const at = new Date(expiresAt).getTime();
    60→    60→  if (Number.isNaN(at)) return 0;
    61→    61→  return Math.max(0, Math.round((at - Date.now()) / 1000));
    62→    62→}

    64→    64→function countdownLabel(total: number): string {
    65→    65→  const m = Math.floor(total / 60);

    67→    67→  return `${toFaDigits(String(m).padStart(2, "0"))}:${toFaDigits(String(s).padStart(2, "0"))}`;
    68→    68→}
