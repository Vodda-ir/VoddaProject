   335→  async function joinClassroom(withMedia: boolean) {
   336→    if (!session) return;

   345→          video: camId
   346→            ? { deviceId: { exact: camId } }
   347→            : { facingMode: facingRef.current, width: { ideal: 1280 } },
   348→          audio: micId ? { deviceId: { exact: micId } } : true,
   349→        });
   350→      } catch {
   351→        setMediaError('ورود با دوربین/میکروفون ناموفق بود؛ بدون رسانه وارد می‌شوید.');
   352→      }
   353→    }
   354→    liveStreamRef.current = stream;
   355→    setPhase('live');
   356→
   357→    const client = new LiveClient({
   358→      sessionId: session.id,
   359→      onEvent: handleEvent,
   360→      onStatus: (value) => setStatus(value),
   361→    });
   362→    clientRef.current = client;
   363→
   364→    const engine = new LiveEngine({
   365→      role,
   366→      onRemoteStream: (remote) => {
   367→        setRemoteActive(remote.getTracks().length > 0);
   368→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   369→      },
   370→      onState: (state) => setConnState(`${state.connection}/${state.ice}`),
   371→      send: (type, payload) => client.send(type, payload),
   372→    });
   373→    engineRef.current = engine;
   374→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   375→    await engine.ensureConnection();
   376→    if (stream) await engine.setLocalMedia(stream);
   377→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   378→    if (engine.isInitiator && peerOnline) void engine.initiate();
   379→
   380→    await client.connect();
   381→  }

   383→  /** تعویض دوربین جلو/عقب — مخصوص موبایل؛ هم در پیش‌نمایش و هم داخل کلاس. */
   384→  async function switchCamera() {
   385→    const next: 'user' | 'environment' = facingRef.current === 'user' ? 'environment' : 'user';
   386→    try {
   387→      const stream = await navigator.mediaDevices.getUserMedia({
   388→        video: { facingMode: next, width: { ideal: 1280 } },
   389→        audio: false,
   390→      });
   391→      const [track] = stream.getVideoTracks();
   392→      if (!track) throw new Error('no track');
   393→      facingRef.current = next;
   394→      setFacing(next);
   395→      if (phase === 'preflight') {
   396→        setCamId('');
   397→        previewStream?.getVideoTracks().forEach((old) => old.stop());
   398→        return;
   399→      }
   400→      const engine = engineRef.current;
   401→      const current = liveStreamRef.current;
   402→      if (engine && current) {
   403→        await engine.replaceVideoTrack(track);
   404→        const oldTracks = current.getVideoTracks();
   405→        current.removeTrack(oldTracks[0]);
   406→        oldTracks[0]?.stop();
   407→        current.addTrack(track);
   408→        if (localVideoRef.current && !sharing) localVideoRef.current.srcObject = current;
   409→      } else {
   410→        track.stop();
   411→      }
   412→    } catch {
   413→      setNotice('تغییر دوربین انجام نشد؛ دوربین دیگری در دسترس نیست.');
   414→      setTimeout(() => setNotice(''), 3500);
   415→    }
   416→  }

   442→  async function toggleShare() {
   443→    const engine = engineRef.current;
   444→    if (!engine) return;
