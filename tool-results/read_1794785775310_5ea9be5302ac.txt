    60→    60→function parseService(value: unknown): TeacherService | null {
    61→    61→  const row = asRecord(value);
    62→    62→  if (!row) return null;
    63→    63→  const id = asInteger(row.id);
    64→    64→  if (id <= 0) return null;
    65→    65→
    66→    66→  return {
    67→    67→    id,
    68→    68→    official_service_id: row.official_service_id == null ? null : asInteger(row.official_service_id) || null,
    69→    69→    title: asString(row.title, "خدمت"),
    70→    70→    duration: asInteger(row.duration, 30),
    71→    71→    price: asInteger(row.price),
    72→    72→    active: row.active === true,
    73→    73→  };
    74→    74→}

   107→   107→async function getCatalog(runtime: PlatformRuntime): Promise<CatalogCategory[]> {
   108→   108→  // فقط خدمات آنلاین — دستهٔ «خدمات حضوری قرآنی» حذف می‌شود تا ثبت خدمت
   109→   109→  // حضوری فقط از مسیر مخصوص خودش («خدمات حضوری» /t/presence با موجودیت و
   110→   110→  // قیمتی مستقل) انجام شود. قبلاً همین دسته این‌جا هم نمایش داده می‌شد و
   111→   111→  // استادها را گیج می‌کرد (دو مسیر برای یک کار).
   112→   112→  const response = await runtime.request<ApiEnvelope<UnknownRecord>>("/catalog/services");
   113→   113→  const payload = asRecord(response.data);
   114→   114→  const categories: CatalogCategory[] = [];
   115→   115→  for (const group of asArray(payload?.categories)) {
   116→   116→    const row = asRecord(group);
   117→   117→    if (!row) continue;
   118→   118→    const category = asString(row.category, "خدمات رسمی");
   119→   119→    if (category === IN_PERSON_CATEGORY) continue;
   120→   120→    const services: CatalogEntry[] = [];
   121→   121→    for (const item of asArray(row.services)) {
   122→   122→      const entry = asRecord(item);
   123→   123→      if (!entry) continue;
   124→   124→      const id = asInteger(entry.id);
   125→   125→      if (id <= 0) continue;
   126→   126→      services.push({
   127→   127→        id,
   128→   128→        category,
   129→   129→        title: asString(entry.title, "خدمت رسمی"),
   130→   130→        description: asString(entry.description, ""),
   131→   131→        icon: entry.icon == null ? undefined : asString(entry.icon, undefined as unknown as string),
   132→   132→      });
   133→   133→    }
   134→   134→    if (services.length > 0) categories.push({ category, services });
   135→   135→  }
   136→   136→  return categories;
   137→   137→}
