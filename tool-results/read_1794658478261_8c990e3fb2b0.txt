   765→   765→function NotificationsBell({
   766→   766→  runtime,

   768→   768→}: {
   769→   769→  runtime: ReturnType<typeof usePlatform>;
   770→   770→  user: AuthUser | null;
   771→   771→}) {
   772→   772→  const [unread, setUnread] = useState(0);
   773→   773→  const [open, setOpen] = useState(false);
   774→   774→  const [items, setItems] = useState<AppNotification[] | null>(null);

   776→   776→  const [marking, setMarking] = useState(false);

   778→   778→  useEffect(() => {
   779→   779→    if (!user) {
   780→   780→      setUnread(0);
   781→   781→      return;

   783→   783→    let active = true;
   784→   784→    getNotifications(runtime)
   785→   785→      .then((list) => {
   786→   786→        if (active) setUnread(list.filter((notification) => !notification.read).length);
   787→   787→      })
   788→   788→      .catch(() => {});
   789→   789→    return () => {
   790→   790→      active = false;
   791→   791→    };

   798→   798→      const list = await getNotifications(runtime);
   799→   799→      setItems(list);
   800→   800→      setUnread(list.filter((notification) => !notification.read).length);
   801→   801→    } catch {
   802→   802→      setItems([]);
   803→   803→    } finally {
   804→   804→      setLoading(false);
   805→   805→    }
   806→   806→  };
   807→   807→

   810→   810→    setMarking(true);
   811→   811→    for (const notification of items.filter((item) => !item.read)) {
   812→   812→      try {
   813→   813→        await markNotificationRead(runtime, notification.id);
   814→   814→      } catch {
   815→   815→        // تلاش برای بقیه ادامه پیدا می‌کند
   816→   816→      }
   817→   817→    }
   818→   818→    setItems(items.map((item) => ({ ...item, read: true })));
   819→   819→    setUnread(0);
   820→   820→    setMarking(false);
   821→   821→  };
   822→   822→
   823→   823→  if (!user) return null;
   824→   824→
   825→   825→  return (
   826→   826→    <>
   827→   827→      <button
   828→   828→        type="button"
   829→   829→        aria-label="اعلان‌ها"
   830→   830→        className="vodda-tappable"
   831→   831→        onClick={openSheet}
   832→   832→        style={{
   833→   833→          position: "relative",
   834→   834→          width: 44,

   836→   836→          display: "grid",
   837→   837→          placeItems: "center",
