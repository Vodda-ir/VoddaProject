    46→// NewSessionUsecase builds a SessionUsecase. The optional FinancialRepository
    47→// preserves compatibility with existing tests while production injects an atomic
    48→// transaction boundary.
    49→func NewSessionUsecase(s domain.SessionRepository, t domain.TeacherRepository, u domain.UserRepository, w domain.WalletRepository, n domain.NotificationRepository, cfg *config.Config, financial ...domain.FinancialRepository) *SessionUsecase {
    50→        var f domain.FinancialRepository

    52→                f = financial[0]
    53→        }
    54→        return &SessionUsecase{sessions: s, teachers: t, users: u, wallet: w, financial: f, notif: n, cfg: cfg}
    55→}

    64→// BookInput is the learner's booking request.
    65→type BookInput struct {
    66→        TeacherID     uint
    67→        ServiceID     uint
    68→        ScheduledAt   time.Time
    69→        Note          string
    70→        PaymentMethod string // "wallet" (default) | "gateway"
    71→        // ReturnPath is the client-chosen callback landing path. Only
    72→        // validated "/app..." paths are stored on the payment order; empty
    73→        // keeps the legacy panel landing.
    74→        ReturnPath string
    75→        // SkipWorkTimeCheck bypasses the teacher's weekly work-time template.
    76→        // The dispatch (فراخوان) confirm path uses it: the teacher already opted
    77→        // in by accepting the call, so the learner's chosen hour stands even if
    78→        // it falls outside the weekly template. Overlap checks still apply.
    79→        SkipWorkTimeCheck bool
    80→        // Instant marks a جلسهٔ فوری — an immediate booking created minutes
    81→        // before start. Requires the teacher's instant switch to be on and
    82→        // ignores the weekly template.
    83→        Instant bool
    84→}

    86→// Book creates a booking: the wallet method charges the learner's wallet up
    87→// front; the gateway method reserves the slot and hands the learner to the
    88→// Saman Sep payment page, activating the session after settlement.
    89→func (uc *SessionUsecase) Book(ctx context.Context, learnerID uint, in BookInput) (*BookResult, error) {
    90→        uc.cancelExpiredAwaiting(ctx)
    91→        uc.expireStaleInstantPending(ctx)
    92→
    93→        gatewayPayment := in.PaymentMethod == "gateway"
    94→        if in.PaymentMethod != "" && in.PaymentMethod != "wallet" && in.PaymentMethod != "gateway" {
    95→                return nil, domain.ErrValidation
    96→        }
    97→        if gatewayPayment && uc.payments == nil {
    98→                return nil, domain.ErrForbidden
    99→        }
   100→
   101→        teacher, err := uc.teachers.ByID(ctx, in.TeacherID)
   102→        if err != nil {
   103→                return nil, err
   104→        }
   105→        if teacher.Status != domain.TeacherApproved || !teacher.Verified {
   106→                return nil, domain.ErrTeacherNotReady
   107→        }
   108→        if !teacher.Available {
   109→                return nil, domain.ErrTeacherOffline
   110→        }
   111→        if in.Instant && !teacher.InstantEnabled {
   112→                return nil, domain.ErrInstantUnavailable
   113→        }
   114→        if teacher.UserID == learnerID {
   115→                return nil, domain.ErrSelfBooking
   116→        }
   117→
   118→        svc, err := uc.teachers.ServiceByID(ctx, in.ServiceID)
   119→        if err != nil || svc.TeacherID != teacher.ID || !svc.Active || svc.Price < 0 || svc.Duration <= 0 || svc.Duration > 240 {
   120→                return nil, domain.ErrValidation
   121→        }
   122→        if len(in.Note) > 500 {
   123→                return nil, domain.ErrValidation
   124→        }
   125→        if svc.Price == 0 && !teacher.FreeEnabled {
   126→                return nil, domain.ErrFreeUnavailable
   127→        }
   128→
   129→        now := time.Now()
   130→        if in.Instant {
   131→                // جلسهٔ فوری — رزرو همین لحظه: پنجرهٔ آغاز کوتاه و بدون قالب هفتگی.
   132→                if in.ScheduledAt.Before(now.Add(-time.Minute)) || in.ScheduledAt.After(now.Add(15*time.Minute)) {
   133→                        return nil, domain.ErrValidation
   134→                }
   135→        } else if in.ScheduledAt.Before(now.Add(5*time.Minute)) || in.ScheduledAt.After(now.AddDate(0, 6, 0)) {
   136→                return nil, domain.ErrValidation
   137→        }
   138→        if !in.SkipWorkTimeCheck && !in.Instant {
   139→                if err := uc.ensureWithinWorkTime(ctx, teacher.ID, in.ScheduledAt, svc.Duration); err != nil {
   140→                        return nil, err
   141→                }
   142→        }
   143→        if err := uc.ensureNoOverlap(ctx, learnerID, teacher.ID, in.ScheduledAt, svc.Duration); err != nil {
   144→                return nil, err
   145→        }
   146→
   147→        learner, err := uc.users.ByID(ctx, learnerID)
   148→        if err != nil {
   149→                return nil, err
   150→        }
   151→        if !learner.Active {
   152→                return nil, domain.ErrForbidden
   153→        }
   154→        if !gatewayPayment && learner.Wallet < svc.Price {
   155→                return nil, domain.ErrInsufficient
   156→        }
   157→
   158→        // Title/body of the teacher notification — the mobile client rings the
   159→        // summon chime for these titles (see CallAlertBar matching).
   160→        notifyTitle := "درخواست جلسه جدید: " + svc.Title
   161→        notifyBody := "یک قرآن‌آموز برای «" + svc.Title + "» درخواست جلسه ثبت کرد؛ از بخش درخواست‌ها تأیید یا رد کنید."
   162→        if in.Instant {
   163→                notifyTitle = "جلسهٔ فوری جدید: " + svc.Title
   164→                notifyBody = "یک قرآن‌آموز همین حالا درخواست جلسهٔ فوری ثبت کرد؛ سریع تأیید کنید تا کلاس آغاز شود."
   165→        } else if svc.Price == 0 {
   166→                notifyTitle = "درخواست جلسهٔ رایگان: " + svc.Title
   167→                notifyBody = "یک قرآن‌آموز جلسهٔ رایگان شما را رزرو کرد؛ از بخش درخواست‌های جلسه تأیید کنید."
   168→        }
   169→
   170→        s := &domain.Session{
   171→                LearnerID:   learnerID,
   172→                TeacherID:   teacher.ID,
   173→                ServiceID:   svc.ID,
   174→                Title:       svc.Title,
   175→                ScheduledAt: in.ScheduledAt.UTC(),
   176→                Duration:    svc.Duration,
   177→                Price:       svc.Price,
   178→                Status:      domain.SessionPending,
   179→                Note:        in.Note,
   180→                Instant:     in.Instant,
   181→        }
   182→
   183→        if gatewayPayment {
   184→                if uc.financial != nil {
   185→                        if err := uc.financial.CreateAwaitingSession(ctx, s); err != nil {
   186→                                return nil, err
   187→                        }
   188→                } else {
   189→                        s.Status = domain.SessionAwaitingPayment
   190→                        if err := uc.sessions.Create(ctx, s); err != nil {
   191→                                return nil, err
   192→                        }
   193→                }
   194→                payment, err := uc.payments.CreateBookingPayment(ctx, learner, s, in.ReturnPath)
   195→                if err != nil {
   196→                        // Free the reserved slot so the learner can retry cleanly.
   197→                        uc.cancelAwaitingSession(ctx, s.ID, "شروع پرداخت ناموفق بود")
   198→                        return nil, err
   199→                }
   200→                return &BookResult{Session: s, Token: payment.Token, ResNum: payment.ResNum, PaidViaWallet: false}, nil
   201→        }
   202→
   203→        // خدمات رایگان — جلسهٔ صفر تومانی: بدون کسر کیف پول، بدون درگاه؛
   204→        // درخواست مستقیم به صف تأیید استاد می‌رود.
   205→        if svc.Price == 0 {
   206→                if uc.financial != nil {
   207→                        if err := uc.financial.CreateFreeSession(ctx, s); err != nil {
   208→                                return nil, err
   209→                        }
   210→                } else {
   211→                        if err := uc.sessions.Create(ctx, s); err != nil {
   212→                                return nil, err
   213→                        }
   214→                }
   215→                uc.notify(ctx, teacher.UserID, notifyTitle, notifyBody)
   216→                return &BookResult{Session: s, PaidViaWallet: false}, nil
   217→        }
   218→
   219→        if uc.financial != nil {
   220→                if err := uc.financial.CreateFundedSession(ctx, s); err != nil {
   221→                        return nil, err
   222→                }
   223→        } else {
   224→                // Compatibility fallback for isolated tests; production always injects the
   225→                // atomic financial repository.
   226→                if err := uc.sessions.Create(ctx, s); err != nil {
   227→                        return nil, err
   228→                }
   229→                if err := uc.users.AddWallet(ctx, learnerID, -svc.Price); err != nil {
   230→                        uc.cancelUnfundedSession(ctx, s)
   231→                        return nil, err
   232→                }
   233→                if err := uc.wallet.AddTx(ctx, &domain.Transaction{
   234→                        UserID: learnerID, Type: domain.TxSession,
   235→                        Title: "رزرو جلسه: " + svc.Title, Amount: -svc.Price, RefID: s.ID,
   236→                }); err != nil {
   237→                        _ = uc.users.AddWallet(ctx, learnerID, svc.Price)
   238→                        uc.cancelUnfundedSession(ctx, s)
   239→                        return nil, err
   240→                }
   241→        }
   242→
   243→        uc.notify(ctx, teacher.UserID, notifyTitle, notifyBody)
   244→        return &BookResult{Session: s, PaidViaWallet: true}, nil
   245→}

   247→// BookingPayment re-initiates the gateway payment for a session that is still
   248→// awaiting payment (e.g. the learner closed the browser mid-checkout).
   249→func (uc *SessionUsecase) BookingPayment(ctx context.Context, learnerID, sessionID uint, returnPath string) (*TopupResult, error) {
   250→        uc.cancelExpiredAwaiting(ctx)
   251→        s, err := uc.sessions.ByID(ctx, sessionID)
   252→        if err != nil {
   253→                return nil, err
   254→        }
   255→        if s.LearnerID != learnerID {
   256→                return nil, domain.ErrForbidden
   257→        }
   258→        if s.Status != domain.SessionAwaitingPayment {
   259→                return nil, domain.ErrConflict
   260→        }
   261→        if uc.payments == nil {
   262→                return nil, domain.ErrForbidden
   263→        }
   264→        learner, err := uc.users.ByID(ctx, learnerID)
   265→        if err != nil {
   266→                return nil, err
   267→        }
   268→        if !learner.Active {
   269→                return nil, domain.ErrForbidden
   270→        }
   271→        _ = uc.financialFailStaleOrders(ctx, s.ID)
   272→        return uc.payments.CreateBookingPayment(ctx, learner, s, returnPath)
   273→}

   275→// Accept confirms a pending session (teacher action).
   276→func (uc *SessionUsecase) Accept(ctx context.Context, userID, sessionID uint) (*domain.Session, error) {
   277→        s, _, err := uc.ownTeacherSession(ctx, userID, sessionID)
   278→        if err != nil {
   279→                return nil, err
   280→        }
   281→        if s.Status != domain.SessionPending {
   282→                return nil, domain.ErrConflict
   283→        }
   284→        // جلسهٔ فوری تا پایان بازهٔ خودش قابل تأیید است (استاد ممکن است با
   285→        // کمی تأخیر ببیند)؛ جلسهٔ معمولی فقط پیش از آغاز رسمی.
   286→        acceptDeadline := s.ScheduledAt
   287→        if s.Instant {
   288→                acceptDeadline = s.ScheduledAt.Add(time.Duration(s.Duration) * time.Minute)
   289→        }
   290→        if time.Now().After(acceptDeadline) {
   291→                return nil, domain.ErrConflict
   292→        }
   293→        s.Status = domain.SessionConfirmed
   294→        if err := uc.sessions.Update(ctx, s); err != nil {
   295→                return nil, err
   296→        }
   297→        uc.notify(ctx, s.LearnerID, "جلسه تأیید شد", "استاد درخواست جلسه‌ی شما را تأیید کرد.")
   298→        return s, nil
   299→}

   301→// Reject declines a pending session and refunds the learner.
   302→func (uc *SessionUsecase) Reject(ctx context.Context, userID, sessionID uint, reason string) (*domain.Session, error) {
   303→        s, _, err := uc.ownTeacherSession(ctx, userID, sessionID)
   304→        if err != nil {
   305→                return nil, err
   306→        }
   307→        if s.Status != domain.SessionPending {
   308→                return nil, domain.ErrConflict
   309→        }
   310→        if len(reason) > 255 {
   311→                return nil, domain.ErrValidation
   312→        }
   313→
   314→        if uc.financial != nil {
   315→                if err := uc.financial.CancelAndRefundSession(ctx, s, []domain.SessionStatus{domain.SessionPending}, reason); err != nil {
   316→                        return nil, err
   317→                }
   318→        } else {
   319→                previousStatus, previousReason := s.Status, s.RejectNote
   320→                s.Status = domain.SessionCanceled
   321→                s.RejectNote = reason
   322→                if err := uc.sessions.Update(ctx, s); err != nil {
   323→                        return nil, err
   324→                }
   325→                if err := uc.refund(ctx, s); err != nil {
   326→                        s.Status, s.RejectNote = previousStatus, previousReason
   327→                        _ = uc.sessions.Update(ctx, s)
   328→                        return nil, err
   329→                }
   330→        }
   331→        uc.notify(ctx, s.LearnerID, "جلسه لغو شد", "استاد درخواست جلسه را رد کرد و مبلغ بازگشت داده شد.")
   332→        return s, nil
   333→}

   335→// Cancel lets the learner cancel a not-yet-held session and refunds them.
   336→func (uc *SessionUsecase) Cancel(ctx context.Context, learnerID, sessionID uint) (*domain.Session, error) {
   337→        s, err := uc.sessions.ByID(ctx, sessionID)
   338→        if err != nil {
   339→                return nil, err
   340→        }
   341→        if s.LearnerID != learnerID {
   342→                return nil, domain.ErrForbidden
   343→        }
   344→        if s.Status != domain.SessionPending && s.Status != domain.SessionConfirmed {
   345→                return nil, domain.ErrConflict
   346→        }
   347→
   348→        if uc.financial != nil {
   349→                if err := uc.financial.CancelAndRefundSession(ctx, s, []domain.SessionStatus{domain.SessionPending, domain.SessionConfirmed}, "لغو توسط قرآن‌آموز"); err != nil {
   350→                        return nil, err
   351→                }
   352→        } else {
   353→                previousStatus := s.Status
   354→                s.Status = domain.SessionCanceled
   355→                if err := uc.sessions.Update(ctx, s); err != nil {
   356→                        return nil, err
   357→                }
   358→                if err := uc.refund(ctx, s); err != nil {
   359→                        s.Status = previousStatus
   360→                        _ = uc.sessions.Update(ctx, s)
   361→                        return nil, err
   362→                }
   363→        }
   364→        if teacher, err := uc.teachers.ByID(ctx, s.TeacherID); err == nil {
   365→                uc.notify(ctx, teacher.UserID, "جلسه لغو شد", "قرآن‌آموز جلسه را لغو کرد.")
   366→        }
   367→        return s, nil

   370→// Complete marks a confirmed session done and settles the teacher's earnings,
   371→// deducting the platform commission.
   372→func (uc *SessionUsecase) Complete(ctx context.Context, userID, sessionID uint) (*domain.Session, error) {
   373→        s, teacher, err := uc.ownTeacherSession(ctx, userID, sessionID)
   374→        if err != nil {
   375→                return nil, err
   376→        }
   377→        if s.Status != domain.SessionConfirmed {
   378→                return nil, domain.ErrConflict
   379→        }
   380→        // پایان جلسه پس از آغاز رسمی مجاز است (جلسه ممکن است زودتر از مدت
   381→        // رزروشده تمام شود)؛ پیش از آغاز رسمی خیر.
   382→        if time.Now().Before(s.ScheduledAt) {
   383→                return nil, domain.ErrConflict
   384→        }
   385→
   386→        commission := s.Price * int64(uc.cfg.PlatformCommissionPercent) / 100
   387→        earning := s.Price - commission
   388→        if earning < 0 {
   389→                return nil, domain.ErrInternal
   390→        }
   391→
   392→        if uc.financial != nil {
   393→                if err := uc.financial.CompleteAndSettleSession(ctx, s, teacher.UserID, earning, commission); err != nil {
   394→                        return nil, err
   395→                }
   396→        } else {
   397→                // Compatibility fallback for isolated tests.
   398→                s.Status = domain.SessionDone
   399→                if err := uc.sessions.Update(ctx, s); err != nil {
   400→                        return nil, err
   401→                }
   402→                if err := uc.users.AddWallet(ctx, teacher.UserID, earning); err != nil {
   403→                        s.Status = domain.SessionConfirmed
   404→                        _ = uc.sessions.Update(ctx, s)
   405→                        return nil, err
   406→                }
   407→                if err := uc.wallet.AddTx(ctx, &domain.Transaction{
   408→                        UserID: teacher.UserID, Type: domain.TxEarning,
   409→                        Title: "درآمد ناخالص جلسه: " + s.Title, Amount: s.Price, RefID: s.ID,
   410→                }); err != nil {
   411→                        _ = uc.users.AddWallet(ctx, teacher.UserID, -earning)
   412→                        s.Status = domain.SessionConfirmed
   413→                        _ = uc.sessions.Update(ctx, s)
   414→                        return nil, err
   415→                }
   416→                _ = uc.wallet.AddTx(ctx, &domain.Transaction{
   417→                        UserID: teacher.UserID, Type: domain.TxCommission,
   418→                        Title: "کارمزد پلتفرم", Amount: -commission, RefID: s.ID,
   419→                })
   420→                teacher.Sessions++
   421→                _ = uc.teachers.Update(ctx, teacher)
   422→        }
   423→        uc.notify(ctx, s.LearnerID, "جلسه برگزار شد", "جلسه‌ی شما به پایان رسید. لطفاً امتیاز دهید.")
   424→        return s, nil
   425→}

   464→// LiveAccess resolves a session for online-classroom participation. The
   465→// learner, the session's teacher, and admins may join; everyone else is
   466→// rejected. The returned role reflects the caller's position in this session.
   467→func (uc *SessionUsecase) LiveAccess(ctx context.Context, userID uint, role domain.Role, sessionID uint) (*domain.Session, domain.Role, error) {
   468→        s, err := uc.sessions.ByID(ctx, sessionID)
   469→        if err != nil {
   470→                return nil, "", err
   471→        }
   472→        if s.LearnerID == userID {
   473→                return s, domain.RoleLearner, nil
   474→        }
   475→        if role == domain.RoleAdmin {
   476→                return s, domain.RoleAdmin, nil
   477→        }
   478→        if teacher, terr := uc.teachers.ByUserID(ctx, userID); terr == nil && teacher != nil && teacher.ID == s.TeacherID {
   479→                return s, domain.RoleTeacher, nil
   480→        }
   481→        return nil, "", domain.ErrForbidden
   482→}

   484→func (uc *SessionUsecase) ensureWithinWorkTime(ctx context.Context, teacherID uint, scheduledAt time.Time, duration int) error {
   485→        location, err := time.LoadLocation(uc.cfg.AppTimezone)
   486→        if err != nil {
   487→                return err
   488→        }
   489→        local := scheduledAt.In(location)
   490→        weekday := (int(local.Weekday()) + 1) % 7 // Go Sunday=0; Vodda Saturday=0.
   491→        startMinute := local.Hour()*60 + local.Minute()
   492→        endMinute := startMinute + duration
   493→
   494→        workTimes, err := uc.teachers.WorkTimes(ctx, teacherID)
   495→        if err != nil {
   496→                return err
   497→        }
   498→        for _, slot := range workTimes {
   499→                if !slot.Active || slot.Weekday != weekday {
   500→                        continue
   501→                }
   502→                from, okFrom := parseClock(slot.FromTime)
   503→                to, okTo := parseClock(slot.ToTime)
   504→                if okFrom && okTo && startMinute >= from && endMinute <= to {
   505→                        return nil
   506→                }
   507→        }
   508→        return domain.ErrValidation
   509→}

   511→func (uc *SessionUsecase) ensureNoOverlap(ctx context.Context, learnerID, teacherID uint, start time.Time, duration int) error {
   512→        end := start.Add(time.Duration(duration) * time.Minute)
   513→        statuses := []domain.SessionStatus{domain.SessionPending, domain.SessionConfirmed, domain.SessionAwaitingPayment}
   514→        teacherSessions, err := uc.sessions.ForTeacher(ctx, teacherID, statuses)
   515→        if err != nil {
   516→                return err
   517→        }
   518→        for _, existing := range teacherSessions {
   519→                if overlaps(start, end, existing.ScheduledAt, existing.ScheduledAt.Add(time.Duration(existing.Duration)*time.Minute)) {
   520→                        return domain.ErrConflict
   521→                }
   522→        }
   523→
   524→        learnerSessions, err := uc.sessions.ForLearner(ctx, learnerID, "upcoming")
   525→        if err != nil {
   526→                return err
   527→        }
   528→        for _, existing := range learnerSessions {
   529→                if existing.Status != domain.SessionPending && existing.Status != domain.SessionConfirmed && existing.Status != domain.SessionAwaitingPayment {
   530→                        continue
   531→                }
   532→                if overlaps(start, end, existing.ScheduledAt, existing.ScheduledAt.Add(time.Duration(existing.Duration)*time.Minute)) {
   533→                        return domain.ErrConflict
   534→                }
   535→        }
   536→        return nil
   537→}

   592→func parseClock(value string) (int, bool) {
   593→        parsed, err := time.Parse("15:04", value)
   594→        if err != nil {
   595→                return 0, false
   596→        }
   597→        return parsed.Hour()*60 + parsed.Minute(), true
   598→}

   600→func overlaps(aStart, aEnd, bStart, bEnd time.Time) bool {
   601→        return aStart.Before(bEnd) && aEnd.After(bStart)
   602→}

   611→// caller is its learner, its teacher, or an admin. A teacher view stamps

   627→        if err != nil {

   641→// SendSessionMessage appends a coordination message and notifies the
   642→// counterpart (best effort) so neither side misses it.
   643→func (uc *SessionUsecase) SendSessionMessage(ctx context.Context, callerID uint, callerRole domain.Role, sessionID uint, body string) (*domain.SessionMessage, error) {
   644→        body = strings.TrimSpace(body)
   645→        if len(body) == 0 || len(body) > 1000 {
   646→                return nil, domain.ErrValidation
   647→        }
   648→        s, role, err := uc.LiveAccess(ctx, callerID, callerRole, sessionID)
   649→        if err != nil {
   650→                return nil, err
   651→        }
   652→        if s.Status == domain.SessionCanceled {
   653→                return nil, domain.ErrConflict
   654→        }
   655→        message := &domain.SessionMessage{SessionID: s.ID, SenderID: callerID, Body: body}
   656→        if err := uc.sessions.CreateSessionMessage(ctx, message); err != nil {
   657→                return nil, err
   658→        }
   659→        if role == domain.RoleLearner {
   660→                if teacher, terr := uc.teachers.ByID(ctx, s.TeacherID); terr == nil {
   661→                        uc.notify(ctx, teacher.UserID, "پیام جدید در جلسه", "قرآن‌آموز برای جلسه «"+s.Title+"» پیامی فرستاد.")
   662→                }
   663→        } else if role == domain.RoleTeacher {
   664→                uc.notify(ctx, s.LearnerID, "پیام جدید از استاد", "استاد برای جلسه «"+s.Title+"» پیامی فرستاد.")
   665→        }
   666→        return message, nil
   667→}

   696→func (uc *SessionUsecase) expireStaleInstantPending(ctx context.Context) {
   697→        stale, err := uc.sessions.StaleInstantPendings(ctx, time.Now())
   698→        if err != nil {
   699→                log.Printf("session: stale instant sweep: %v", err)
   700→                return
   701→        }
   702→        for _, s := range stale {
   703→                if uc.financial != nil {
   704→                        if err := uc.financial.CancelAndRefundSession(ctx, &s, []domain.SessionStatus{domain.SessionPending}, "انقضای جلسهٔ فوری"); err != nil {
   705→                                log.Printf("session: refund stale instant #%d: %v", s.ID, err)
   706→                                continue
   707→                        }
   708→                } else {
   709→                        s.Status = domain.SessionCanceled
   710→                        if err := uc.sessions.Update(ctx, &s); err != nil {
   711→                                continue
   712→                        }
   713→                        _ = uc.refund(ctx, &s)
   714→                }
   715→                uc.notify(ctx, s.LearnerID, "جلسهٔ فوری تأیید نشد", "استاد در بازهٔ جلسهٔ فوری پاسخ نداد و مبلغ به کیف پول شما بازگشت.")
   716→                if teacher, terr := uc.teachers.ByID(ctx, s.TeacherID); terr == nil {
   717→                        uc.notify(ctx, teacher.UserID, "جلسهٔ فوری منقضی شد", "به درخواست جلسهٔ فوری پاسخ ندادید؛ مبلغ به قرآن‌آموز بازگشت.")
   718→                }
   719→        }
   720→}
