    48→const numberFormatter = new Intl.NumberFormat("fa-IR");

    50→function fa(n: number): string {
    51→  return numberFormatter.format(n);

    54→function priceLabel(n: number): string {
    55→  return n === 0 ? "رایگان" : `${formatWalletAmount(Math.max(0, n))} تومان`;
    56→}

    58→function timeLabelOf(call: { flexible: boolean; scheduled_at?: string | null }): string {
    59→  if (call.flexible || !call.scheduled_at) return "جلسهٔ فوری — همین حالا";
    60→  const at = new Date(call.scheduled_at);
    61→  if (Number.isNaN(at.getTime())) return "در اسرع وقت";
    62→  const now = new Date();
    63→  const diffMin = Math.round((at.getTime() - now.getTime()) / 60000);
    64→  if (diffMin < 60 && diffMin > -60) return `حدود ${fa(Math.max(0, diffMin))} دقیقهٔ دیگر`;
    65→  return formatJalaliDateTime(call.scheduled_at, { withWeekday: true });
    66→}
    67→
    68→function remainingSeconds(expiresAt: string): number {
    69→  const at = new Date(expiresAt).getTime();
    70→  if (Number.isNaN(at)) return 0;
    71→  return Math.max(0, Math.round((at - Date.now()) / 1000));
    72→}
    73→
    74→function countdownLabel(total: number): string {
    75→  const m = Math.floor(total / 60);
    76→  const s = total % 60;
    77→  return `${toFaDigits(String(m).padStart(2, "0"))}:${toFaDigits(String(s).padStart(2, "0"))}`;
    78→}

   196→export function LearnerDispatchScreen({ config }: { config: UniversalAppConfig }) {
   197→  const runtime = usePlatform();
   198→  const [catalog, setCatalog] = useState<OfficialService[]>([]);
   199→  const [catalogLoading, setCatalogLoading] = useState(true);
   200→  const [catalogError, setCatalogError] = useState("");
   201→  const [query, setQuery] = useState("");
   202→  const [selectedTopic, setSelectedTopic] = useState<OfficialService | null>(null);
   203→  const [note, setNote] = useState("");
   204→  const [whenMode, setWhenMode] = useState<"now" | "later">("now");
   205→  const [when, setWhen] = useState<WhenValue>(() => {
   206→    const nowHour = new Date().getHours();
   207→    return { localDate: upcomingJalaliDays(1)[0]?.localDate ?? "", hh: Math.min(Math.max(nowHour + 2, 8), 22), mm: 0 };
   208→  });
   209→  const [creating, setCreating] = useState(false);
   210→  const [actionError, setActionError] = useState("");
   211→  const [view, setView] = useState<DispatchCallView | null>(null);
   212→  const [busyOffer, setBusyOffer] = useState(0);
   213→  const [canceling, setCanceling] = useState(false);
   214→  const [resuming, setResuming] = useState(true);
   215→  const [payMethod, setPayMethod] = useState<"wallet" | "gateway">("gateway");
   216→  const [tick, setTick] = useState(0);
   217→  const pollRef = useRef<number | null>(null);
   218→  /** فراخوانی که کاربر با «فراخوان جدید» از آن عبور کرده — تا کارت موفقیت
   219→   *  دوباره روی آهنگ‌ساز تحمیل نشود و بتوان فراخوان تازه ساخت. */
   220→  const dismissedCallIdRef = useRef(0);
   221→
   222→  const call = view?.call ?? null;
   223→  const mode: ScreenMode = call ? (call.status === "searching" ? "searching" : "matched") : "idle";
   224→
   225→  const loadCatalog = useCallback(async () => {
   226→    setCatalogLoading(true);
   227→    setCatalogError("");
   228→    try {
   229→      setCatalog(await getCatalog(runtime));
   230→    } catch (error) {
   231→      setCatalogError(messageOf(error, "دریافت فهرست موضوعات ناموفق بود."));
   232→    } finally {
   233→      setCatalogLoading(false);
   234→    }
   235→  }, [runtime]);
   236→
   237→  const refresh = useCallback(async () => {
   238→    try {
   239→      const current = await getCurrentDispatchCall(runtime);
   240→      const dismissed = dismissedCallIdRef.current;
   241→      if (current.call && dismissed > 0 && current.call.id === dismissed && current.call.status !== "searching") {
   242→        // کاربر آگاهانه «فراخوان جدید» را زده — آهنگ‌ساز باید باز بماند
   243→        setView({ call: null, candidates: [] });
   244→      } else {
   245→        setView(current);
   246→      }
   247→    } catch {
   248→      // polling errors are non-fatal
   249→    } finally {
   250→      setResuming(false);
   251→    }
   252→  }, [runtime]);
   253→
   254→  useEffect(() => {
   255→    void loadCatalog();
   256→    void refresh();
   257→  }, [loadCatalog, refresh]);
   258→
   259→  // Live polling while a call is active.
   260→  useEffect(() => {
   261→    if (mode === "idle") return;
   262→    pollRef.current = window.setInterval(() => {
   263→      void refresh();
   264→      setTick((value) => value + 1);
   265→    }, 2500);
   266→    return () => {
   267→      if (pollRef.current) window.clearInterval(pollRef.current);
   268→      pollRef.current = null;
   269→    };
   270→  }, [mode, refresh]);
   271→
   272→  const secondsLeft = useMemo(
   273→    () => (call ? remainingSeconds(call.expires_at) : 0),
   274→    // recompute when the poll tick advances so the countdown stays live
   275→    [call, tick],
   276→  );
   277→
   278→  const filteredCatalog = useMemo(() => {
   279→    const q = query.trim();
   280→    if (!q) return catalog;
   281→    return catalog.filter(
   282→      (item) => item.title.includes(q) || item.category.includes(q) || item.description.includes(q),
   283→    );
   284→  }, [catalog, query]);
   285→
   286→  const grouped = useMemo(() => {
   287→    const groups = new Map<string, OfficialService[]>();
   288→    for (const item of filteredCatalog) {
   289→      const rows = groups.get(item.category) ?? [];
   290→      rows.push(item);
   291→      groups.set(item.category, rows);
   292→    }
   293→    return [...groups.entries()];
   294→  }, [filteredCatalog]);
   295→
   296→  /* گزینه‌های شمسی «زمان مشخص» — هم‌سو با فرم مراسم (A3/B1) */
   297→  const dayOptions = useMemo(
   298→    () =>
   299→      upcomingJalaliDays(7).map((day, index) => ({
   300→        value: day.localDate,
   301→        label: `${index === 0 ? "امروز" : index === 1 ? "فردا" : day.weekday} — ${day.dayLabel}`,
   302→      })),
   303→    [],
   304→  );
   305→
   306→  const timeOptions = useMemo(() => {
   307→    const options: Array<{ value: string; label: string }> = [];
   308→    for (let hh = 8; hh <= 23; hh += 1) {
   309→      for (const mm of [0, 30]) {
   310→        options.push({
   311→          value: `${hh}:${mm}`,
   312→          label: `${toFaDigits(String(hh).padStart(2, "0"))}:${toFaDigits(String(mm).padStart(2, "0"))}`,
   313→        });
   314→      }
   315→    }
   316→    return options;
   317→  }, []);
   318→
   319→  const start = useCallback(async () => {
   320→    if (!selectedTopic || creating) return;
   321→    if (whenMode === "later" && !isWhenInFuture(when)) {
   322→      setActionError("زمان انتخاب‌شده باید حداقل ۱۵ دقیقه در آینده باشد.");
   323→      return;
   324→    }
   325→    setCreating(true);
   326→    setActionError("");
   327→    try {
   328→      const scheduledAt = whenMode === "later" ? tehranWallToRfc3339(when) || null : null;
   329→      const created = await createDispatchCall(runtime, {
   330→        officialServiceId: selectedTopic.id,
   331→        note,
   332→        scheduledAt,
   333→        flexible: whenMode === "now",
   334→      });
   335→      showToast("فراخوان شما برای اساتید این موضوع ارسال شد");
   336→      setView(created);
   337→    } catch (error) {
   338→      setActionError(messageOf(error, "ثبت فراخوان انجام نشد."));
   339→    } finally {
   340→      setCreating(false);
   341→    }
   342→  }, [creating, note, runtime, selectedTopic, when, whenMode]);
   343→
   344→  const cancel = useCallback(async () => {
   345→    if (!call || canceling) return;
   346→    setCanceling(true);
   347→    setActionError("");
   348→    try {
   349→      await cancelDispatchCall(runtime, call.id);
   350→      setView({ call: null, candidates: [] });
   351→    } catch (error) {
   352→      setActionError(messageOf(error, "لغو فراخوان انجام نشد."));
   353→    } finally {
   354→      setCanceling(false);
   355→    }
   356→  }, [call, canceling, runtime]);
   357→
   358→  const choose = useCallback(
   359→    async (offerId: number) => {
   360→      if (!call || busyOffer) return;
   361→      setBusyOffer(offerId);
   362→      setActionError("");
   363→      try {
   364→        // پاسخ رایگان (قیمت صفر) بدون انتخاب روش پرداخت رزرو می‌شود.
   365→        const offer = (view?.candidates ?? []).find((item) => item.offer_id === offerId);
   366→        const method = offer && offer.price === 0 ? "" : payMethod;
   367→        const result = await chooseDispatchOffer(runtime, call.id, offerId, method);
   368→        await refresh();
   369→        if (result.token) {
   370→          // پست فرم توکن به OnlinePG از مرورگر کاربر — دقیقاً مثل منافسه، در همان تب
   371→          payViaSamanGate(result.token);
   372→        }
   373→      } catch (error) {
   374→        if (payMethod === "wallet" && error instanceof PlatformHttpError && error.status === 400) {
   375→          setActionError("کیف پول شما کافی نیست. ابتدا کیف پول را شارژ کنید یا پرداخت آنلاین را برگزینید.");
   376→        } else {
   377→          setActionError(messageOf(error, "ثبت رزرو انجام نشد."));
   378→        }
   379→        await refresh();
   380→      } finally {
   381→        setBusyOffer(0);
   382→      }
   383→    },
   384→    [busyOffer, call, payMethod, refresh, runtime, view],
   385→  );
   386→
   387→  const decline = useCallback(
   388→    async (offerId: number) => {
   389→      if (!call || busyOffer) return;
   390→      setBusyOffer(offerId);
   391→      setActionError("");
   392→      try {
   393→        const updated = await declineDispatchOffer(runtime, call.id, offerId);
   394→        setView(updated);
   395→      } catch (error) {
   396→        setActionError(messageOf(error, "رد پاسخ انجام نشد."));
   397→      } finally {
   398→        setBusyOffer(0);
   399→      }
   400→    },
   401→    [busyOffer, call, runtime],
   402→  );
   403→

   411→            aria-hidden="true"
   412→            style={{
   413→              width: 76, height: 76, display: "grid", placeItems: "center", marginInline: "auto",
   414→              borderRadius: "50%", color: colors.surface, background: gradients.hero, boxShadow: shadows.olive,
   415→            }}
   416→          >
   417→            <Icon name="checkCircle" size={36} stroke={2.2} />
   418→          </div>
   419→          <div>
   420→            <h2 style={{ ...typography.title, margin: 0 }}>جلسه شما رزرو شد 🎉</h2>
   421→            <p style={{ ...typography.body, color: colors.inkSecondary, marginBottom: 0, marginTop: 6 }}>
   422→              موضوع «{call.topic_title}» با استاد {candidate?.name ?? "انتخاب‌شده"} ثبت شد و در انتظار تأیید نهایی استاد است.
   423→            </p>
   424→          </div>
   425→          {candidate ? (
   426→            <div style={{ display: "flex", alignItems: "center", gap: 12, textAlign: "right", background: colors.sand100, borderRadius: 16, padding: 12 }}>
   427→              <Avatar name={candidate.name} src={candidate.avatar_url || undefined} size={48} ring={false} />
   428→              <div style={{ minWidth: 0, flex: 1 }}>
   429→                <strong style={{ display: "block", fontSize: 14.5, color: colors.ink }}>{candidate.name}</strong>
   430→                <span style={{ fontSize: 12, color: colors.inkMuted }}>
   431→                  {priceLabel(candidate.price)} · {timeLabelOf(call)}
   432→                </span>
   433→              </div>
   434→            </div>
   435→          ) : null}
   436→          <Button fullWidth onClick={() => navigateApp("/sessions", config)}>پیگیری در جلسات من</Button>
   437→          <Button
   438→            variant="secondary"
   439→            fullWidth
   440→            onClick={() => {
   441→              if (call) dismissedCallIdRef.current = call.id;
   442→              setView({ call: null, candidates: [] });
   443→              void refresh();
   444→            }}
   445→          >
   446→            فراخوان جدید
   447→          </Button>
   448→        </Card>
   449→      </div>
   450→    );
   451→  }
   452→
   453→  // ── live searching state ──
   454→  if (mode === "searching" && call) {
   455→    return (
   456→      <div className="vodda-screen" style={{ display: "grid", gap: spacing.lg }}>
   457→        <div
   458→          className="vodda-enter"
   459→          style={{
   460→            position: "relative",
   461→            overflow: "hidden",
   462→            borderRadius: 22,
   463→            padding: 20,
   464→            color: colors.surface,
   465→            background: gradients.deep,
   466→            boxShadow: shadows.olive,
   467→          }}
   468→        >
   469→          <span aria-hidden="true" className="vodda-dispatch-pulse" style={{ position: "absolute", insetInlineStart: 18, top: 18 }} />
   470→          <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
   471→            <div style={{ display: "grid", placeItems: "center", width: 46, height: 46, borderRadius: 15, background: "rgba(255,255,255,.14)" }}>
   472→              <TopicIcon icon={call.topic_icon} size={24} />
   473→            </div>
   474→            <div style={{ minWidth: 0 }}>
   475→              <strong style={{ display: "block", fontSize: 16.5, fontWeight: 800 }}>{call.topic_title}</strong>
   476→              <span style={{ fontSize: 12.5, opacity: 0.85 }}>{timeLabelOf(call)}</span>
   477→            </div>
   478→            <div style={{ marginInlineStart: "auto", textAlign: "center" }}>
   479→              <strong dir="ltr" style={{ display: "block", fontSize: 19, fontVariantNumeric: "tabular-nums" }}>

   481→              </strong>
   482→              <span style={{ fontSize: 11.5, opacity: 0.78 }}>مهلت فراخوان</span>

   484→          </div>
