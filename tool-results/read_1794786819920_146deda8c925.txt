   119→/** وزن رتبهٔ استاد — بین‌المللی (۳) > کشوری (۲) > استانی (۱). */
   120→const TIER_WEIGHT: Record<string, number> = { international: 3, grade2: 2, grade3: 1 };
   121→
   122→/**
   123→ * امتیاز عملکرد کلی استاد — ترکیب رتبه + عملکرد + فعالیت:
   124→ *   رتبه (تیک): بین‌المللی/کشوری/استانی
   125→ *   عملکرد: امتیاز رضایت و تعداد نظرها
   126→ *   فعالیت: تعداد جلسات موفق + آمادهٔ رزرو بودن امروز
   127→ */
   128→function performanceScore(teacher: Teacher) {
   129→  const tier = TIER_WEIGHT[teacher.tier] ?? 0;
   130→  const rating = Math.max(0, teacher.rating);
   131→  const reviews = Math.log10(10 + Math.max(0, teacher.reviews));
   132→  const sessions = Math.log10(10 + Math.max(0, teacher.sessions));
   133→  const availability = teacher.available ? 0.6 : 0;
   134→  return tier * 2.4 + rating * 1.3 + sessions * 1.9 + reviews * 0.9 + availability;
   135→}

   137→const SPOTLIGHT_POOL_SIZE = 24;
   138→const SPOTLIGHT_COUNT = 7;
   139→
   140→/** مولد شبه‌تصادفی با seed — چیدمان هر بازدید متفاوت اما تابع خالص (بدون خطای hydration). */
   141→function mulberry32(seed: number) {
   142→  let state = seed >>> 0;
   143→  return () => {
   144→    state = (state + 0x6d2b79f5) | 0;
   145→    let t = Math.imul(state ^ (state >>> 15), 1 | state);
   146→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   147→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   148→  };
   149→}
   150→
   151→/**
   152→ * انتخاب ۷ استاد صفحهٔ نخست: از میان ۲۴ استادِ برتر (بر اساس عملکرد، رتبه و فعالیت)،
   153→ * هر بار ۷ نفر به‌صورت رندوم؛ ترتیب نمایش هم بر اساس همان امتیاز.
   154→ */
   155→function pickSpotlight(teachers: Teacher[], seed: number): Teacher[] {
   156→  const pool = [...teachers]
   157→    .sort((a, b) => performanceScore(b) - performanceScore(a))
   158→    .slice(0, SPOTLIGHT_POOL_SIZE);
   159→  const random = mulberry32(seed);
   160→  for (let i = pool.length - 1; i > 0; i -= 1) {
   161→    const j = Math.floor(random() * (i + 1));
   162→    [pool[i], pool[j]] = [pool[j], pool[i]];
   163→  }
   164→  return pool.slice(0, SPOTLIGHT_COUNT).sort((a, b) => performanceScore(b) - performanceScore(a));
   165→}
