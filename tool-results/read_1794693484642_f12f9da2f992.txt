    97→function recommendationScore(teacher: Teacher) {
    98→  return teacher.rating * Math.log10(10 + Math.max(0, teacher.sessions));

   101→function sortTeachers(rows: Teacher[], sort: SortKey): Teacher[] {
   102→  const copy = [...rows];
   103→  switch (sort) {
   104→    case "rating":
   105→      return copy.sort((a, b) => b.rating - a.rating);
   106→    case "sessions":
   107→      return copy.sort((a, b) => b.sessions - a.sessions);
   108→    case "newest":
   109→      return copy.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
   110→    default:
   111→      return copy.sort((a, b) => recommendationScore(b) - recommendationScore(a));
   112→  }
   113→}

   500→    if (filters.availableOnly) params.set("available", "true");
   501→    if (filters.sort !== "recommended") params.set("sort", filters.sort);
   502→    const next = window.location.pathname + (params.size ? "?" + params.toString() : "");
   503→    window.history.replaceState({}, "", next);
   504→  }, [filters, mode, topicId, freeOnly, servicesQuery]);
