    27→function countdown(total: number) {
    28→  const m = Math.floor(total / 60);
    29→  const s = total % 60;
    30→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    31→}

    33→function faNumber(n: number) {

    37→function remaining(expiresAt: string) {
    38→  const at = new Date(expiresAt).getTime();
    39→  if (Number.isNaN(at)) return 0;
    40→  return Math.max(0, Math.round((at - Date.now()) / 1000));
    41→}

    43→function whenLabel(call: DispatchCallItem) {
    44→  if (call.flexible || !call.scheduled_at) return '⚡ در اسرع وقت';
    45→  const at = new Date(call.scheduled_at);
    46→  if (Number.isNaN(at.getTime())) return '⚡ در اسرع وقت';
    47→  return `📅 ${at.toLocaleString('fa-IR', { weekday: 'long', day: 'numeric', month: 'long', hour: '2-digit', minute: '2-digit' })}`;
    48→}
