    47→/** پیام فارسی رده‌بندی‌شده برای خطاهای getUserMedia — هم‌رده با helper مشترک media-permissions. */
    48→function mediaFailureMessage(error: unknown): string {
    49→  const name = error instanceof Error ? error.name : '';
    50→  if (name === 'NotAllowedError' || name === 'SecurityError') return 'دسترسی دوربین/میکروفون مسدود شده است؛ از تنظیمات مرورگر اجازه دهید و صفحه را دوباره باز کنید.';
    51→  if (name === 'NotFoundError' || name === 'OverconstrainedError') return 'دوربین یا میکروفون درخواستی پیدا نشد.';
    52→  if (name === 'NotReadableError' || name === 'TrackStartError') return 'دوربین/میکروفون در برنامهٔ دیگری باز است.';
    53→  return 'گرفتن دسترسی دوربین/میکروفون ناموفق بود؛ دوباره تلاش کنید.';
    54→}

   845→function ControlButton({ active, label, emoji, onClick, danger = false, small = false, badge = 0 }: {
   846→  active: boolean;
   847→  label: string;
   848→  emoji: string;
   849→  onClick: () => void;
   850→  danger?: boolean;
   851→  small?: boolean;

   860→      style={{
   861→        position: 'relative',
   862→        display: 'inline-flex',
   863→        alignItems: 'center',
   864→        gap: 7,
   865→        padding: small ? '9px 12px' : '11px 16px',

   869→        color: danger ? '#fff' : active ? '#fff' : 'var(--ink-2)',
   870→        fontSize: small ? 15 : 13,
   871→        fontWeight: 700,
   872→        cursor: 'pointer',
   873→        boxShadow: active && !danger ? 'var(--sh-olive)' : 'var(--sh-sm)',
   874→      }}
   875→    >
   876→      <span>{emoji}</span>
   877→      {!small && <span>{label}</span>}
   878→      {badge > 0 && <span style={{ position: 'absolute', top: -6, left: -6, minWidth: 19, height: 19, borderRadius: 999, background: 'var(--rose)', color: '#fff', fontSize: 10.5, display: 'grid', placeItems: 'center', padding: '0 5px' }}>{faDigits(badge)}</span>}

   881→}
