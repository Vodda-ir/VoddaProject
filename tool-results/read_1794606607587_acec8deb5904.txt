   209→/** مبلغ با پیشنهادهای آماده — فهرست کشویی انتخاب سریع + ورودی دستی. */
   210→export function AmountPicker({
   211→  label,
   212→  hint,
   213→  value,
   214→  onChange,
   215→  presets,
   216→  ariaLabel,
   217→  emptyHint,
   218→}: {
   219→  label: string;
   220→  hint?: string;
   221→  value: string;
   222→  onChange: (next: string) => void;
   223→  presets: readonly number[];
   224→  ariaLabel: string;
   225→  emptyHint?: string;
   226→}) {
   227→  const numeric = Number(digitsOnly(value) || 0);
   228→  const [customMode, setCustomMode] = useState(false);
   229→  const uniquePresets = useMemo(
   230→    () => [...new Set(presets.filter((preset) => preset > 0))].sort((a, b) => a - b),
   231→    [presets],
   232→  );
   233→  const showManual = customMode || (numeric > 0 && !uniquePresets.includes(numeric));
   234→  const selectValue = numeric > 0
   235→    ? uniquePresets.includes(numeric) && !customMode
   236→      ? String(numeric)
   237→      : "__custom__"
   238→    : "";
   239→  return (
   240→    <FieldShell label={label} hint={hint}>
   241→      {uniquePresets.length > 0 ? (
   242→        <SelectField
   243→          ariaLabel={ariaLabel}
   244→          value={selectValue}
   245→          onChange={(next) => {
   246→            if (next === "__custom__") {
   247→              setCustomMode(true);
   248→              return;
   249→            }
   250→            setCustomMode(false);
   251→            onChange(next);
   252→          }}
   253→          options={[
   254→            { value: "", label: "مبلغ توافقی — بدون مبلغ ثابت" },
   255→            ...uniquePresets.map((preset) => ({ value: String(preset), label: formatToman(preset) })),
   256→            { value: "__custom__", label: "مبلغ دلخواه…" },
   257→          ]}
   258→        />
   259→      ) : null}
   260→      {showManual || uniquePresets.length === 0 ? (
   261→        <AmountInput value={value} onChange={onChange} ariaLabel={ariaLabel} />
   262→      ) : null}
   263→      {!value && emptyHint ? (
   264→        <span style={{ display: "block", fontSize: 11, color: colors.inkFaint, lineHeight: 1.8 }}>{emptyHint}</span>
   265→      ) : null}
   266→    </FieldShell>
   267→  );
   268→}

   272→type BudgetTier = { key: string; label: string; sub: string; min: number; max: number };
   273→
   274→const BUDGET_TIERS: readonly BudgetTier[] = [
   275→  { key: "open", label: "توافقی", sub: "بدون کف و سقف بودجه", min: 0, max: 0 },
   276→  { key: "t500", label: "تا ۵۰۰ هزار", sub: "سقف ۵۰۰ هزار تومان", min: 0, max: 500_000 },
   277→  { key: "500-1m", label: "۵۰۰ تا ۱ میلیون", sub: "بازهٔ اقتصادی", min: 500_000, max: 1_000_000 },
   278→  { key: "1-2m", label: "۱ تا ۲ میلیون", sub: "پرترددترین بازه", min: 1_000_000, max: 2_000_000 },
   279→  { key: "2-3m", label: "۲ تا ۳ میلیون", sub: "اجرای ویژه", min: 2_000_000, max: 3_000_000 },
   280→  { key: "3m+", label: "بالای ۳ میلیون", sub: "مجلس بزرگ", min: 3_000_000, max: 0 },
   281→  { key: "custom", label: "بازهٔ دلخواه", sub: "تعیین دقیق کف و سقف", min: -1, max: -1 },
   282→];
   283→
   284→function activeBudgetKey(minValue: number, maxValue: number): string {
   285→  for (const tier of BUDGET_TIERS) {
   286→    if (tier.key === "custom") continue;
   287→    if (tier.min === minValue && tier.max === maxValue) return tier.key;
   288→  }
   289→  return minValue > 0 || maxValue > 0 ? "custom" : "open";
   290→}

   292→/** بازهٔ حق‌الزحمهٔ فراخوان مراسم — فهرست کشویی پیشنهادی + بازهٔ دلخواه. */
   293→export function PriceRangePicker({
   294→  minValue,
   295→  maxValue,
   296→  onChange,
   297→}: {
   298→  minValue: number;
   299→  maxValue: number;
   300→  onChange: (next: { min: string; max: string }) => void;
   301→}) {
   302→  // بازهٔ دلخواه حتی با مقادیر خالی هم باید فعال بماند — پس علاوه بر تطبیق
   303→  // مقادیر، یک وضعیت محلی هم نگه می‌داریم.
   304→  const [customMode, setCustomMode] = useState(false);
   305→  const activeKey = customMode ? "custom" : activeBudgetKey(minValue, maxValue);
   306→  const select = (tier: BudgetTier) => {
   307→    if (tier.key === "custom") {
   308→      setCustomMode(true);
   309→      onChange({ min: minValue > 0 ? String(minValue) : "", max: maxValue > 0 ? String(maxValue) : "" });
   310→      return;
   311→    }
   312→    setCustomMode(false);
   313→    onChange({ min: tier.min > 0 ? String(tier.min) : "", max: tier.max > 0 ? String(tier.max) : "" });
   314→  };
   315→  return (
   316→    <FieldShell label="بودجهٔ پیشنهادی" hint="حق‌الزحمهٔ منصفانه برای قاریان">
   317→      <SelectField
   318→        ariaLabel="بازهٔ بودجهٔ پیشنهادی"
   319→        value={activeKey}
   320→        onChange={(next) => select(BUDGET_TIERS.find((tier) => tier.key === next) ?? BUDGET_TIERS[0])}
   321→        options={BUDGET_TIERS.map((tier) => ({
   322→          value: tier.key,
   323→          label: tier.key === "custom" ? "بازهٔ دلخواه — تعیین دقیق کف و سقف" : `${tier.label} — ${tier.sub}`,
   324→        }))}
   325→      />
   326→
   327→      {activeKey === "custom" ? (
   328→        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: spacing.md, marginTop: spacing.xs }}>
   329→          <AmountInput value={minValue > 0 ? String(minValue) : ""} onChange={(next) => onChange({ min: next, max: String(maxValue || "") })} ariaLabel="حداقل بودجه به تومان" placeholder="حداقل" />
   330→          <AmountInput value={maxValue > 0 ? String(maxValue) : ""} onChange={(next) => onChange({ min: String(minValue || ""), max: next })} ariaLabel="حداکثر بودجه به تومان" placeholder="حداکثر" />
   331→        </div>
   332→      ) : null}
   333→
   334→      <span style={{ display: "block", fontSize: 11.5, color: colors.inkMuted, marginTop: 2 }}>
   335→        بازهٔ اعلام‌شده به قاریان: {formatBudgetRange(minValue, maxValue)}
   336→      </span>
   337→    </FieldShell>
   338→  );
   339→}

   343→export function CountField({
   344→  label,
   345→  hint,
   346→  value,
   347→  onChange,
   348→  presets,
   349→  min,
   350→  max,
   351→  step = 1,
   352→  unit = "نفر",
   353→  ariaLabel,
   354→}: {
   355→  label: string;
   356→  hint?: string;
   357→  value: string;
   358→  onChange: (next: string) => void;
   359→  presets: readonly number[];
   360→  min: number;
   361→  max: number;
   362→  step?: number;
   363→  unit?: string;
   364→  ariaLabel: string;
   365→}) {
   366→  const numeric = Number(digitsOnly(value) || 0);
   367→  const set = (next: number) => onChange(String(Math.min(max, Math.max(min, Math.round(next)))));
   368→  const presetOptions = presets.filter((preset) => preset >= min && preset <= max);
   369→  const isPreset = presetOptions.includes(numeric);
   370→  const stepButtonStyle: CSSProperties = {
   371→    width: 46,
   372→    height: 46,
   373→    display: "grid",
   374→    placeItems: "center",
   375→    flex: "0 0 46px",
   376→    borderRadius: 14,
   377→    border: `1px solid ${colors.sand200}`,
   378→    background: colors.surface,
   379→    color: colors.olive700,
   380→    cursor: "pointer",
   381→    fontFamily: typography.fontFamily,
   382→  };
   383→  return (
   384→    <FieldShell label={label} hint={hint}>
   385→      <SelectField
   386→        ariaLabel={ariaLabel}
   387→        value={numeric > 0 ? String(numeric) : ""}
   388→        onChange={(next) => {
   389→          if (next) set(Number(next));
   390→        }}
   391→        options={[
   392→          ...(numeric > 0 ? [] : [{ value: "", label: "انتخاب کنید…" }]),
   393→          ...presetOptions.map((preset) => ({ value: String(preset), label: `${toFaDigits(preset)} ${unit}` })),
   394→          ...(numeric > 0 && !isPreset ? [{ value: String(numeric), label: `${toFaDigits(numeric)} ${unit} (دلخواه)` }] : []),
   395→        ]}
   396→      />
   397→      <div style={{ display: "flex", alignItems: "center", gap: spacing.md }}>
   398→        <button
   399→          type="button"
   400→          aria-label={`کاهش ${ariaLabel}`}
   401→          onClick={() => set((numeric > 0 ? numeric : min) - step)}
   402→          disabled={numeric <= min}
   403→          style={{ ...stepButtonStyle, opacity: numeric <= min ? 0.45 : 1 }}
   404→        >
   405→          <Icon name="x" size={18} stroke={2.4} />
   406→        </button>
   407→        <strong
   408→          aria-live="polite"
   409→          style={{
   410→            flex: 1,
   411→            textAlign: "center",
   412→            fontSize: 23,
   413→            fontWeight: 900,
   414→            color: colors.olive700,
   415→            fontVariantNumeric: "tabular-nums",
   416→          }}
   417→        >
   418→          {numeric > 0 ? toFaDigits(numeric) : "—"}
   419→          {numeric > 0 && unit ? <span style={{ fontSize: 12, fontWeight: 700, color: colors.inkMuted }}> {unit}</span> : null}
   420→        </strong>
   421→        <button
   422→          type="button"
   423→          aria-label={`افزایش ${ariaLabel}`}
   424→          onClick={() => set((numeric > 0 ? numeric : min - step) + step)}
   425→          disabled={numeric >= max}
   426→          style={{ ...stepButtonStyle, opacity: numeric >= max ? 0.45 : 1 }}
   427→        >
   428→          <Icon name="plus" size={18} stroke={2.4} />
   429→        </button>
   430→      </div>
   431→    </FieldShell>
   432→  );
   433→}

   481→function osmEmbedSrc(lat: number, lon: number): string {
   482→  const dLat = 0.0032;
   483→  const dLon = 0.0042;
   484→  return `https://www.openstreetmap.org/export/embed.html?bbox=${(lon - dLon).toFixed(5)}%2C${(lat - dLat).toFixed(5)}%2C${(lon + dLon).toFixed(5)}%2C${(lat + dLat).toFixed(5)}&layer=mapnik&marker=${lat.toFixed(5)}%2C${lon.toFixed(5)}`;

   491→function getCurrentPosition(): Promise<GeolocationPosition> {
   492→  return new Promise((resolve, reject) => {
   493→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   494→      reject(new Error("unsupported"));

   501→    );
   502→  });

   505→async function reverseGeocodeNominatim(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   506→  const url = `https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${lat}&lon=${lon}&zoom=17&addressdetails=1&accept-language=fa`;
   507→  const response = await fetch(url, { headers: { Accept: "application/json" } });
   508→  if (!response.ok) return null;
   509→  const data = (await response.json()) as Record<string, unknown>;
   510→  const address = (data.address && typeof data.address === "object" ? data.address : {}) as Record<string, unknown>;
   511→  const pick = (key: string): string => (typeof address[key] === "string" ? (address[key] as string).trim() : "");
   512→  const city = pick("city") || pick("town") || pick("village") || pick("municipality") || pick("county") || pick("state");
   513→  const parts = [pick("amenity"), pick("road"), pick("neighbourhood"), pick("quarter"), pick("suburb")]
   514→    .filter((part) => part.length > 0);
   515→  const unique = [...new Set(parts)];
   516→  if (!city && unique.length === 0) return null;
   517→  return { city, address: unique.join("، ") };
   518→}

   520→async function reverseGeocodeFallback(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   521→  const url = `https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${lat}&longitude=${lon}&localityLanguage=fa`;
   522→  const response = await fetch(url);
   523→  if (!response.ok) return null;
   524→  const data = (await response.json()) as Record<string, unknown>;
   525→  const str = (value: unknown): string => (typeof value === "string" ? value.trim() : "");
   526→  const city = str(data.city) || str(data.locality) || str(data.principalSubdivision);
   527→  if (!city) return null;
   528→  const locality = str(data.locality);
   529→  return { city, address: locality && locality !== city ? locality : "" };
   530→}

   532→function geoErrorText(error: unknown): string {
   533→  const code = typeof (error as GeolocationPositionError)?.code === "number" ? (error as GeolocationPositionError).code : 0;

   535→  if (code === 3) return "یافتن موقعیت بیش از حد طول کشید؛ لطفاً دوباره تلاش کنید.";
   536→  return "موقعیت دستگاه در دسترس نیست؛ نشانی را دستی وارد کنید.";

   809→  const toggle = (text: string) => {
   810→    const current = value.trim();

   812→      const next = current
   813→        .replace(`؛ ${text}`, "")
   814→        .replace(`؛ ${text}؛`, "")
   815→        .replace(`${text}؛ `, "")
   816→        .replace(text, "")
   817→        .replace(/؛\s*؛/g, "؛")
   818→        .replace(/^[؛\s]+|[؛\s]+$/g, "")
   819→        .trim();
   820→      onChange(next);
   821→      return;
   822→    }
   823→    onChange(current ? `${current}؛ ${text}` : text);
   824→  };
   825→  return (
   826→    <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   827→      {NOTE_TEMPLATES.map((template) => (
   828→        <Chip key={template.chip} active={value.includes(template.text)} onClick={() => toggle(template.text)}>
   829→          {template.chip}
   830→        </Chip>
   831→      ))}
   832→    </div>
   833→  );
