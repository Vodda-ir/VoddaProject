    59→function tehranHour() {
    60→  try {
    61→    return Number(
    62→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),
    63→    );
    64→  } catch {
    65→    return new Date().getHours();
    66→  }
    67→}

    81→function dayOfYearIndex() {
    82→  const now = new Date();
    83→  const start = new Date(now.getFullYear(), 0, 0);
    84→  return Math.floor((now.getTime() - start.getTime()) / 86_400_000);
    85→}

    87→function tehranDayKey(dateIso: string) {
    88→  if (!dateIso) return "";
    89→  try {
    90→    return new Intl.DateTimeFormat("en-US", {
    91→      timeZone: "Asia/Tehran",
    92→      year: "numeric",
    93→      month: "2-digit",
    94→      day: "2-digit",
    95→    }).format(new Date(dateIso));
    96→  } catch {
    97→    return "";
    98→  }
    99→}

   101→function isTodayInTehran(dateIso: string) {
   102→  return dateIso !== "" && tehranDayKey(dateIso) === tehranDayKey(new Date().toISOString());
   103→}
