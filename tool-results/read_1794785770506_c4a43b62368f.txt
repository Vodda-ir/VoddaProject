    91→function QuickFeatureToggles({ runtime }: { runtime: PlatformRuntime }) {
    92→  const [state, setState] = useState<FeatureSwitches | null>(null);
    93→  const [busyKey, setBusyKey] = useState<string | null>(null);
    94→  const [error, setError] = useState<string | null>(null);
    95→
    96→  useEffect(() => {
    97→    let active = true;
    98→    loadFeatureSwitches(runtime)
    99→      .then((value) => {
   100→        if (active) setState(value);
   101→      })
   102→      .catch(() => {
   103→        if (active) setState(null);
   104→      });
   105→    return () => {
   106→      active = false;
   107→    };
   108→  }, [runtime]);
   109→
   110→  const toggle = async (key: "available" | "instant_enabled" | "free_enabled", next: boolean) => {
   111→    if (state === null || busyKey !== null) return;
   112→    const previous = state;
   113→    setBusyKey(key);
   114→    setError(null);
   115→    setState({ ...state, [key === "available" ? "available" : key === "instant_enabled" ? "instantEnabled" : "freeEnabled"]: next });
   116→    try {
   117→      const response = await runtime.request<ApiEnvelope<unknown>>("/teacher/me", {
   118→        method: "PUT",
   119→        body: JSON.stringify({ [key]: next }),
   120→      });

   122→      // پایگاه‌داده ثبت شده؛ حتی اگر به هر دلیل مقدار دیگری ذخیره شده باشد.
   123→      const row = asRecord(response.data);
   124→      if (row) {
   125→        setState({
   126→          available: row.available !== false,
   127→          instantEnabled: row.instant_enabled === true,
   128→          freeEnabled: row.free_enabled !== false,
   129→        });
   130→      }
   131→    } catch {
   132→      setState(previous);
   133→      setError("تغییر وضعیت انجام نشد؛ دوباره تلاش کنید.");
   134→    } finally {
   135→      setBusyKey(null);
   136→    }
   137→  };

   139→  if (state === null) return null;

   158→        : "خاموش — خدمات رایگان شما مخفی و رزرو رایگان بسته است",
   159→      checked: state.freeEnabled,
   160→      gold: false,

   163→      key: "available" as const,
   164→      icon: "checkCircle" as const,
   165→      title: "پذیرش درخواست جدید",
   166→      hint: state.available

   177→        <span aria-hidden="true" style={{ width: 38, height: 38, display: "grid", placeItems: "center", flex: "0 0 38px", borderRadius: 12, background: "linear-gradient(145deg,#d8b869,#b8923c)", color: "#fff" }}>

   257→/** وضعیت تکمیل پروفایل حرفه‌ای استاد — چک‌لیست راهنمای ورود به کار */
   258→type CompletionState = {
   259→  activeServices: number;
   260→  bioChars: number;
   261→  hasTitle: boolean;
   262→  priceSet: boolean;
   263→  workSlots: number;
   264→};
   265→
   266→async function loadCompletionState(runtime: PlatformRuntime): Promise<CompletionState> {
   267→  const [teacherResponse, workTimesResponse] = await Promise.all([
   268→    runtime.request<ApiEnvelope<unknown>>("/teacher/me"),
   269→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/work-times").catch(() => ({ data: [] as unknown[] })),
   270→  ]);
   271→  const row = asRecord(teacherResponse.data) ?? {};
   272→  const services = asArray(row.services).map(asRecord);
   273→  const workTimes = Array.isArray(workTimesResponse.data) ? workTimesResponse.data : [];
   274→  return {
   275→    activeServices: services.filter((service) => service?.active === true).length,
   276→    bioChars: asString(row.bio).trim().length,
   277→    hasTitle: asString(row.field).trim().length > 0 || asString(row.tag).trim().length > 0,
   278→    priceSet: asNumber(row.price30) > 0 || asNumber(row.price60) > 0,
   279→    workSlots: workTimes.length,
   280→  };
   281→}

   283→function parseTeacherSummary(value: unknown): TeacherSummary | null {
   284→  const user = asRecord(value);
   285→  if (!user) return null;
   286→  const teacher = asRecord(user.teacher);
   287→
   288→  return {
   289→    id: asNumber(user.id),
   290→    name: asString(user.name, "استاد"),
   291→    avatarUrl: asString(user.avatar_url) || undefined,
   292→    wallet: asNumber(user.wallet),
   293→    rating: asNumber(teacher?.rating),
   294→    reviews: asNumber(teacher?.reviews),
   295→    sessions: asNumber(teacher?.sessions),
   296→  };
   297→}
   298→
   299→function parseTeacherSession(value: unknown): TeacherSession | null {
   300→  const session = asRecord(value);
   301→  if (!session) return null;
   302→  const learner = asRecord(session.learner);
   303→
   304→  return {
   305→    id: asNumber(session.id),
   306→    learnerName: asString(learner?.name, "قرآن‌آموز"),
   307→    service: asString(session.service ?? session.title, "جلسه"),
   308→    scheduledAt: asString(session.scheduled_at),
   309→    localDate: asString(session.local_date),
   310→    localStart: asString(session.local_start),
   311→    localEnd: asString(session.local_end),
   312→    status: asString(session.status, "pending"),
   313→  };
   314→}

   316→async function getTeacherDashboardSnapshot(runtime: PlatformRuntime): Promise<DashboardSnapshot> {
   317→  const [meResponse, pendingResponse, upcomingResponse] = await Promise.all([
   318→    runtime.request<ApiEnvelope<unknown>>("/me"),
   319→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=pending"),
   320→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=upcoming"),
   321→  ]);
   322→
   323→  const teacher = parseTeacherSummary(meResponse.data);
   324→  if (!teacher) throw new Error("اطلاعات حساب استاد دریافت نشد.");
   325→
   326→  return {
   327→    teacher,
   328→    pending: (pendingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   329→    upcoming: (upcomingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   330→  };
   331→}

   333→function officialTime(session: TeacherSession): string {
   334→  if (!session.localDate || !session.localStart || !session.localEnd) {
   335→    return "زمان رسمی جلسه دریافت نشد";
   336→  }
   337→  return `${formatJalaliDate(session.localDate)} · ${toFaDigits(session.localStart)} تا ${toFaDigits(session.localEnd)}`;
   338→}
