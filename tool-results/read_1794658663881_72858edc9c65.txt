    59→function parseService(value: unknown): TeacherService | null {
    60→  const row = asRecord(value);
    61→  if (!row) return null;
    62→  const id = asInteger(row.id);
    63→  if (id <= 0) return null;
    64→
    65→  return {
    66→    id,
    67→    official_service_id: row.official_service_id == null ? null : asInteger(row.official_service_id) || null,
    68→    title: asString(row.title, "خدمت"),
    69→    duration: asInteger(row.duration, 30),
    70→    price: asInteger(row.price),
    71→    active: row.active === true,
    72→  };
    73→}
    74→
    75→async function getServices(runtime: PlatformRuntime): Promise<TeacherService[]> {
    76→  const response = await runtime.request<ApiEnvelope<UnknownRecord>>("/teacher/me");
    77→  const payload = asRecord(response.data);
    78→  return asArray(payload?.services).map(parseService).filter((item): item is TeacherService => item !== null);
    79→}

   104→const IN_PERSON_CATEGORY = "خدمات حضوری قرآنی";

   106→async function getCatalog(runtime: PlatformRuntime): Promise<CatalogCategory[]> {
   107→  // فقط خدمات آنلاین — دستهٔ «خدمات حضوری قرآنی» حذف می‌شود تا ثبت خدمت
   108→  // حضوری فقط از مسیر مخصوص خودش («خدمات حضوری» /t/presence با موجودیت و
   109→  // قیمتی مستقل) انجام شود. قبلاً همین دسته این‌جا هم نمایش داده می‌شد و
   110→  // استادها را گیج می‌کرد (دو مسیر برای یک کار).
   111→  const response = await runtime.request<ApiEnvelope<UnknownRecord>>("/catalog/services");
   112→  const payload = asRecord(response.data);
   113→  const categories: CatalogCategory[] = [];
   114→  for (const group of asArray(payload?.categories)) {
   115→    const row = asRecord(group);
   116→    if (!row) continue;
   117→    const category = asString(row.category, "خدمات رسمی");
   118→    if (category === IN_PERSON_CATEGORY) continue;
   119→    const services: CatalogEntry[] = [];
   120→    for (const item of asArray(row.services)) {
   121→      const entry = asRecord(item);
   122→      if (!entry) continue;
   123→      const id = asInteger(entry.id);
   124→      if (id <= 0) continue;
   125→      services.push({
   126→        id,
   127→        category,
   128→        title: asString(entry.title, "خدمت رسمی"),
   129→        description: asString(entry.description, ""),
   130→        icon: entry.icon == null ? undefined : asString(entry.icon, undefined as unknown as string),
   131→      });
   132→    }
   133→    if (services.length > 0) categories.push({ category, services });
   134→  }
   135→  return categories;
   136→}

   196→function normalizePersian(value: string): string {
   197→  return value
   198→    .replace(/[\u200c\u064b-\u0652]/g, "")
   199→    .replace(/[أإآ]/g, "ا")
   200→    .replace(/ة/g, "ه")
   201→    .replace(/ى/g, "ی")
   202→    .replace(/ك/g, "ک")
   203→    .trim()
   204→    .toLowerCase();
   205→}

   235→      setLoading(false);

   269→  const orphanServices = useMemo(() => {

   296→    if (searching) return true;

   299→    return group.services.some((entry) => {
   300→      const own = serviceByOfficialId.get(entry.id);
   301→      return own != null && own.active;

   306→  async function selectService(entry: CatalogEntry) {
   307→    if (creatingId !== null) return;
   308→    const own = serviceByOfficialId.get(entry.id);
   309→    if (own) {
   310→      openEditor(own);
   311→      return;

   315→    setActionError(null);
   316→    try {
   317→      const response = await createService(runtime, {
   318→        official_service_id: entry.id,
   319→        duration: preset.duration,
   320→        price: preset.price,
   321→      });
   322→      const created = parseService((response.data as UnknownRecord) ?? null);
   323→      if (created) {
   324→        setServices((current) => [...current, created]);
   325→      } else {
   326→        await load();
   327→      }
   328→      showToast(`«${entry.title}» به خدمات شما اضافه شد`);

   330→      setActionError(messageOf(mutationError));
   331→    } finally {
   332→      setCreatingId(null);
   333→    }

   342→  function closeEditor() {

   348→  async function saveService() {
   349→    if (!editing || saving) return;
   350→    const duration = Number.parseInt(draft.duration.trim(), 10);
   351→    const price = Number.parseInt(draft.price.trim(), 10);
   352→
   353→    if (!Number.isFinite(duration) || duration < 10 || duration > 240) {
   354→      setFormError("مدت خدمت باید بین ۱۰ تا ۲۴۰ دقیقه باشد.");

   356→    }
   357→    if (!Number.isFinite(price) || price <= 0) {
   358→      setFormError("مبلغ معتبر وارد کنید.");
   359→      return;
   360→    }

   362→    setSaving(true);
   363→    setFormError(null);
   364→    try {
   365→      await updateService(runtime, editing.id, { duration, price });
   366→      setServices((current) =>
   367→        current.map((service) => (service.id === editing.id ? { ...service, duration, price } : service)),

   370→      showToast("تغییرات خدمت ذخیره شد");
