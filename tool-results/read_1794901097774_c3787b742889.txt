    53→function tehranHour() {
    54→  try {
    55→    return Number(
    56→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),
    57→    );
    58→  } catch {
    59→    return new Date().getHours();
