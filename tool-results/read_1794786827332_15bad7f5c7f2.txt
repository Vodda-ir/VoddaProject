    59→function categoryOrderIndex(name: string) {
    60→  const index = CATEGORY_ORDER_HINTS.findIndex((hint) => hint.match.test(name));
    61→  return index === -1 ? CATEGORY_ORDER_HINTS.length : index;
    62→}

    66→function faNumber(value: number, maximumFractionDigits = 0) {
    67→  return new Intl.NumberFormat("fa-IR", { maximumFractionDigits }).format(value);

    70→type CatalogEntry = { id: number; category: string; title: string; description: string };
    71→
    72→async function fetchCatalog(): Promise<CatalogEntry[]> {
    73→  const response = await fetch("/api/backend/catalog/services", { cache: "no-store" });
    74→  if (!response.ok) throw new Error("catalog failed");
    75→  const payload = (await response.json()) as { data?: unknown };
    76→  const data = asRecord(payload.data);
    77→  const entries: CatalogEntry[] = [];
    78→  for (const group of asArray(data?.categories)) {
    79→    const row = asRecord(group);
    80→    const category = String(row?.category ?? "");
    81→    for (const item of asArray(row?.services)) {
    82→      const entry = asRecord(item);
    83→      const id = Number(entry?.id ?? 0);
    84→      if (id > 0 && entry?.title) {
    85→        entries.push({ id, category, title: String(entry.title), description: String(entry.description ?? "") });
    86→      }
    87→    }
    88→  }
    89→  return entries;
    90→}

   472→export type FreeServiceCardData = {
   473→  service_id: number;
   474→  title: string;
   475→  duration: number;
   476→  teacher_id: number;
   477→  teacher_name: string;
   478→  avatar_url: string;

   486→
   487→async function fetchFreeServices(): Promise<FreeServiceCardData[]> {
   488→  const response = await fetch("/api/backend/catalog/free?limit=12", { cache: "no-store" });
   489→  if (!response.ok) throw new Error("free services failed");
   490→  const payload = (await response.json()) as { data?: unknown };
   491→  const rows: FreeServiceCardData[] = [];
   492→  for (const item of asArray(payload.data)) {
   493→    const row = asRecord(item);
   494→    const serviceId = Number(row?.service_id ?? 0);
   495→    const teacherId = Number(row?.teacher_id ?? 0);
   496→    const title = String(row?.title ?? "");
   497→    const teacherName = String(row?.teacher_name ?? "");
   498→    if (serviceId > 0 && teacherId > 0 && title && teacherName) {
   499→      rows.push({
   500→        service_id: serviceId,
   501→        title,
   502→        duration: Number(row?.duration ?? 0),
   503→        teacher_id: teacherId,
   504→        teacher_name: teacherName,
   505→        avatar_url: String(row?.avatar_url ?? ""),
   506→        field: String(row?.field ?? ""),
   507→        tag: String(row?.tag ?? ""),
   508→        city: String(row?.city ?? ""),
   509→        rating: Number(row?.rating ?? 0),
   510→        reviews: Number(row?.reviews ?? 0),
   511→        sessions: Number(row?.sessions ?? 0),
   512→      });
   513→    }
   514→  }
   515→  return rows;
   516→}
