   118→/** وزن رتبهٔ استاد — بین‌المللی (۳) > کشوری (۲) > استانی (۱). */
   119→const TIER_WEIGHT: Record<string, number> = { international: 3, grade2: 2, grade3: 1 };
   120→
   121→/**
   122→ * امتیاز عملکرد کلی استاد — ترکیب رتبه + عملکرد + فعالیت:
   123→ *   رتبه (تیک): بین‌المللی/کشوری/استانی
   124→ *   عملکرد: امتیاز رضایت و تعداد نظرها
   125→ *   فعالیت: تعداد جلسات موفق + آمادهٔ رزرو بودن امروز
   126→ */
   127→function performanceScore(teacher: Teacher) {
   128→  const tier = TIER_WEIGHT[teacher.tier] ?? 0;
   129→  const rating = Math.max(0, teacher.rating);
   130→  const reviews = Math.log10(10 + Math.max(0, teacher.reviews));
   131→  const sessions = Math.log10(10 + Math.max(0, teacher.sessions));
   132→  const availability = teacher.available ? 0.6 : 0;
   133→  return tier * 2.4 + rating * 1.3 + sessions * 1.9 + reviews * 0.9 + availability;
   134→}

   136→const SPOTLIGHT_POOL_SIZE = 24;
   137→const SPOTLIGHT_COUNT = 7;
   138→
   139→/** مولد شبه‌تصادفی با seed — چیدمان هر بازدید متفاوت اما تابع خالص (بدون خطای hydration). */
   140→function mulberry32(seed: number) {
   141→  let state = seed >>> 0;
   142→  return () => {
   143→    state = (state + 0x6d2b79f5) | 0;
   144→    let t = Math.imul(state ^ (state >>> 15), 1 | state);
   145→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   146→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   147→  };
   148→}
   149→
   150→/**
   151→ * انتخاب ۷ استاد صفحهٔ نخست: از میان ۲۴ استادِ برتر (بر اساس عملکرد، رتبه و فعالیت)،
   152→ * هر بار ۷ نفر به‌صورت رندوم؛ ترتیب نمایش هم بر اساس همان امتیاز.
   153→ */
   154→function pickSpotlight(teachers: Teacher[], seed: number): Teacher[] {
   155→  const pool = [...teachers]
   156→    .sort((a, b) => performanceScore(b) - performanceScore(a))
   157→    .slice(0, SPOTLIGHT_POOL_SIZE);
   158→  const random = mulberry32(seed);
   159→  for (let i = pool.length - 1; i > 0; i -= 1) {
   160→    const j = Math.floor(random() * (i + 1));
   161→    [pool[i], pool[j]] = [pool[j], pool[i]];
   162→  }
   163→  return pool.slice(0, SPOTLIGHT_COUNT).sort((a, b) => performanceScore(b) - performanceScore(a));
   164→}

   595→function NotificationsBell({
   596→  runtime,
   597→  user,
   598→}: {
   599→  runtime: ReturnType<typeof usePlatform>;
   600→  user: AuthUser | null;
   601→}) {
   602→  const [unread, setUnread] = useState(0);
   603→  const [open, setOpen] = useState(false);
   604→  const [items, setItems] = useState<AppNotification[] | null>(null);
   605→  const [loading, setLoading] = useState(false);
   606→  const [marking, setMarking] = useState(false);
   607→  const { L, lang } = useAppearance();
   608→
   609→  useEffect(() => {
   610→    if (!user) {
   611→      setUnread(0);
   612→      return;
   613→    }
   614→    let active = true;
   615→    getNotifications(runtime)
   616→      .then((list) => {
   617→        if (active) setUnread(list.filter((notification) => !notification.read).length);
   618→      })
   619→      .catch(() => {});
   620→    return () => {
   621→      active = false;
   622→    };
   623→  }, [runtime, user]);
   624→
   625→  const openSheet = async () => {
   626→    setOpen(true);
   627→    setLoading(true);
   628→    try {
   629→      const list = await getNotifications(runtime);
   630→      setItems(list);
   631→      setUnread(list.filter((notification) => !notification.read).length);
   632→    } catch {
   633→      setItems([]);
   634→    } finally {
   635→      setLoading(false);
   636→    }
   637→  };
   638→
   639→  const markAllRead = async () => {

   641→    setMarking(true);
   642→    for (const notification of items.filter((item) => !item.read)) {
   643→      try {
   644→        await markNotificationRead(runtime, notification.id);
   645→      } catch {
   646→        // تلاش برای بقیه ادامه پیدا می‌کند
   647→      }
   648→    }
   649→    setItems(items.map((item) => ({ ...item, read: true })));
   650→    setUnread(0);
   651→    setMarking(false);
   652→  };
   653→
   654→  if (!user) return null;
   655→
   656→  return (
   657→    <>
   658→      <button
   659→        type="button"
   660→        aria-label={L("اعلان‌ها")}

   665→          width: 44,
   666→          height: 44,
   667→          display: "grid",
   668→          placeItems: "center",
   669→          flex: "0 0 44px",
   670→          padding: 0,

   672→          borderRadius: 16,

   674→          background: colors.olive50,
   675→          cursor: "pointer",
   676→        }}

   678→        <Icon name="bell" size={20} />
   679→        {unread > 0 ? (
   680→          <span
   681→            style={{
   682→              position: "absolute",
   683→              top: -5,
   684→              insetInlineEnd: -5,
   685→              minWidth: 18,

   694→              fontSize: 11,
   695→              fontWeight: 800,
   696→              lineHeight: 1,
   697→            }}
   698→          >
   699→            {unread > 9 ? `${localizedDigits(9, lang)}+` : faNumber(unread, lang, 0)}

   708→            <div style={{ display: "grid", justifyItems: "center", gap: spacing.sm, paddingBlock: spacing.xl, textAlign: "center" }}>
   709→              <IconBadge icon="bell" tone="olive" size={52} />
   710→              <strong style={{ ...typography.body, color: colors.ink }}>{L("اعلانی نداری")}</strong>

   713→              </span>
   714→            </div>
   715→          ) : (
   716→            <>
   717→              <div style={{ display: "grid", gap: 8, maxHeight: 320, overflowY: "auto" }}>
   718→                {items.map((notification) => (
   719→                  <div

   735→                    </span>
   736→                    {notification.body ? (
   737→                      <span style={{ ...typography.caption, color: colors.inkMuted, lineHeight: 1.7 }}>{L(notification.body)}</span>
   738→                    ) : null}
