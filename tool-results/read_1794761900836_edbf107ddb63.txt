   273→   273→function parseTeacherSummary(value: unknown): TeacherSummary | null {

   306→   306→async function getTeacherDashboardSnapshot(runtime: PlatformRuntime): Promise<DashboardSnapshot> {
   307→   307→  const [meResponse, pendingResponse, upcomingResponse] = await Promise.all([
   308→   308→    runtime.request<ApiEnvelope<unknown>>("/me"),
   309→   309→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=pending"),
   310→   310→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=upcoming"),
   311→   311→  ]);
   312→   312→
   313→   313→  const teacher = parseTeacherSummary(meResponse.data);
   314→   314→  if (!teacher) throw new Error("اطلاعات حساب استاد دریافت نشد.");
   315→   315→
   316→   316→  return {
   317→   317→    teacher,
   318→   318→    pending: (pendingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   319→   319→    upcoming: (upcomingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   320→   320→  };
   321→   321→}
