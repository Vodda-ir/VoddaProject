   214→function activeBudgetKey(minValue: number, maxValue: number): string {
   215→  for (const tier of BUDGET_TIERS) {
   216→    if (tier.key === "custom") continue;
   217→    if (tier.min === minValue && tier.max === maxValue) return tier.key;
   218→  }
   219→  return minValue > 0 || maxValue > 0 ? "custom" : "open";
   220→}

   310→export function CountField({
   311→  label,
   312→  hint,
   313→  value,
   314→  onChange,

   317→  max,
   318→  step = 1,

   389→          <Icon name="plus" size={18} stroke={2.4} />

   445→function getCurrentPosition(): Promise<GeolocationPosition> {

   459→async function reverseGeocodeNominatim(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   460→  const url = `https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${lat}&lon=${lon}&zoom=17&addressdetails=1&accept-language=fa`;
   461→  const response = await fetch(url, { headers: { Accept: "application/json" } });
   462→  if (!response.ok) return null;
   463→  const data = (await response.json()) as Record<string, unknown>;
   464→  const address = (data.address && typeof data.address === "object" ? data.address : {}) as Record<string, unknown>;
   465→  const pick = (key: string): string => (typeof address[key] === "string" ? (address[key] as string).trim() : "");
   466→  const city = pick("city") || pick("town") || pick("village") || pick("municipality") || pick("county") || pick("state");
   467→  const parts = [pick("amenity"), pick("road"), pick("neighbourhood"), pick("quarter"), pick("suburb")]
   468→    .filter((part) => part.length > 0);
   469→  const unique = [...new Set(parts)];
   470→  if (!city && unique.length === 0) return null;
   471→  return { city, address: unique.join("، ") };
   472→}

   474→async function reverseGeocodeFallback(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   475→  const url = `https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${lat}&longitude=${lon}&localityLanguage=fa`;
   476→  const response = await fetch(url);
   477→  if (!response.ok) return null;
   478→  const data = (await response.json()) as Record<string, unknown>;
   479→  const str = (value: unknown): string => (typeof value === "string" ? value.trim() : "");
   480→  const city = str(data.city) || str(data.locality) || str(data.principalSubdivision);
   481→  if (!city) return null;
   482→  const locality = str(data.locality);
   483→  return { city, address: locality && locality !== city ? locality : "" };
   484→}
   485→
   486→function geoErrorText(error: unknown): string {
   487→  const code = typeof (error as GeolocationPositionError)?.code === "number" ? (error as GeolocationPositionError).code : 0;
   488→  if (code === 1) return "دسترسی به موقعیت دستگاه داده نشد؛ مجوز موقعیت را در تنظیمات فعال کنید یا نشانی را دستی وارد کنید.";
   489→  if (code === 3) return "یافتن موقعیت بیش از حد طول کشید؛ لطفاً دوباره تلاش کنید.";
   490→  return "موقعیت دستگاه در دسترس نیست؛ نشانی را دستی وارد کنید.";
   491→}
   492→
   493→function stripVenuePrefix(value: string): string {
   494→  let rest = value.trim();
   495→  for (const venueType of VENUE_TYPES) {
   496→    rest = rest.replace(new RegExp(`^${venueType}\\s*[-–—،,]?\\s*`), "");
   497→  }
   498→  return rest.trim();
   499→}

   506→export function VenuePicker({

   516→  venue: string;
   517→  onCityChange: (next: string) => void;
   518→  onVenueChange: (next: string) => void;
   519→  label?: string;
   520→  hint?: string;
   521→  cityLabel?: string;
   522→}) {
   523→  const [geoState, setGeoState] = useState<"idle" | "locating" | "done" | "error">("idle");
   524→  const [geoError, setGeoError] = useState("");
   525→  const [detected, setDetected] = useState<DetectedPlace | null>(null);
   526→  const [showMap, setShowMap] = useState(true);
   527→
   528→  const locate = async () => {
   529→    if (geoState === "locating") return;
   530→    setGeoState("locating");
   531→    setGeoError("");
   532→    try {
   533→      const position = await getCurrentPosition();
   534→      const lat = position.coords.latitude;
   535→      const lon = position.coords.longitude;
   536→      let resolved: { city: string; address: string } | null = null;
   537→      try {
   538→        resolved = await reverseGeocodeNominatim(lat, lon);
   539→      } catch {
   540→        resolved = null;
   541→      }
   542→      if (!resolved) {
   543→        try {
   544→          resolved = await reverseGeocodeFallback(lat, lon);
   545→        } catch {
   546→          resolved = null;
   547→        }
   548→      }
   549→      if (resolved) {
   550→        setDetected({ city: resolved.city, address: resolved.address, lat, lon });
   551→        setShowMap(true);
   552→        setGeoState("done");
   553→        if (resolved.city) onCityChange(resolved.city.slice(0, 60));
   554→        if (resolved.address && !venue.trim()) onVenueChange(resolved.address.slice(0, 200));
   555→      } else {
   556→        setDetected({ city: "", address: "", lat, lon });
   557→        setShowMap(true);
   558→        setGeoState("done");
   559→        setGeoError("موقعیت یافت شد اما نشانی خودکار در دسترس نبود؛ شهر و نشانی را دستی وارد کنید.");
   560→      }
   561→    } catch (caught) {
   562→      setGeoState("error");
   563→      setGeoError(geoErrorText(caught));
   564→    }
   565→  };
   566→
   567→  const detectedAddress = detected?.address ?? "";
   568→  const applyDetected = () => {
   569→    if (!detected) return;
   570→    if (detected.address) onVenueChange(detected.address.slice(0, 200));
   571→    if (detected.city) onCityChange(detected.city.slice(0, 60));
   572→  };
   573→
   574→  const actionCard: CSSProperties = {
   575→    display: "flex",
   576→    alignItems: "center",
   577→    gap: 10,
   578→    padding: "11px 13px",
   579→    borderRadius: 14,
   580→    border: `1px solid ${geoState === "error" ? "rgba(166,81,81,.35)" : colors.goldSoft}`,
   581→    background: geoState === "error" ? colors.dangerBackground : colors.goldBackground,
   582→    color: geoState === "error" ? colors.danger : colors.goldDark,
   583→  };
   584→
   585→  return (
   586→    <div style={{ display: "grid", gap: spacing.lg }}>
   587→      <CityPicker value={city} onChange={onCityChange} label={cityLabel ?? "شهر برگزاری"} hint="اجباری" />
   588→
   589→      <FieldShell label={label} hint={hint}>
   590→        <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   591→          {VENUE_TYPES.map((venueType) => {
   592→            const active = venue.trim().startsWith(venueType);
   593→            return (
   594→              <Chip
   595→                key={venueType}
   596→                active={active}
   597→                onClick={() => onVenueChange(active ? stripVenuePrefix(venue) : `${venueType} ${stripVenuePrefix(venue)}`.trim())}
   598→              >
   599→                {venueType}
   600→              </Chip>
   601→            );
   602→          })}
   603→        </div>
   604→        <input
   605→          type="text"
   606→          value={venue}
   607→          onChange={(event) => onVenueChange(event.target.value.slice(0, 200))}
   608→          placeholder="مثلاً: مسجد جامع، سالن شماره ۲…"
   609→          style={{ ...inputStyle, minHeight: 46 }}
   610→          aria-label={label}
   611→        />
   612→      </FieldShell>
   613→
   614→      <div style={{ display: "grid", gap: spacing.sm }}>
   615→        {geoState === "done" && detected ? (
   616→          <div style={{ ...actionCard, border: `1px solid rgba(94,110,67,.35)`, background: colors.successBackground, color: colors.olive800 }}>
   617→            <Icon name="checkCircle" size={19} style={{ flex: "0 0 auto" }} />
   618→            <div style={{ minWidth: 0, flex: 1 }}>
   619→              <strong style={{ display: "block", fontSize: 12.5, fontWeight: 800 }}>
   620→                {detected.city ? `شهر «${detected.city}» از موقعیت دستگاه ثبت شد` : "موقعیت دستگاه ثبت شد"}
   621→              </strong>
   622→              <span style={{ display: "block", fontSize: 11, opacity: 0.85, lineHeight: 1.7 }}>
   623→                {detected.address ? `نشانی شناسایی‌شده: ${detected.address}` : `مختصات: ${toFaDigits(detected.lat.toFixed(4))} ، ${toFaDigits(detected.lon.toFixed(4))}`}
   624→              </span>
   625→            </div>
   626→            <button
   627→              type="button"
   628→              onClick={() => void locate()}
   629→              aria-label="تازه‌سازی موقعیت"
   630→              style={{ background: "none", border: "none", cursor: "pointer", color: "inherit", padding: 6, fontFamily: typography.fontFamily }}
   631→            >
   632→              <Icon name="refresh" size={17} />
   633→            </button>
   634→          </div>
   635→        ) : null}
   636→
   637→        {detectedAddress && venue.trim() && venue.trim() !== detectedAddress ? (
   638→          <div style={{ display: "flex", alignItems: "center", gap: 8, padding: "9px 12px", borderRadius: 13, background: colors.sand100, border: `1px solid ${colors.sand200}` }}>
   639→            <Icon name="pin" size={15} style={{ color: colors.olive600, flex: "0 0 auto" }} />
   640→            <span style={{ minWidth: 0, flex: 1, fontSize: 11.5, color: colors.inkSecondary, lineHeight: 1.7 }}>
   641→              نشانی شناسایی‌شده: {detectedAddress}
   642→            </span>
   643→            <button
   644→              type="button"
   645→              onClick={applyDetected}
   646→              style={{
   647→                flex: "0 0 auto",
   648→                minHeight: 34,
   649→                padding: "0 12px",
   650→                borderRadius: 999,
   651→                border: `1px solid ${colors.olive200}`,
   652→                background: colors.surface,
   653→                color: colors.olive700,
   654→                fontSize: 11.5,
   655→                fontWeight: 800,
   656→                cursor: "pointer",
   657→                fontFamily: typography.fontFamily,
   658→              }}
   659→            >
   660→              درج در نشانی
   661→            </button>
   662→          </div>
   663→        ) : null}
   664→
   665→        {geoState !== "done" ? (
   666→          <button
   667→            type="button"
   668→            onClick={() => void locate()}
   669→            disabled={geoState === "locating"}
   670→            aria-label="استفاده از موقعیت دستگاه برای پرکردن نشانی"
   671→            style={{ ...actionCard, cursor: geoState === "locating" ? "wait" : "pointer", fontFamily: typography.fontFamily, textAlign: "right" }}
   672→          >
   673→            {geoState === "locating" ? (
   674→              <span aria-hidden="true" className="vodda-spinner" style={{ flex: "0 0 auto" }} />
   675→            ) : (
   676→              <Icon name="pin" size={19} style={{ flex: "0 0 auto" }} />
   677→            )}
   678→            <span style={{ minWidth: 0, flex: 1 }}>
   679→              <strong style={{ display: "block", fontSize: 12.5, fontWeight: 800 }}>
   680→                {geoState === "locating" ? "در حال یافتن موقعیت…" : "استفاده از موقعیت من"}
   681→              </strong>
   682→              <span style={{ display: "block", fontSize: 11, opacity: 0.85, lineHeight: 1.7 }}>
   683→                {geoState === "error" ? geoError : "شهر و نشانی به‌صورت خودکار از GPS پر می‌شود"}
   684→              </span>
   685→            </span>
   686→          </button>
   687→        ) : null}
   688→
   689→        {detected && showMap ? (
   690→          <div style={{ borderRadius: 14, overflow: "hidden", border: `1px solid ${colors.sand200}`, position: "relative" }}>
   691→            <iframe
   692→              title="پیش‌نمایش نقشهٔ موقعیت مراسم"
   693→              src={osmEmbedSrc(detected.lat, detected.lon)}
   694→              loading="lazy"

   696→            />
   697→            <div style={{ display: "flex", alignItems: "center", gap: 8, padding: "8px 12px", background: colors.surface, borderTop: `1px solid ${colors.sand150}` }}>
   698→              <a
   699→                href={osmLink(detected.lat, detected.lon)}
   700→                target="_blank"
   701→                rel="noopener noreferrer"
   702→                style={{ display: "inline-flex", alignItems: "center", gap: 5, fontSize: 11.5, fontWeight: 800, color: colors.olive700, textDecoration: "none" }}
   703→              >
   704→                <Icon name="arrowUpRight" size={13} />
   705→                مشاهده روی نقشه
   706→              </a>
   707→              <span aria-hidden="true" style={{ flex: 1 }} />
   708→              <button
   709→                type="button"
   710→                onClick={() => setShowMap(false)}
   711→                style={{ background: "none", border: "none", cursor: "pointer", color: colors.inkFaint, padding: 4, fontFamily: typography.fontFamily, fontSize: 11 }}
   712→              >
   713→                بستن نقشه
   714→              </button>
   715→            </div>
   716→          </div>
   717→        ) : null}
   718→
   719→        <span style={{ display: "block", fontSize: 10.5, color: colors.inkFaint, lineHeight: 1.8 }}>
   720→          موقعیت دستگاه شما فقط برای پرکردن متن نشانی استفاده می‌شود و تنها به قاریِ انتخاب‌شده نمایش داده می‌شود.
   721→        </span>
   722→      </div>
   723→    </div>
   724→  );
   725→}

   754→
   755→export function NoteTemplateChips({ value, onChange }: { value: string; onChange: (next: string) => void }) {
   756→  const toggle = (text: string) => {
   757→    const current = value.trim();
   758→    if (current.includes(text)) {
   759→      const next = current
   760→        .replace(`؛ ${text}`, "")
   761→        .replace(`؛ ${text}؛`, "")
   762→        .replace(`${text}؛ `, "")
   763→        .replace(text, "")
   764→        .replace(/؛\s*؛/g, "؛")
   765→        .replace(/^[؛\s]+|[؛\s]+$/g, "")
   766→        .trim();
   767→      onChange(next);
   768→      return;
   769→    }
   770→    onChange(current ? `${current}؛ ${text}` : text);
   771→  };
   772→  return (
   773→    <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   774→      {NOTE_TEMPLATES.map((template) => (
   775→        <Chip key={template.chip} active={value.includes(template.text)} onClick={() => toggle(template.text)}>
   776→          {template.chip}
   777→        </Chip>
   778→      ))}
   779→    </div>
   780→  );
   781→}

   810→/* ─────────────────────────  شهرهای پذیرش (چند انتخابی)  ───────────────────────── */
   811→
   812→export function CitiesChipsField({
   813→  value,
   814→  onChange,
   815→  label = "شهرهای پذیرش مراسم",
   816→}: {
   817→  value: string;
   818→  onChange: (next: string) => void;
   819→  label?: string;

   822→    () => value.split(/[،,]/).map((part) => part.trim()).filter(Boolean),

   840→            {cityOption}
   841→          </Chip>
   842→        ))}
   843→      </div>
   844→      <input
   845→        type="text"
   846→        value={value}
