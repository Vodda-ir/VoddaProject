    48→function shortTime(value: string): string {
    49→  const date = new Date(value);
    50→  if (Number.isNaN(date.getTime())) return "";
    51→  return new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit" }).format(date);
    52→}

    54→function shortDate(value: string): string {
    55→  const date = new Date(value);
    56→  if (Number.isNaN(date.getTime())) return "";
    57→  return new Intl.DateTimeFormat("fa-IR", { day: "numeric", month: "short" }).format(date);
    58→}

    60→function formatDuration(ms: number, lang: "fa" | "en"): string {
    61→  const total = Math.max(0, Math.round(ms / 1000));
    62→  const minutes = Math.floor(total / 60);
    63→  const seconds = total % 60;
    64→  return `${localizedDigits(minutes, lang)}:${localizedDigits(String(seconds).padStart(2, "0"), lang)}`;
    65→}

    67→function pickRecorderMime(): string {
    68→  if (typeof MediaRecorder === "undefined" || !MediaRecorder.isTypeSupported) return "";
    69→  const candidates = ["audio/webm;codecs=opus", "audio/webm", "audio/ogg;codecs=opus", "audio/mp4"];
    70→  return candidates.find((mime) => MediaRecorder.isTypeSupported(mime)) ?? "";
    71→}

    73→function mimeToExtension(mime: string): string {
    74→  if (mime.includes("webm")) return ".webm";
    75→  if (mime.includes("ogg")) return ".ogg";
    76→  if (mime.includes("mp4")) return ".m4a";
    77→  return ".bin";
    78→}

    80→/** downscales a picked image so uploads stay small and fast. */
    81→async function compressImageFile(file: File): Promise<File> {
    82→  if (!file.type.startsWith("image/") || file.size < 900_000) return file;
    83→  try {
    84→    const bitmap = await createImageBitmap(file);
    85→    const maxSide = 1600;
    86→    const scale = Math.min(1, maxSide / Math.max(bitmap.width, bitmap.height));
    87→    const canvas = document.createElement("canvas");
    88→    canvas.width = Math.round(bitmap.width * scale);
    89→    canvas.height = Math.round(bitmap.height * scale);
    90→    const ctx = canvas.getContext("2d");
    91→    if (!ctx) return file;
    92→    ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
    93→    const blob = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.85));
    94→    if (!blob) return file;
    95→    return new File([blob], `photo_${Date.now()}.jpg`, { type: "image/jpeg" });
    96→  } catch {
    97→    return file;
    98→  }
    99→}

   101→/** compact voice-note player used inside chat bubbles. */
   102→function AudioBubble({
   103→  url,
   104→  durationMs,
   105→  mine,
   106→}: {
   107→  url: string;
   108→  durationMs: number;
   109→  mine: boolean;
   110→}) {
   111→  const audioRef = useRef<HTMLAudioElement | null>(null);
   112→  const [playing, setPlaying] = useState(false);
   113→  const [progress, setProgress] = useState(0);
   114→  const { L } = useAppearance();
   115→
   116→  useEffect(() => () => audioRef.current?.pause(), []);
   117→
   118→  function toggle() {
   119→    const audio = audioRef.current;
   120→    if (!audio) return;
   121→    if (playing) {
   122→      audio.pause();
   123→    } else {
   124→      void audio.play().catch(() => setPlaying(false));
   125→    }
   126→  }
   127→
   128→  return (
   129→    <div
   130→      style={{
   131→        display: "flex",
   132→        alignItems: "center",
   133→        gap: 9,
   134→        minWidth: 168,
   135→        padding: "7px 10px",
   136→        borderRadius: 999,
   137→        border: `1px solid ${mine ? "rgba(255,250,240,.35)" : colors.olive200}`,
   138→        background: mine ? colors.olive600 : colors.surface,
   139→        color: mine ? colors.onOlive : colors.ink,
   140→      }}
   141→    >
   142→      <button
   143→        type="button"
   144→        onClick={toggle}
   145→        aria-label={playing ? L("توقف پخش پیام صوتی") : L("پخش پیام صوتی")}
   146→        style={{
   147→          width: 34,
   148→          height: 34,
   149→          flex: "0 0 auto",
   150→          display: "grid",
   151→          placeItems: "center",
   152→          padding: 0,
   153→          border: 0,
   154→          borderRadius: "50%",
   155→          cursor: "pointer",
   156→          color: mine ? colors.onOlive : "#fffaf0",
   157→          background: mine ? "rgba(255,255,255,.22)" : colors.olive600,
   158→        }}
   159→      >
   160→        {playing ? (
   161→          <span aria-hidden style={{ display: "inline-flex", gap: 2.5 }}>
   162→            <span style={{ width: 3.5, height: 12, background: "currentColor", borderRadius: 2 }} />
   163→            <span style={{ width: 3.5, height: 12, background: "currentColor", borderRadius: 2 }} />
   164→          </span>
   165→        ) : (
   166→          <Icon name="play" size={15} stroke={2.2} />
   167→        )}
   168→      </button>
   169→      <div style={{ flex: 1, minWidth: 0 }}>
   170→        <div
   171→          aria-hidden
   172→          style={{
   173→            height: 4,
   174→            borderRadius: 999,
   175→            overflow: "hidden",
   176→            background: mine ? "rgba(255,255,255,.25)" : colors.olive100,
   177→          }}
   178→        >
   179→          <div
   180→            style={{
   181→              width: `${Math.round(progress * 100)}%`,
   182→              height: "100%",
   183→              borderRadius: 999,
   184→              background: mine ? "#fffaf0" : colors.olive600,
   185→              transition: "width .2s linear",
   186→            }}
   187→          />
   188→        </div>
   189→        <span style={{ display: "block", marginTop: 3, fontSize: 10, fontWeight: 700, opacity: 0.9 }}>
   190→          {formatDuration(durationMs || 0, "fa")}
   191→        </span>
   192→      </div>
   193→      <audio
   194→        ref={audioRef}
   195→        src={url}
   196→        preload="metadata"
   197→        onPlay={() => setPlaying(true)}
   198→        onPause={() => setPlaying(false)}
   199→        onEnded={() => {
   200→          setPlaying(false);
   201→          setProgress(0);
   202→        }}
   203→        onTimeUpdate={(event) => {
   204→          const audio = event.currentTarget;
   205→          if (audio.duration > 0 && Number.isFinite(audio.duration)) {
   206→            setProgress(audio.currentTime / audio.duration);
   207→          }
   208→        }}
   209→      />
   210→    </div>
   211→  );
   212→}

   224→  const [input, setInput] = useState("");
   225→  const [subject, setSubject] = useState("");
   226→  const [busy, setBusy] = useState(false);

   234→  const [imagePreview, setImagePreview] = useState<{ file: File; url: string } | null>(null);

   237→  const recorderRef = useRef<MediaRecorder | null>(null);

   246→  // هویت کاربر؛ دکمهٔ شناور فقط برای کاربران وارد‌شده نمایش داده می‌شود.
   247→  useEffect(() => {
   248→    let active = true;

   262→  // نشان قرمز پیام‌های خوانده‌نشدهٔ پشتیبانی
   263→  useEffect(() => {
   264→    if (!authed) return;
   265→    let active = true;
   266→    const load = () => {
   267→      fetchUnreadTotal(runtime)
   268→        .then((value) => {
   269→          if (active) setBadge(value);

   295→    void refreshTickets();

   299→
   300→  // گفتگوی باز: واکشی اولیه + نظرسنجی افزایندهٔ پیام‌های تازه.
   301→  const threadTicketId = view === "thread" && ticket ? ticket.id : 0;
   302→  useEffect(() => {
   303→    if (!open || threadTicketId === 0 || !authed) return;
   304→    let active = true;
   305→    const ticketId = threadTicketId;
   306→    lastMessageIdRef.current = 0;
   307→    setMessages([]);
   308→
   309→    const pull = (reset: boolean) => {
   310→      const afterId = reset ? 0 : lastMessageIdRef.current;
   311→      fetchThread(runtime, ticketId, afterId)
   312→        .then((payload) => {
   313→          if (!active) return;
   314→          setTicket(payload.ticket);
   315→          if (reset) {
   316→            setMessages(payload.messages);
   317→          } else if (payload.messages.length > 0) {
   318→            setMessages((current) => {
   319→              const seen = new Set(current.map((item) => item.id));
   320→              return [...current, ...payload.messages.filter((item) => !seen.has(item.id))];
   321→            });
   322→          }
   323→          const all = payload.messages;
   324→          if (all.length > 0) lastMessageIdRef.current = Math.max(lastMessageIdRef.current, all[all.length - 1].id);
   325→        })
   326→        .catch(() => {});
   327→    };
   328→
   329→    pull(true);
   330→    const timer = window.setInterval(() => pull(false), POLL_THREAD_MS);
   331→    return () => {
   332→      active = false;
   333→      window.clearInterval(timer);
   334→    };
   335→  }, [open, threadTicketId, authed, runtime]);

   338→  useEffect(() => {
   339→    const node = listRef.current;

   395→  async function startRecording() {
   396→    setError("");
   397→    try {
   398→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   399→      const mime = pickRecorderMime();
   400→      recordMimeRef.current = mime;
   401→      const recorder = new MediaRecorder(stream, mime ? { mimeType: mime } : undefined);
   402→      recordChunksRef.current = [];
   403→      recorder.ondataavailable = (event) => {
   404→        if (event.data.size > 0) recordChunksRef.current.push(event.data);
   405→      };
   406→      recorder.onstop = () => {
   407→        stream.getTracks().forEach((track) => track.stop());
   408→        if (recordTimerRef.current !== null) {
   409→          window.clearInterval(recordTimerRef.current);
   410→          recordTimerRef.current = null;
   411→        }
   412→        const seconds = recordSecondsRef.current;
   413→        setRecording(false);
   414→        setRecordSeconds(0);
   415→        if (recordChunksRef.current.length === 0) return;
   416→        const blob = new Blob(recordChunksRef.current, { type: mime || "audio/webm" });
   417→        setVoicePreview({
   418→          blob,
   419→          url: URL.createObjectURL(blob),
   420→          ms: Math.max(600, seconds * 1000),
   421→          mime: mime || "audio/webm",
   422→        });
   423→      };
   424→      recorderRef.current = recorder;
   425→      recordSecondsRef.current = 0;
   426→      setRecordSeconds(0);
   427→      setRecording(true);
   428→      recorder.start(250);
   429→      recordTimerRef.current = window.setInterval(() => {
   430→        recordSecondsRef.current += 1;
   431→        setRecordSeconds(recordSecondsRef.current);
   432→        if (recordSecondsRef.current >= MAX_RECORD_SECONDS) {
   433→          recorderRef.current?.stop();
   434→        }
   435→      }, 1000);
   436→    } catch {
   437→      setError(L("دسترسی به میکروفون ممکن نشد؛ اجازهٔ مرورگر را بررسی کنید."));
   438→    }
   439→  }

   441→  function stopRecording(send: boolean) {
   442→    const recorder = recorderRef.current;
   443→    if (!recorder) return;
   444→    if (!send) {
   445→      // لغو: جریان بسته و ذخیره‌ها دور ریخته می‌شوند.
   446→      recorder.onstop = () => {
   447→        recorder.stream.getTracks().forEach((track) => track.stop());
   448→        if (recordTimerRef.current !== null) {
   449→          window.clearInterval(recordTimerRef.current);
   450→          recordTimerRef.current = null;
   451→        }
   452→        setRecording(false);
   453→        setRecordSeconds(0);
   454→      };
   455→      recordChunksRef.current = [];
   456→    }
   457→    recorder.stop();
   458→  }
