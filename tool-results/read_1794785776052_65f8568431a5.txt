   283→   283→function parseTeacherSummary(value: unknown): TeacherSummary | null {

   316→   316→async function getTeacherDashboardSnapshot(runtime: PlatformRuntime): Promise<DashboardSnapshot> {
   317→   317→  const [meResponse, pendingResponse, upcomingResponse] = await Promise.all([
   318→   318→    runtime.request<ApiEnvelope<unknown>>("/me"),
   319→   319→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=pending"),
   320→   320→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=upcoming"),
   321→   321→  ]);
   322→   322→
   323→   323→  const teacher = parseTeacherSummary(meResponse.data);
   324→   324→  if (!teacher) throw new Error("اطلاعات حساب استاد دریافت نشد.");
   325→   325→
   326→   326→  return {
   327→   327→    teacher,
   328→   328→    pending: (pendingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   329→   329→    upcoming: (upcomingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   330→   330→  };
   331→   331→}
