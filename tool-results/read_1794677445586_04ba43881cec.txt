    75→type CatalogEntry = { id: number; category: string; title: string; description: string };
    76→
    77→async function fetchCatalog(): Promise<CatalogEntry[]> {
    78→  const response = await fetch("/api/backend/catalog/services", { cache: "no-store" });
    79→  if (!response.ok) throw new Error("catalog failed");
    80→  const payload = (await response.json()) as { data?: unknown };
    81→  const data = asRecord(payload.data);
    82→  const entries: CatalogEntry[] = [];
    83→  for (const group of asArray(data?.categories)) {
    84→    const row = asRecord(group);
    85→    const category = String(row?.category ?? "");
    86→    for (const item of asArray(row?.services)) {
    87→      const entry = asRecord(item);
    88→      const id = Number(entry?.id ?? 0);
    89→      if (id > 0 && entry?.title) {
    90→        entries.push({ id, category, title: String(entry.title), description: String(entry.description ?? "") });
    91→      }
    92→    }
    93→  }
    94→  return entries;
    95→}
