   132→function faDate(iso?: string): string {
   133→  if (!iso) return '—';
   134→  try {
   135→    return new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'short', year: 'numeric' }).format(new Date(iso));
   136→  } catch {
   137→    return '—';
   138→  }
   139→}
   140→
   141→function faDateTime(iso?: string): string {
   142→  if (!iso) return '—';
   143→  try {
   144→    const day = new Intl.DateTimeFormat('fa-IR', { day: 'numeric', month: 'short' }).format(new Date(iso));
   145→    const time = new Intl.DateTimeFormat('fa-IR', { hour: '2-digit', minute: '2-digit' }).format(new Date(iso));
   146→    return `${day} · ${time}`;
   147→  } catch {
   148→    return '—';
   149→  }
   150→}

   152→function teacherStatusLabel(status?: string): string {

   156→  return '—';
   157→}
