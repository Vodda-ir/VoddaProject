   436→  async function joinClassroom(withMedia: boolean) {
   437→    if (!session) return;
   438→    stopPreview();
   439→    void audioCtxRef.current?.close();
   440→    audioCtxRef.current = null;
   441→
   442→    let stream: MediaStream | null = null;
   443→    if (withMedia) {
   444→      try {
   445→        stream = await navigator.mediaDevices.getUserMedia({
   446→          video: camId ? { deviceId: { exact: camId } } : true,
   447→          audio: micId ? { deviceId: { exact: micId } } : true,
   448→        });
   449→      } catch {
   450→        setMediaError('ورود با دوربین/میکروفون ناموفق بود؛ بدون رسانه وارد می‌شوید.');
   451→      }
   452→    }
   453→    liveStreamRef.current = stream;
   454→    setPhase('live');
   455→
   456→    const client = new LiveClient({
   457→      sessionId: session.id,
   458→      onEvent: handleEvent,
   459→      onStatus: (value) => setStatus(value),
   460→    });
   461→    clientRef.current = client;
   462→
   463→    const engine = new LiveEngine({
   464→      role,
   465→      onRemoteStream: (remote) => {
   466→        setRemoteActive(remote.getTracks().length > 0);
   467→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   468→      },
   469→      onState: (state) => setConnState(`${state.connection}/${state.ice}`),
   470→      send: (type, payload) => client.send(type, payload),
   471→    });
   472→    engineRef.current = engine;
   473→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   474→    await engine.ensureConnection();
   475→    if (stream) await engine.setLocalMedia(stream);
   476→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   477→    if (engine.isInitiator && peerOnline) void engine.initiate();
   478→
   479→    await client.connect();
   480→  }

   506→  async function toggleShare() {
   507→    const engine = engineRef.current;
   508→    if (!engine) return;
   509→    if (sharing) {
   510→      screenRef.current?.getTracks().forEach((track) => track.stop());
   511→      screenRef.current = null;
   512→      await engine.stopScreenShare();
   513→      if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;
   514→      setSharing(false);
   515→      return;
   516→    }
   517→    try {
   518→      const screen = await engine.startScreenShare();
   519→      screenRef.current = screen;
   520→      if (localVideoRef.current) localVideoRef.current.srcObject = screen;
   521→      setSharing(true);
   522→      screen.getVideoTracks()[0]?.addEventListener('ended', () => {
   523→        setSharing(false);
   524→        if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;

   527→      setNotice(cause instanceof Error ? cause.message : 'اشتراک صفحه انجام نشد.');
   528→      setTimeout(() => setNotice(''), 4000);
   529→    }

   752→      {/* control bar */}

   754→        <ControlButton active={micOn} label={micOn ? 'میکروفون روشن' : 'میکروفون خاموش'} emoji={micOn ? '🎙️' : '🔇'} onClick={() => { setMicOn(!micOn); void engineRef.current?.toggleMic(!micOn); }} />

   758→        {REACTIONS.map((emoji) => <ControlButton key={emoji} active={false} label={`واکنش ${emoji}`} emoji={emoji} onClick={() => sendReaction(emoji)} small />)}
   759→        <ControlButton active={sideTab === 'chat'} label="گفت‌وگو" emoji="💬" onClick={() => openSide('chat')} badge={unread > 0 && sideTab !== 'chat' ? unread : 0} />
   760→        <ControlButton active={sideTab === 'settings'} label="تنظیمات دوربین و صدا" emoji="⚙️" onClick={() => openSide('settings')} />

   762→        <ControlButton active={false} label={isTeacherSide ? 'پایان کلاس و تسویه' : 'خروج از کلاس'} emoji="🔚" danger onClick={() => { if (window.confirm(isTeacherSide ? 'کلاس پایان یابد و تسویهٔ جلسه انجام شود؟' : 'از کلاس خارج می‌شوید؟')) void leaveClassroom(isTeacherSide); }} />
   763→      </div>
   764→
   765→      <style jsx global>{`

   768→          100% { transform: translateY(-140px) scale(1.5); opacity: 0; }

   770→      `}</style>
   771→    </main>
   772→  );

   776→  active: boolean;
   777→  label: string;
   778→  emoji: string;
   779→  onClick: () => void;
   780→  danger?: boolean;
   781→  small?: boolean;

   793→        alignItems: 'center',
