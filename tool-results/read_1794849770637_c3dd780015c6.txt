    82→function formatDate(value: string) {
    83→  const date = new Date(value);
    84→  return Number.isNaN(date.getTime()) ? '—' : new Intl.DateTimeFormat('fa-IR', { dateStyle: 'short', timeStyle: 'short' }).format(date);
    85→}
    86→
    87→function shortTime(value: string) {
    88→  const date = new Date(value);
    89→  return Number.isNaN(date.getTime()) ? '' : new Intl.DateTimeFormat('fa-IR', { hour: '2-digit', minute: '2-digit' }).format(date);
    90→}

    92→function formatDuration(ms: number) {
    93→  const total = Math.max(0, Math.round(ms / 1000));
    94→  const minutes = Math.floor(total / 60);
    95→  const seconds = total % 60;
    96→  return `${toFaDigits(minutes)}:${toFaDigits(String(seconds).padStart(2, '0'))}`;
    97→}

   115→async function apiJson(path: string, init?: RequestInit): Promise<Record<string, unknown>> {
   116→  const response = await fetch(`/api/backend${path}`, {
   117→    ...init,
   118→    cache: 'no-store',
   119→    credentials: 'same-origin',
   120→  });
   121→  const body = (await response.json().catch(() => ({}))) as Record<string, unknown>;
   122→  if (!response.ok) {
   123→    const message = typeof body.error === 'string' ? body.error : 'درخواست ناموفق بود.';
   124→    throw new Error(message);
   125→  }
   126→  return body;
   127→}

   129→function pickRecorderMime(): string {
   130→  if (typeof MediaRecorder === 'undefined' || !MediaRecorder.isTypeSupported) return '';
   131→  const candidates = ['audio/webm;codecs=opus', 'audio/webm', 'audio/ogg;codecs=opus', 'audio/mp4'];
   132→  return candidates.find((mime) => MediaRecorder.isTypeSupported(mime)) ?? '';
   133→}

   135→function mimeToExtension(mime: string): string {
   136→  if (mime.includes('webm')) return '.webm';
   137→  if (mime.includes('ogg')) return '.ogg';
   138→  if (mime.includes('mp4')) return '.m4a';
   139→  return '.bin';
   140→}

   142→/** compact voice-note player used in the staff thread. */
   143→function StaffAudioBubble({ url, durationMs, mine }: { url: string; durationMs: number; mine: boolean }) {
   144→  const audioRef = useRef<HTMLAudioElement | null>(null);
   145→  const [playing, setPlaying] = useState(false);
   146→  const [progress, setProgress] = useState(0);
   147→
   148→  useEffect(() => () => audioRef.current?.pause(), []);
   149→
   150→  return (
   151→    <div
   152→      style={{
   153→        display: 'flex',
   154→        alignItems: 'center',
   155→        gap: 9,
   156→        minWidth: 170,
   157→        padding: '7px 10px',
   158→        borderRadius: 999,
   159→        border: `1px solid ${mine ? 'rgba(255,255,255,.35)' : 'var(--olive-200, #d6dcc4)'}`,
   160→        background: mine ? 'var(--olive-600, #5e6e43)' : '#fff',
   161→      }}
   162→    >
   163→      <button
   164→        type="button"
   165→        onClick={() => {
   166→          const audio = audioRef.current;
   167→          if (!audio) return;
   168→          if (playing) audio.pause();
   169→          else void audio.play().catch(() => setPlaying(false));
   170→        }}
   171→        aria-label={playing ? 'توقف پخش پیام صوتی' : 'پخش پیام صوتی'}
   172→        style={{
   173→          width: 34,
   174→          height: 34,
   175→          flex: '0 0 auto',
   176→          display: 'grid',
   177→          placeItems: 'center',
   178→          padding: 0,
   179→          border: 0,
   180→          borderRadius: '50%',
   181→          cursor: 'pointer',
   182→          color: mine ? '#fffaf0' : '#fffaf0',
   183→          background: mine ? 'rgba(255,255,255,.22)' : 'var(--olive-600, #5e6e43)',
   184→        }}
   185→      >
   186→        {playing ? (
   187→          <span aria-hidden style={{ display: 'inline-flex', gap: 2.5 }}>
   188→            <span style={{ width: 3.5, height: 12, background: 'currentColor', borderRadius: 2 }} />
   189→            <span style={{ width: 3.5, height: 12, background: 'currentColor', borderRadius: 2 }} />
   190→          </span>
   191→        ) : (
   192→          <Icon name="play" size={15} stroke={2.2} />
   193→        )}
   194→      </button>
   195→      <div style={{ flex: 1, minWidth: 0 }}>
   196→        <div aria-hidden style={{ height: 4, borderRadius: 999, overflow: 'hidden', background: mine ? 'rgba(255,255,255,.25)' : 'var(--olive-100, #e8ecdb)' }}>
   197→          <div
   198→            style={{
   199→              width: `${Math.round(progress * 100)}%`,
   200→              height: '100%',
   201→              borderRadius: 999,
   202→              background: mine ? '#fffaf0' : 'var(--olive-600, #5e6e43)',
   203→              transition: 'width .2s linear',
   204→            }}
   205→          />
   206→        </div>
   207→        <span style={{ display: 'block', marginTop: 3, fontSize: 10, fontWeight: 700, opacity: 0.9 }}>{formatDuration(durationMs || 0)}</span>
   208→      </div>
   209→      <audio
   210→        ref={audioRef}
   211→        src={url}
   212→        preload="metadata"
   213→        onPlay={() => setPlaying(true)}
   214→        onPause={() => setPlaying(false)}
   215→        onEnded={() => { setPlaying(false); setProgress(0); }}
   216→        onTimeUpdate={(event) => {
   217→          const audio = event.currentTarget;
   218→          if (audio.duration > 0 && Number.isFinite(audio.duration)) setProgress(audio.currentTime / audio.duration);
   219→        }}
   220→      />
   221→    </div>
   222→  );
   223→}

   502→  function startRecording() {
   503→    setThreadError('');
   504→    navigator.mediaDevices
   505→      ?.getUserMedia({ audio: true })
   506→      .then((stream) => {
   507→        const mime = pickRecorderMime();
   508→        const recorder = new MediaRecorder(stream, mime ? { mimeType: mime } : undefined);
   509→        recordChunksRef.current = [];
   510→        recorder.ondataavailable = (event) => {
   511→          if (event.data.size > 0) recordChunksRef.current.push(event.data);
   512→        };
   513→        recorder.onstop = () => {
   514→          stream.getTracks().forEach((track) => track.stop());
   515→          if (recordTimerRef.current !== null) {
   516→            window.clearInterval(recordTimerRef.current);
   517→            recordTimerRef.current = null;
   518→          }
   519→          const seconds = recordSecondsRef.current;
   520→          setRecording(false);
   521→          setRecordSeconds(0);
   522→          if (recordChunksRef.current.length === 0) return;
   523→          const blob = new Blob(recordChunksRef.current, { type: mime || 'audio/webm' });
   524→          setVoicePreview({ blob, url: URL.createObjectURL(blob), ms: Math.max(600, seconds * 1000), mime: mime || 'audio/webm' });
   525→        };
   526→        recorderRef.current = recorder;
   527→        recordSecondsRef.current = 0;
   528→        setRecordSeconds(0);
   529→        setRecording(true);
   530→        recorder.start(250);
   531→        recordTimerRef.current = window.setInterval(() => {
   532→          recordSecondsRef.current += 1;
   533→          setRecordSeconds(recordSecondsRef.current);
   534→          if (recordSecondsRef.current >= 300) recorderRef.current?.stop();
   535→        }, 1000);
   536→      })
   537→      .catch(() => setThreadError('دسترسی به میکروفون ممکن نشد؛ اجازهٔ مرورگر را بررسی کنید.'));
   538→  }

   540→  function stopRecording(send: boolean) {
   541→    const recorder = recorderRef.current;
   542→    if (!recorder) return;
   543→    if (!send) {
   544→      recorder.onstop = () => {
   545→        recorder.stream.getTracks().forEach((track) => track.stop());
   546→        if (recordTimerRef.current !== null) {
   547→          window.clearInterval(recordTimerRef.current);
   548→          recordTimerRef.current = null;
   549→        }
   550→        setRecording(false);
   551→        setRecordSeconds(0);
   552→      };
   553→      recordChunksRef.current = [];
   554→    }
   555→    recorder.stop();
   556→  }
