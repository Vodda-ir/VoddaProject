    91→function QuickFeatureToggles({ runtime }: { runtime: PlatformRuntime }) {
    92→  const [state, setState] = useState<FeatureSwitches | null>(null);
    93→  const [busyKey, setBusyKey] = useState<string | null>(null);
    94→  const [error, setError] = useState<string | null>(null);
    95→
    96→  useEffect(() => {
    97→    let active = true;
    98→    loadFeatureSwitches(runtime)
    99→      .then((value) => {
   100→        if (active) setState(value);
   101→      })
   102→      .catch(() => {
   103→        if (active) setState(null);
   104→      });
   105→    return () => {
   106→      active = false;
   107→    };
   108→  }, [runtime]);
   109→
   110→  const toggle = async (key: "available" | "instant_enabled" | "free_enabled", next: boolean) => {
   111→    if (state === null || busyKey !== null) return;
   112→    const previous = state;
   113→    setBusyKey(key);
   114→    setError(null);
   115→    setState({ ...state, [key === "available" ? "available" : key === "instant_enabled" ? "instantEnabled" : "freeEnabled"]: next });

   117→      await runtime.request("/teacher/me", {
   118→        method: "PUT",
   119→        body: JSON.stringify({ [key]: next }),
   120→      });

   122→      setState(previous);
   123→      setError("تغییر وضعیت انجام نشد؛ دوباره تلاش کنید.");

   125→      setBusyKey(null);

   129→  if (state === null) return null;

   247→/** وضعیت تکمیل پروفایل حرفه‌ای استاد — چک‌لیست راهنمای ورود به کار */
   248→type CompletionState = {
   249→  activeServices: number;
   250→  bioChars: number;
   251→  hasTitle: boolean;
   252→  priceSet: boolean;
   253→  workSlots: number;
   254→};
   255→
   256→async function loadCompletionState(runtime: PlatformRuntime): Promise<CompletionState> {
   257→  const [teacherResponse, workTimesResponse] = await Promise.all([
   258→    runtime.request<ApiEnvelope<unknown>>("/teacher/me"),
   259→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/work-times").catch(() => ({ data: [] as unknown[] })),
   260→  ]);
   261→  const row = asRecord(teacherResponse.data) ?? {};
   262→  const services = asArray(row.services).map(asRecord);
   263→  const workTimes = Array.isArray(workTimesResponse.data) ? workTimesResponse.data : [];
   264→  return {
   265→    activeServices: services.filter((service) => service?.active === true).length,
   266→    bioChars: asString(row.bio).trim().length,
   267→    hasTitle: asString(row.field).trim().length > 0 || asString(row.tag).trim().length > 0,
   268→    priceSet: asNumber(row.price30) > 0 || asNumber(row.price60) > 0,
   269→    workSlots: workTimes.length,
   270→  };
   271→}

   273→function parseTeacherSummary(value: unknown): TeacherSummary | null {
   274→  const user = asRecord(value);
   275→  if (!user) return null;
   276→  const teacher = asRecord(user.teacher);
   277→
   278→  return {
   279→    id: asNumber(user.id),
   280→    name: asString(user.name, "استاد"),
   281→    avatarUrl: asString(user.avatar_url) || undefined,
   282→    wallet: asNumber(user.wallet),
   283→    rating: asNumber(teacher?.rating),
   284→    reviews: asNumber(teacher?.reviews),
   285→    sessions: asNumber(teacher?.sessions),
   286→  };
   287→}
   288→
   289→function parseTeacherSession(value: unknown): TeacherSession | null {
   290→  const session = asRecord(value);
   291→  if (!session) return null;
   292→  const learner = asRecord(session.learner);
   293→
   294→  return {
   295→    id: asNumber(session.id),
   296→    learnerName: asString(learner?.name, "قرآن‌آموز"),
   297→    service: asString(session.service ?? session.title, "جلسه"),
   298→    scheduledAt: asString(session.scheduled_at),
   299→    localDate: asString(session.local_date),
   300→    localStart: asString(session.local_start),
   301→    localEnd: asString(session.local_end),
   302→    status: asString(session.status, "pending"),
   303→  };
   304→}

   306→async function getTeacherDashboardSnapshot(runtime: PlatformRuntime): Promise<DashboardSnapshot> {
   307→  const [meResponse, pendingResponse, upcomingResponse] = await Promise.all([
   308→    runtime.request<ApiEnvelope<unknown>>("/me"),
   309→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=pending"),
   310→    runtime.request<ApiEnvelope<unknown[]>>("/teacher/sessions?when=upcoming"),
   311→  ]);
   312→
   313→  const teacher = parseTeacherSummary(meResponse.data);
   314→  if (!teacher) throw new Error("اطلاعات حساب استاد دریافت نشد.");
   315→
   316→  return {
   317→    teacher,
   318→    pending: (pendingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   319→    upcoming: (upcomingResponse.data ?? []).map(parseTeacherSession).filter((item): item is TeacherSession => item !== null),
   320→  };
   321→}

   323→function officialTime(session: TeacherSession): string {
   324→  if (!session.localDate || !session.localStart || !session.localEnd) {
   325→    return "زمان رسمی جلسه دریافت نشد";
   326→  }
   327→  return `${formatJalaliDate(session.localDate)} · ${toFaDigits(session.localStart)} تا ${toFaDigits(session.localEnd)}`;
   328→}
