     3→// کلاس زندهٔ تصویری وُدّا در اپ جامع — پوشش تمام‌صفحه روی صفحهٔ جزئیات جلسه.
     4→// ورود با پیش‌آزمون دوربین/میکروفون، ویدیوی P2P، چت درون‌کلاس (وب‌سوکت)
     5→// و کلیدهای قطع صدا/تصویر و خروج. موتور و کلاینت سیگنالینگ از lib/live.

     7→import { useCallback, useEffect, useRef, useState } from "react";
     8→import { LiveClient, type LiveEvent, type LiveStatus } from "../../../lib/live/client";
     9→import { LiveEngine, type EngineRole } from "../../../lib/live/engine";
    10→import { acquireMediaStream } from "../../platform/media-permissions";
    11→import { localizedDigits } from "../../ui/persian";
    12→import { Avatar } from "../../ui";

    14→
    15→type ChatLine = { id: string; from: number; text: string; at: number; mine: boolean };

    17→type LiveRoomProps = {
    18→  sessionId: number;
    19→  role: EngineRole;
    20→  myUserId: number;
    21→  peerName: string;
    22→  peerAvatar?: string;
    23→  startAt: string;
    24→  durationMinutes: number;
    25→  onClose: () => void;
    26→};
    27→
    28→function uid(): string {
    29→  return Math.random().toString(36).slice(2) + Date.now().toString(36);
    30→}
    31→
    32→function chatTime(at: number, lang: "fa" | "en"): string {
    33→  const date = new Date(at);
    34→  return `${localizedDigits(String(date.getHours()).padStart(2, "0"), lang)}:${localizedDigits(String(date.getMinutes()).padStart(2, "0"), lang)}`;
    35→}

    55→  background: "rgba(255,255,255,.12)",

    63→
    64→export function LiveRoom({
    65→  sessionId,
    66→  role,
    67→  myUserId,
    68→  peerName,
    69→  peerAvatar,
    70→  startAt,
    71→  durationMinutes,
    72→  onClose,
    73→}: LiveRoomProps) {
    74→  const [phase, setPhase] = useState<"preflight" | "live" | "ended">("preflight");
    75→  const [status, setStatus] = useState<LiveStatus>("connecting");
    76→  const [mediaError, setMediaError] = useState("");
    77→  const [mediaDenied, setMediaDenied] = useState(false);
    78→  const [mediaBusy, setMediaBusy] = useState(false);
    79→  const [hasLiveMedia, setHasLiveMedia] = useState(false);
    80→  const [micLevel, setMicLevel] = useState(0);
    81→  const [micOn, setMicOn] = useState(true);
    82→  const [camOn, setCamOn] = useState(true);
    83→  const [remoteActive, setRemoteActive] = useState(false);
    84→  const [chat, setChat] = useState<ChatLine[]>([]);
    85→  const [chatDraft, setChatDraft] = useState("");
    86→  const [chatOpen, setChatOpen] = useState(false);
    87→  const [notice, setNotice] = useState("");
    88→  const [countdown, setCountdown] = useState("");
    89→
    90→  const clientRef = useRef<LiveClient | null>(null);
    91→  const engineRef = useRef<LiveEngine | null>(null);
    92→  const localVideoRef = useRef<HTMLVideoElement | null>(null);
    93→  const remoteVideoRef = useRef<HTMLVideoElement | null>(null);
    94→  const liveStreamRef = useRef<MediaStream | null>(null);
    95→  const previewStreamRef = useRef<MediaStream | null>(null);
    96→  const mediaBusyRef = useRef(false);
    97→  const previewRunIdRef = useRef(0);
    98→  const stopPreviewRunRef = useRef<() => void>(() => {});
    99→  const chatBodyRef = useRef<HTMLDivElement | null>(null);
   100→  const peerOnlineRef = useRef(false);
   101→  const joinAnnouncedRef = useRef(false);
   102→  const peerSignaledRef = useRef(false);
   103→  const engineRoleRef = useRef<EngineRole>(role);
   104→  const wakeLockRef = useRef<{ release: () => Promise<void>; released: boolean } | null>(null);
   105→  const [needsTapToPlay, setNeedsTapToPlay] = useState(false);
   106→  const { L, lang, dir, N } = useAppearance();
   107→
   108→  const isTeacherSide = role !== "learner";
   109→
   110→  /* ---------- پیش‌آزمون: پیش‌نمایش دوربین و سطح میکروفون ---------- */
   111→  // گرفتن رسانه با helper مشترک و نردبان پلکانی (قید مطلوب ← ساده ← فقط صدا)؛
   112→  // دیالوگ مجوز مرورگر همین‌جا باز می‌شود و هر شکست با پیام راهنما و کلید «تلاش دوباره» قابل جبران است.
   113→  const startPreview = useCallback(() => {
   114→    stopPreviewRunRef.current();
   115→    const runId = ++previewRunIdRef.current;
   116→    mediaBusyRef.current = true;
   117→    setMediaBusy(true);
   118→    let ctx: AudioContext | null = null;
   119→    let raf = 0;
   120→    let cancelled = false;
   121→    const cancelRun = () => {
   122→      cancelled = true;
   123→      cancelAnimationFrame(raf);
   124→      void ctx?.close();
   125→    };
   126→    stopPreviewRunRef.current = cancelRun;
   127→    void (async () => {
   128→      previewStreamRef.current?.getTracks().forEach((track) => track.stop());
   129→      previewStreamRef.current = null;
   130→      const result = await acquireMediaStream("both");
   131→      if (cancelled) {
   132→        if (result.ok) result.stream.getTracks().forEach((track) => track.stop());
   133→        if (runId === previewRunIdRef.current) {
   134→          mediaBusyRef.current = false;
   135→          setMediaBusy(false);
   136→        }
   137→        return;
   138→      }
   139→      mediaBusyRef.current = false;
   140→      setMediaBusy(false);
   141→      if (!result.ok) {
   142→        setMediaError(L(result.message));
   143→        setMediaDenied(result.denied);
   144→        return;
   145→      }
   146→      const stream = result.stream;
   147→      previewStreamRef.current = stream;
   148→      setMediaError("");
   149→      setMediaDenied(false);
   150→      if (localVideoRef.current) localVideoRef.current.srcObject = stream;
   151→      const [audioTrack] = stream.getAudioTracks();
   152→      if (audioTrack) {
   153→        ctx = new AudioContext();
   154→        const source = ctx.createMediaStreamSource(new MediaStream([audioTrack]));
   155→        const analyser = ctx.createAnalyser();
   156→        analyser.fftSize = 256;
   157→        source.connect(analyser);
   158→        const buffer = new Uint8Array(analyser.frequencyBinCount);
   159→        const tick = () => {
   160→          analyser.getByteFrequencyData(buffer);
   161→          const avg = buffer.reduce((sum, value) => sum + value, 0) / buffer.length;
   162→          setMicLevel(Math.min(100, Math.round((avg / 128) * 100)));
   163→          raf = requestAnimationFrame(tick);
   164→        };
   165→        tick();
   166→      }
   167→    })();
   168→  }, [L]);
   169→
   170→  const retryMedia = useCallback(() => {
   171→    if (mediaBusy) return;
   172→    startPreview();
   173→  }, [mediaBusy, startPreview]);
   174→
   175→  useEffect(() => {
   176→    if (phase !== "preflight") return;
   177→    startPreview();
   178→    return () => {
   179→      stopPreviewRunRef.current();
   180→      const preview = previewStreamRef.current;
   181→      preview?.getTracks().forEach((track) => track.stop());
   182→      previewStreamRef.current = null;
   183→    };
   184→    // startPreview عمداً در وابستگی‌ها نمی‌آید؛ اجرای دوبارهٔ افکت جریان تازهٔ پیش‌نمایش را قطع می‌کرد.
   185→  }, [phase]);
   186→
   187→  /* ---------- شمارش معکوس جلسه ---------- */
   188→  useEffect(() => {
   189→    if (!startAt) return;
   190→    const endAt = new Date(startAt).getTime() + Math.max(1, durationMinutes) * 60_000;
   191→    const tick = () => {
   192→      const diff = endAt - Date.now();
   193→      if (diff <= 0) {
   194→        setCountdown(L("زمان جلسه به پایان رسیده است"));
   195→        return;
   196→      }
   197→      const minutes = Math.floor(diff / 60_000);
   198→      const seconds = Math.floor((diff % 60_000) / 1000);
   199→      setCountdown(
   200→        L("{time} باقی‌مانده").replace(
   201→          "{time}",
   202→          `${localizedDigits(minutes, lang)}:${localizedDigits(String(seconds).padStart(2, "0"), lang)}`,
   203→        ),
   204→      );
   205→    };
   206→    tick();
   207→    const timer = window.setInterval(tick, 1000);
   208→    return () => window.clearInterval(timer);
   209→  }, [startAt, durationMinutes, L, lang]);
   210→
   211→  /* ---------- رویدادهای سیگنالینگ ---------- */
   212→  const handleEvent = useCallback(
   213→    (event: LiveEvent) => {
   214→      if (event.from === myUserId && event.type !== "system") return;
   215→      switch (event.type) {
   216→        case "chat": {
   217→          const payload = event.payload as { text?: string } | undefined;
   218→          const text = String(payload?.text ?? "").slice(0, 1000);
   219→          if (!text) return;
   220→          setChat((items) => [...items, { id: uid(), from: event.from, text, at: event.at, mine: false }]);
   221→          break;
   222→        }
   223→        case "control": {
   224→          const payload = event.payload as { action: string; value?: boolean | string } | undefined;
   225→          if (payload?.action === "bye") {
   226→            setNotice(L("طرف مقابل کلاس را ترک کرد."));
   227→            window.setTimeout(() => setNotice(""), 4000);
   228→          }
   229→          if (payload?.action === "end") setPhase("ended");
   230→          break;
   231→        }
   232→        case "offer":
   233→        case "answer":
   234→        case "ice":
   235→          peerSignaledRef.current = true;
   236→          void engineRef.current?.handleSignal(event.type, event.payload);
   237→          break;
   238→        case "system": {
   239→          const payload = event.payload as { action?: string } | undefined;
   240→          if (payload?.action === "join") {
   241→            peerOnlineRef.current = true;
   242→            setChat((items) => [...items, { id: uid(), from: 0, text: L("طرف مقابل وارد کلاس شد."), at: event.at, mine: false }]);
   243→            // هر ورود به کلاس فقط یک‌بار خودکار پیشنهاد می‌فرستد؛ join بازپخش‌شده/اکو‌شده
   244→            // (replay کانال WS یا کانال زنده) نباید دوباره offer بفرستد.
   245→            if (isTeacherSide && !joinAnnouncedRef.current) {
   246→              joinAnnouncedRef.current = true;
   247→              void engineRef.current?.initiate();
   248→            }
   249→          }
   250→          if (payload?.action === "leave") {
   251→            peerOnlineRef.current = false;
   252→            setChat((items) => [...items, { id: uid(), from: 0, text: L("طرف مقابل خارج شد."), at: event.at, mine: false }]);
   253→          }
   254→          break;
   255→        }
   256→      }
   257→    },
   258→    [isTeacherSide, myUserId],
   259→  );
   260→
   261→  useEffect(() => {
   262→    return () => {
   263→      clientRef.current?.close();
   264→      engineRef.current?.close();
   265→      liveStreamRef.current?.getTracks().forEach((track) => track.stop());
   266→    };
   267→  }, []);
   268→
   269→  useEffect(() => {
   270→    const body = chatBodyRef.current;
   271→    if (body) body.scrollTop = body.scrollHeight;
   272→  }, [chat, chatOpen]);
   273→
   274→  /* ---------- موبایل: بیدار نگه‌داشتن صفحه در طول کلاس ---------- */
   275→  useEffect(() => {
   276→    if (phase !== "live") return;
   277→    type WakeLockSentinel = { release: () => Promise<void>; released: boolean };
   278→    const wakeLockApi = (navigator as Navigator & { wakeLock?: { request: (type: "screen") => Promise<WakeLockSentinel> } }).wakeLock;
   279→    if (!wakeLockApi) return;
   280→    let released = false;
   281→    wakeLockApi.request("screen")
   282→      .then((sentinel) => {
   283→        if (released) {
   284→          void sentinel.release().catch(() => {});
   285→          return;
   286→        }
   287→        wakeLockRef.current = sentinel;
   288→      })
   289→      .catch(() => {});
   290→    return () => {
   291→      released = true;
   292→      const sentinel = wakeLockRef.current;
   293→      wakeLockRef.current = null;
   294→      void sentinel?.release().catch(() => {});
   295→    };
   296→  }, [phase]);
   297→
   298→  /* ---------- موبایل: برگشت از پس‌زمینه/تعویض شبکه → بازیابی اتصال ---------- */
   299→  useEffect(() => {
   300→    if (phase !== "live") return;
   301→    const recover = () => {
   302→      void engineRef.current?.reconnect();
   303→      const remote = remoteVideoRef.current;
   304→      if (remote && remote.paused) {
   305→        remote.play().catch(() => setNeedsTapToPlay(true));
   306→      }
   307→    };
   308→    const onVisible = () => {
   309→      if (document.visibilityState === "visible") recover();
   310→    };
   311→    document.addEventListener("visibilitychange", onVisible);
   312→    window.addEventListener("online", recover);
   313→    return () => {
   314→      document.removeEventListener("visibilitychange", onVisible);
   315→      window.removeEventListener("online", recover);
   316→    };
   317→  }, [phase]);
   318→
   319→  /* ---------- معلم: اگر ۹ ثانیه پس از ورود هیچ قاب سیگنالینگ از طرف مقابل نرسید، یک‌بار initiate می‌فرستد ---------- */
   320→  useEffect(() => {
   321→    if (phase !== "live" || !isTeacherSide) return;
   322→    const timer = window.setTimeout(() => {
   323→      if (!peerSignaledRef.current) void engineRef.current?.initiate();
   324→    }, 9_000);
   325→    return () => window.clearTimeout(timer);
   326→  }, [phase, isTeacherSide]);
   327→
   328→  /* ---------- پخش خودکار ویدیوی طرف مقابل (سیاست autoplay موبایل) ---------- */
   329→  useEffect(() => {
   330→    if (!remoteActive) return;
   331→    const remote = remoteVideoRef.current;
   332→    if (!remote) return;
   333→    remote.play().then(() => setNeedsTapToPlay(false)).catch(() => setNeedsTapToPlay(true));
   334→  }, [remoteActive]);
   335→
   336→  /* ---------- ورود به کلاس ---------- */
   337→  // جریان پیش‌نمایش اگر آماده باشد مستقیم برای جلسه به‌کار می‌رود تا دیالوگ مجوز دوباره باز نشود؛
   338→  // اگر نباشد، «ورود با رسانه» همان‌جا درخواست می‌دهد — و شکست هیچ‌وقت ورود به کلاس را блок نمی‌کند.
   339→  async function join(withMedia: boolean) {
   340→    joinAnnouncedRef.current = false;
   341→    stopPreviewRunRef.current();
   342→
   343→    let stream: MediaStream | null = previewStreamRef.current;
   344→    previewStreamRef.current = null;
   345→    if (!withMedia) {
   346→      stream?.getTracks().forEach((track) => track.stop());
   347→      stream = null;
   348→    } else if (!stream) {
   349→      setMediaBusy(true);
   350→      const result = await acquireMediaStream("both");
   351→      setMediaBusy(false);
   352→      if (!result.ok) {
   353→        setMediaError(L(result.message));
   354→        setMediaDenied(result.denied);
   355→        // بدون رسانه هم وارد کلاس می‌شویم — هیچ‌چیز نباید برگزاری کلاس را متوقف کند.
   356→      } else {
   357→        stream = result.stream;
   358→        setMediaError("");
   359→        setMediaDenied(false);
   360→      }
   361→    }
   362→    liveStreamRef.current = stream;
   363→    setHasLiveMedia(Boolean(stream));
   364→    setPhase("live");
   365→
   366→    const client = new LiveClient({
   367→      sessionId,
   368→      onEvent: handleEvent,
   369→      onStatus: (value) => setStatus(value),
   370→    });
   371→    clientRef.current = client;
   372→
   373→    const engine = new LiveEngine({
   374→      role: engineRoleRef.current,
   375→      onRemoteStream: (remote) => {
   376→        setRemoteActive(remote.getTracks().length > 0);
   377→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   378→      },
   379→      onState: () => {},
   380→      send: (type, payload) => client.send(type, payload),
   381→    });
   382→    engineRef.current = engine;
   383→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   384→    await engine.ensureConnection();
   385→    if (stream) await engine.setLocalMedia(stream);
   386→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   387→    if (engine.isInitiator && peerOnlineRef.current) void engine.initiate();
   388→
   389→    await client.connect();
   390→  }
   391→
   392→  function sendChat() {
   393→    const text = chatDraft.trim();
   394→    if (!text || !clientRef.current) return;
   395→    setChat((items) => [...items, { id: uid(), from: myUserId, text, at: Date.now(), mine: true }]);
   396→    clientRef.current.send("chat", { text: text.slice(0, 1000) });
   397→    setChatDraft("");
   398→  }
   399→
   400→  function toggleMic() {
   401→    const next = !micOn;
   402→    setMicOn(next);
   403→    liveStreamRef.current?.getAudioTracks().forEach((track) => (track.enabled = next));
   404→  }
   405→
   406→  function toggleCam() {
   407→    const next = !camOn;
   408→    setCamOn(next);
   409→    liveStreamRef.current?.getVideoTracks().forEach((track) => (track.enabled = next));
   410→  }
   411→
   412→  /* ---------- وصل دوبارهٔ دوربین و میکروفون در میانهٔ کلاس ---------- */
   413→  // وقتی کاربر بدون رسانه وارد شده، این مسیر از نوار کنترل دیالوگ مجوز را باز می‌کند
   414→  // و تراک‌ها را به موتور P2P می‌دهد — خروج از کلاس لازم نیست.
   415→  async function enableLiveMedia() {
   416→    if (mediaBusyRef.current) return;
   417→    mediaBusyRef.current = true;
   418→    setMediaBusy(true);
   419→    const result = await acquireMediaStream("both");
   420→    mediaBusyRef.current = false;
   421→    setMediaBusy(false);
   422→    if (!result.ok) {
   423→      setNotice(L(result.message));
   424→      window.setTimeout(() => setNotice(""), 4000);
   425→      return;
   426→    }
   427→    const stream = result.stream;
   428→    liveStreamRef.current = stream;
   429→    setHasLiveMedia(true);
   430→    setMediaError("");
   431→    setMediaDenied(false);
   432→    await engineRef.current?.setLocalMedia(stream);
   433→    engineRef.current?.setCameraTrack(stream.getVideoTracks()[0] ?? null);
   434→    if (localVideoRef.current) localVideoRef.current.srcObject = stream;
   435→    setMicOn(true);
   436→    setCamOn(result.hasVideo);
   437→  }
   438→
   439→  function leave(complete: boolean) {
   440→    clientRef.current?.send("control", { action: complete ? "end" : "bye" });
   441→    clientRef.current?.close();
   442→    engineRef.current?.close();
   443→    liveStreamRef.current?.getTracks().forEach((track) => track.stop());
   444→    setPhase("ended");
   445→    onClose();
   446→  }
   447→
   448→  const statusLabel =
   449→    status === "live"
   450→      ? L("اتصال زنده")
   451→      : status === "polling"
   452→        ? L("اتصال پایدار (کانال پشتیبان)")
   453→        : status === "connecting"
   454→          ? L("در حال اتصال…")
   455→          : status === "error"
   456→            ? L("خطای دسترسی")
   457→            : L("پایان");
   458→
   459→  /* ---------- پوشش تمام‌صفحه ---------- */
   460→  return (
   461→    <div
   462→      role="dialog"
   463→      aria-label={L("کلاس زندهٔ تصویری")}
   464→      style={{
   465→        position: "fixed",
   466→        inset: 0,
   467→        zIndex: 80,
   468→        background: "#171b14",
   469→        color: "#fff",
   470→        display: "flex",
   471→        flexDirection: "column",
   472→        direction: dir,
   473→        fontFamily: "inherit",
   474→      }}
   475→    >
   476→      {/* نوار بالا */}
   477→      <div
   478→        style={{
   479→          display: "flex",
   480→          alignItems: "center",
   481→          gap: 10,
   482→          padding: "12px 16px",
   483→          background: "rgba(23,27,20,.92)",
   484→          borderBottom: "1px solid rgba(255,255,255,.08)",
   485→        }}
   486→      >
   487→        <Avatar name={peerName} src={peerAvatar || undefined} size={38} ring={false} />
   488→        <div style={{ minWidth: 0, flex: 1 }}>
   489→          <strong style={{ display: "block", fontSize: 14, fontWeight: 800 }}>{N(peerName)}</strong>
   490→          <span style={{ display: "block", fontSize: 11, color: remoteActive ? "#a8d08d" : "rgba(255,255,255,.55)" }}>
   491→            {remoteActive ? L("متصل") : statusLabel}
   492→            {countdown ? ` · ${countdown}` : ""}
   493→          </span>
   494→        </div>
   495→        <button
   496→          type="button"
   497→          onClick={() => setChatOpen((open) => !open)}
   498→          aria-label={L("چت کلاس")}
