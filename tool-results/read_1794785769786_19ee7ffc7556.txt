    99→function recommendationScore(teacher: Teacher) {
   100→  return teacher.rating * Math.log10(10 + Math.max(0, teacher.sessions));
   101→}

   103→function sortTeachers(rows: Teacher[], sort: SortKey): Teacher[] {
   104→  const copy = [...rows];
   105→  switch (sort) {
   106→    case "rating":
   107→      return copy.sort((a, b) => b.rating - a.rating);
   108→    case "sessions":
   109→      return copy.sort((a, b) => b.sessions - a.sessions);
   110→    case "newest":
   111→      return copy.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
   112→    default:
   113→      return copy.sort((a, b) => recommendationScore(b) - recommendationScore(a));
   114→  }
   115→}
