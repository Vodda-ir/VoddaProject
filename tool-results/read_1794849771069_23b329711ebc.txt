   437→  async function joinClassroom(withMedia: boolean) {
   438→    if (!session) return;
   439→    stopPreview();
   440→    void audioCtxRef.current?.close();
   441→    audioCtxRef.current = null;
   442→
   443→    let stream: MediaStream | null = null;
   444→    if (withMedia) {
   445→      try {
   446→        stream = await navigator.mediaDevices.getUserMedia({
   447→          video: camId ? { deviceId: { exact: camId } } : true,
   448→          audio: micId ? { deviceId: { exact: micId } } : true,
   449→        });
   450→      } catch {
   451→        setMediaError('ورود با دوربین/میکروفون ناموفق بود؛ بدون رسانه وارد می‌شوید.');
   452→      }
   453→    }
   454→    liveStreamRef.current = stream;
   455→    setPhase('live');
   456→
   457→    const client = new LiveClient({
   458→      sessionId: session.id,
   459→      onEvent: handleEvent,
   460→      onStatus: (value) => setStatus(value),
   461→    });
   462→    clientRef.current = client;
   463→
   464→    const engine = new LiveEngine({
   465→      role,
   466→      onRemoteStream: (remote) => {
   467→        setRemoteActive(remote.getTracks().length > 0);
   468→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   469→      },
   470→      onState: (state) => setConnState(`${state.connection}/${state.ice}`),
   471→      send: (type, payload) => client.send(type, payload),
   472→    });
   473→    engineRef.current = engine;
   474→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   475→    await engine.ensureConnection();
   476→    if (stream) await engine.setLocalMedia(stream);
   477→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   478→    if (engine.isInitiator && peerOnline) void engine.initiate();
   479→
   480→    await client.connect();
   481→  }

   507→  async function toggleShare() {
   508→    const engine = engineRef.current;
   509→    if (!engine) return;
   510→    if (sharing) {
   511→      screenRef.current?.getTracks().forEach((track) => track.stop());
   512→      screenRef.current = null;
   513→      await engine.stopScreenShare();
   514→      if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;
   515→      setSharing(false);
   516→      return;
   517→    }
   518→    try {
   519→      const screen = await engine.startScreenShare();
   520→      screenRef.current = screen;
   521→      if (localVideoRef.current) localVideoRef.current.srcObject = screen;
   522→      setSharing(true);
   523→      screen.getVideoTracks()[0]?.addEventListener('ended', () => {
   524→        setSharing(false);
   525→        if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;

   528→      setNotice(cause instanceof Error ? cause.message : 'اشتراک صفحه انجام نشد.');
   529→      setTimeout(() => setNotice(''), 4000);
   530→    }

   760→        <ControlButton active={sideTab === 'chat'} label="گفت‌وگو" emoji="💬" onClick={() => openSide('chat')} badge={unread > 0 && sideTab !== 'chat' ? unread : 0} />
   761→        <ControlButton active={sideTab === 'settings'} label="تنظیمات دوربین و صدا" emoji="⚙️" onClick={() => openSide('settings')} />

   763→        <ControlButton active={false} label={isTeacherSide ? 'پایان کلاس و تسویه' : 'خروج از کلاس'} emoji="🔚" danger onClick={() => { void confirmDialog(isTeacherSide ?    764→      </div>

   772→    </main>
   773→  );
