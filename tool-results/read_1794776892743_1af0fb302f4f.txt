    79→    79→        SkipWorkTimeCheck bool

    82→    82→        // ignores the weekly template.

    86→    86→// Book creates a booking: the wallet method charges the learner's wallet up
    87→    87→// front; the gateway method reserves the slot and hands the learner to the
    88→    88→// Saman Sep payment page, activating the session after settlement.
    89→    89→func (uc *SessionUsecase) Book(ctx context.Context, learnerID uint, in BookInput) (*BookResult, error) {
    90→    90→        uc.cancelExpiredAwaiting(ctx)
    91→    91→        uc.expireStaleInstantPending(ctx)
    92→    92→
    93→    93→        gatewayPayment := in.PaymentMethod == "gateway"
    94→    94→        if in.PaymentMethod != "" && in.PaymentMethod != "wallet" && in.PaymentMethod != "gateway" {
    95→    95→                return nil, domain.ErrValidation
    96→    96→        }
    97→    97→        if gatewayPayment && uc.payments == nil {
    98→    98→                return nil, domain.ErrForbidden
    99→    99→        }
   100→   100→
   101→   101→        teacher, err := uc.teachers.ByID(ctx, in.TeacherID)
   102→   102→        if err != nil {
   103→   103→                return nil, err
   104→   104→        }
   105→   105→        if teacher.Status != domain.TeacherApproved || !teacher.Verified {
   106→   106→                return nil, domain.ErrTeacherNotReady
   107→   107→        }
   108→   108→        if !teacher.Available {
   109→   109→                return nil, domain.ErrTeacherOffline
   110→   110→        }
   111→   111→        if in.Instant && !teacher.InstantEnabled {
   112→   112→                return nil, domain.ErrInstantUnavailable
   113→   113→        }
   114→   114→        if teacher.UserID == learnerID {
   115→   115→                return nil, domain.ErrSelfBooking
   116→   116→        }
   117→   117→
   118→   118→        svc, err := uc.teachers.ServiceByID(ctx, in.ServiceID)
   119→   119→        if err != nil || svc.TeacherID != teacher.ID || !svc.Active || svc.Price < 0 || svc.Duration <= 0 || svc.Duration > 240 {
   120→   120→                return nil, domain.ErrValidation
   121→   121→        }
   122→   122→        if len(in.Note) > 500 {
   123→   123→                return nil, domain.ErrValidation
   124→   124→        }
   125→   125→        if svc.Price == 0 && !teacher.FreeEnabled {
   126→   126→                return nil, domain.ErrFreeUnavailable
   127→   127→        }
   128→   128→
   129→   129→        now := time.Now()
   130→   130→        if in.Instant {
   131→   131→                // جلسهٔ فوری — رزرو همین لحظه: پنجرهٔ آغاز کوتاه و بدون قالب هفتگی.
   132→   132→                if in.ScheduledAt.Before(now.Add(-time.Minute)) || in.ScheduledAt.After(now.Add(15*time.Minute)) {
   133→   133→                        return nil, domain.ErrValidation
   134→   134→                }
   135→   135→        } else if in.ScheduledAt.Before(now.Add(5*time.Minute)) || in.ScheduledAt.After(now.AddDate(0, 6, 0)) {
   136→   136→                return nil, domain.ErrValidation
   137→   137→        }
   138→   138→        if !in.SkipWorkTimeCheck && !in.Instant {
   139→   139→                if err := uc.ensureWithinWorkTime(ctx, teacher.ID, in.ScheduledAt, svc.Duration); err != nil {
   140→   140→                        return nil, err
   141→   141→                }
   142→   142→        }
   143→   143→        if err := uc.ensureNoOverlap(ctx, learnerID, teacher.ID, in.ScheduledAt, svc.Duration); err != nil {
   144→   144→                return nil, err
   145→   145→        }
   146→   146→
   147→   147→        learner, err := uc.users.ByID(ctx, learnerID)
   148→   148→        if err != nil {
   149→   149→                return nil, err
   150→   150→        }
   151→   151→        if !learner.Active {
   152→   152→                return nil, domain.ErrForbidden
   153→   153→        }
   154→   154→        if !gatewayPayment && learner.Wallet < svc.Price {
   155→   155→                return nil, domain.ErrInsufficient
   156→   156→        }
   157→   157→
   158→   158→        // Title/body of the teacher notification — the mobile client rings the
   159→   159→        // summon chime for these titles (see CallAlertBar matching).
   160→   160→        notifyTitle := "درخواست جلسه جدید: " + svc.Title
   161→   161→        notifyBody := "یک قرآن‌آموز برای «" + svc.Title + "» درخواست جلسه ثبت کرد؛ از بخش درخواست‌ها تأیید یا رد کنید."
   162→   162→        if in.Instant {
   163→   163→                notifyTitle = "جلسهٔ فوری جدید: " + svc.Title
   164→   164→                notifyBody = "یک قرآن‌آموز همین حالا درخواست جلسهٔ فوری ثبت کرد؛ سریع تأیید کنید تا کلاس آغاز شود."
   165→   165→        } else if svc.Price == 0 {
   166→   166→                notifyTitle = "درخواست جلسهٔ رایگان: " + svc.Title
   167→   167→                notifyBody = "یک قرآن‌آموز جلسهٔ رایگان شما را رزرو کرد؛ از بخش درخواست‌های جلسه تأیید کنید."
   168→   168→        }
   169→   169→
   170→   170→        s := &domain.Session{
   171→   171→                LearnerID:   learnerID,
   172→   172→                TeacherID:   teacher.ID,
   173→   173→                ServiceID:   svc.ID,
   174→   174→                Title:       svc.Title,
   175→   175→                ScheduledAt: in.ScheduledAt.UTC(),
   176→   176→                Duration:    svc.Duration,
   177→   177→                Price:       svc.Price,
   178→   178→                Status:      domain.SessionPending,
   179→   179→                Note:        in.Note,
   180→   180→                Instant:     in.Instant,
   181→   181→        }
   182→   182→
   183→   183→        if gatewayPayment {
   184→   184→                if uc.financial != nil {
   185→   185→                        if err := uc.financial.CreateAwaitingSession(ctx, s); err != nil {
   186→   186→                                return nil, err
   187→   187→                        }
   188→   188→                } else {
   189→   189→                        s.Status = domain.SessionAwaitingPayment
   190→   190→                        if err := uc.sessions.Create(ctx, s); err != nil {
   191→   191→                                return nil, err
   192→   192→                        }
   193→   193→                }
   194→   194→                payment, err := uc.payments.CreateBookingPayment(ctx, learner, s, in.ReturnPath)
   195→   195→                if err != nil {
   196→   196→                        // Free the reserved slot so the learner can retry cleanly.
   197→   197→                        uc.cancelAwaitingSession(ctx, s.ID, "شروع پرداخت ناموفق بود")
   198→   198→                        return nil, err
   199→   199→                }
   200→   200→                return &BookResult{Session: s, Token: payment.Token, ResNum: payment.ResNum, PaidViaWallet: false}, nil
   201→   201→        }
   202→   202→
   203→   203→        // خدمات رایگان — جلسهٔ صفر تومانی: بدون کسر کیف پول، بدون درگاه؛
   204→   204→        // درخواست مستقیم به صف تأیید استاد می‌رود.
   205→   205→        if svc.Price == 0 {
   206→   206→                if uc.financial != nil {
   207→   207→                        if err := uc.financial.CreateFreeSession(ctx, s); err != nil {
   208→   208→                                return nil, err
   209→   209→                        }
   210→   210→                } else {
   211→   211→                        if err := uc.sessions.Create(ctx, s); err != nil {
   212→   212→                                return nil, err
   213→   213→                        }
   214→   214→                }
   215→   215→                uc.notify(ctx, teacher.UserID, notifyTitle, notifyBody)
   216→   216→                return &BookResult{Session: s, PaidViaWallet: false}, nil
   217→   217→        }
   218→   218→
   219→   219→        if uc.financial != nil {
   220→   220→                if err := uc.financial.CreateFundedSession(ctx, s); err != nil {
   221→   221→                        return nil, err
   222→   222→                }
   223→   223→        } else {
   224→   224→                // Compatibility fallback for isolated tests; production always injects the
   225→   225→                // atomic financial repository.
   226→   226→                if err := uc.sessions.Create(ctx, s); err != nil {
   227→   227→                        return nil, err
   228→   228→                }
   229→   229→                if err := uc.users.AddWallet(ctx, learnerID, -svc.Price); err != nil {
   230→   230→                        uc.cancelUnfundedSession(ctx, s)
   231→   231→                        return nil, err
   232→   232→                }
   233→   233→                if err := uc.wallet.AddTx(ctx, &domain.Transaction{
   234→   234→                        UserID: learnerID, Type: domain.TxSession,
   235→   235→                        Title: "رزرو جلسه: " + svc.Title, Amount: -svc.Price, RefID: s.ID,
   236→   236→                }); err != nil {
   237→   237→                        _ = uc.users.AddWallet(ctx, learnerID, svc.Price)
   238→   238→                        uc.cancelUnfundedSession(ctx, s)
   239→   239→                        return nil, err
   240→   240→                }
   241→   241→        }
   242→   242→
   243→   243→        uc.notify(ctx, teacher.UserID, notifyTitle, notifyBody)
   244→   244→        return &BookResult{Session: s, PaidViaWallet: true}, nil
   245→   245→}

   247→   247→// BookingPayment re-initiates the gateway payment for a session that is still
   248→   248→// awaiting payment (e.g. the learner closed the browser mid-checkout).
   249→   249→func (uc *SessionUsecase) BookingPayment(ctx context.Context, learnerID, sessionID uint, returnPath string) (*TopupResult, error) {
   250→   250→        uc.cancelExpiredAwaiting(ctx)
   251→   251→        s, err := uc.sessions.ByID(ctx, sessionID)
   252→   252→        if err != nil {
   253→   253→                return nil, err
   254→   254→        }
   255→   255→        if s.LearnerID != learnerID {
   256→   256→                return nil, domain.ErrForbidden
   257→   257→        }
   258→   258→        if s.Status != domain.SessionAwaitingPayment {
   259→   259→                return nil, domain.ErrConflict
   260→   260→        }
   261→   261→        if uc.payments == nil {
   262→   262→                return nil, domain.ErrForbidden
   263→   263→        }
   264→   264→        learner, err := uc.users.ByID(ctx, learnerID)
   265→   265→        if err != nil {
   266→   266→                return nil, err
   267→   267→        }
   268→   268→        if !learner.Active {
   269→   269→                return nil, domain.ErrForbidden
   270→   270→        }
   271→   271→        _ = uc.financialFailStaleOrders(ctx, s.ID)
   272→   272→        return uc.payments.CreateBookingPayment(ctx, learner, s, returnPath)
   273→   273→}

   275→   275→// Accept confirms a pending session (teacher action).
   276→   276→func (uc *SessionUsecase) Accept(ctx context.Context, userID, sessionID uint) (*domain.Session, error) {
   277→   277→        s, _, err := uc.ownTeacherSession(ctx, userID, sessionID)
   278→   278→        if err != nil {
   279→   279→                return nil, err
   280→   280→        }
   281→   281→        if s.Status != domain.SessionPending {
   282→   282→                return nil, domain.ErrConflict
   283→   283→        }
   284→   284→        // جلسهٔ فوری تا پایان بازهٔ خودش قابل تأیید است (استاد ممکن است با
   285→   285→        // کمی تأخیر ببیند)؛ جلسهٔ معمولی فقط پیش از آغاز رسمی.
   286→   286→        acceptDeadline := s.ScheduledAt
   287→   287→        if s.Instant {
   288→   288→                acceptDeadline = s.ScheduledAt.Add(time.Duration(s.Duration) * time.Minute)
   289→   289→        }
   290→   290→        if time.Now().After(acceptDeadline) {
   291→   291→                return nil, domain.ErrConflict
   292→   292→        }
   293→   293→        s.Status = domain.SessionConfirmed
   294→   294→        if err := uc.sessions.Update(ctx, s); err != nil {
   295→   295→                return nil, err
   296→   296→        }
   297→   297→        uc.notify(ctx, s.LearnerID, "جلسه تأیید شد", "استاد درخواست جلسه‌ی شما را تأیید کرد.")
   298→   298→        return s, nil
   299→   299→}

   301→   301→// Reject declines a pending session and refunds the learner.
   302→   302→func (uc *SessionUsecase) Reject(ctx context.Context, userID, sessionID uint, reason string) (*domain.Session, error) {
   303→   303→        s, _, err := uc.ownTeacherSession(ctx, userID, sessionID)
   304→   304→        if err != nil {
   305→   305→                return nil, err
   306→   306→        }
   307→   307→        if s.Status != domain.SessionPending {
   308→   308→                return nil, domain.ErrConflict
   309→   309→        }
   310→   310→        if len(reason) > 255 {
   311→   311→                return nil, domain.ErrValidation
   312→   312→        }
   313→   313→
   314→   314→        if uc.financial != nil {
   315→   315→                if err := uc.financial.CancelAndRefundSession(ctx, s, []domain.SessionStatus{domain.SessionPending}, reason); err != nil {
   316→   316→                        return nil, err
   317→   317→                }
   318→   318→        } else {
   319→   319→                previousStatus, previousReason := s.Status, s.RejectNote
   320→   320→                s.Status = domain.SessionCanceled
   321→   321→                s.RejectNote = reason
   322→   322→                if err := uc.sessions.Update(ctx, s); err != nil {
   323→   323→                        return nil, err
   324→   324→                }
   325→   325→                if err := uc.refund(ctx, s); err != nil {
   326→   326→                        s.Status, s.RejectNote = previousStatus, previousReason
   327→   327→                        _ = uc.sessions.Update(ctx, s)
   328→   328→                        return nil, err
   329→   329→                }
   330→   330→        }
   331→   331→        uc.notify(ctx, s.LearnerID, "جلسه لغو شد", "استاد درخواست جلسه را رد کرد و مبلغ بازگشت داده شد.")
   332→   332→        return s, nil
   333→   333→}

   370→   370→// Complete marks a confirmed session done and settles the teacher's earnings,
   371→   371→// deducting the platform commission.
   372→   372→func (uc *SessionUsecase) Complete(ctx context.Context, userID, sessionID uint) (*domain.Session, error) {
   373→   373→        s, teacher, err := uc.ownTeacherSession(ctx, userID, sessionID)
   374→   374→        if err != nil {
   375→   375→                return nil, err
   376→   376→        }
   377→   377→        if s.Status != domain.SessionConfirmed {
   378→   378→                return nil, domain.ErrConflict
   379→   379→        }
   380→   380→        // پایان جلسه پس از آغاز رسمی مجاز است (جلسه ممکن است زودتر از مدت
   381→   381→        // رزروشده تمام شود)؛ پیش از آغاز رسمی خیر.
   382→   382→        if time.Now().Before(s.ScheduledAt) {
   383→   383→                return nil, domain.ErrConflict
   384→   384→        }
   385→   385→
   386→   386→        commission := s.Price * int64(uc.cfg.PlatformCommissionPercent) / 100
   387→   387→        earning := s.Price - commission
   388→   388→        if earning < 0 {
   389→   389→                return nil, domain.ErrInternal
   390→   390→        }
   391→   391→
   392→   392→        if uc.financial != nil {
   393→   393→                if err := uc.financial.CompleteAndSettleSession(ctx, s, teacher.UserID, earning, commission); err != nil {
   394→   394→                        return nil, err
   395→   395→                }
   396→   396→        } else {
   397→   397→                // Compatibility fallback for isolated tests.
   398→   398→                s.Status = domain.SessionDone
   399→   399→                if err := uc.sessions.Update(ctx, s); err != nil {
   400→   400→                        return nil, err
   401→   401→                }
   402→   402→                if err := uc.users.AddWallet(ctx, teacher.UserID, earning); err != nil {
   403→   403→                        s.Status = domain.SessionConfirmed
   404→   404→                        _ = uc.sessions.Update(ctx, s)
   405→   405→                        return nil, err
   406→   406→                }
   407→   407→                if err := uc.wallet.AddTx(ctx, &domain.Transaction{
   408→   408→                        UserID: teacher.UserID, Type: domain.TxEarning,
   409→   409→                        Title: "درآمد ناخالص جلسه: " + s.Title, Amount: s.Price, RefID: s.ID,
   410→   410→                }); err != nil {
   411→   411→                        _ = uc.users.AddWallet(ctx, teacher.UserID, -earning)
   412→   412→                        s.Status = domain.SessionConfirmed
   413→   413→                        _ = uc.sessions.Update(ctx, s)
   414→   414→                        return nil, err
   415→   415→                }
   416→   416→                _ = uc.wallet.AddTx(ctx, &domain.Transaction{
   417→   417→                        UserID: teacher.UserID, Type: domain.TxCommission,
   418→   418→                        Title: "کارمزد پلتفرم", Amount: -commission, RefID: s.ID,

   484→   484→func (uc *SessionUsecase) ensureWithinWorkTime(ctx context.Context, teacherID uint, scheduledAt time.Time, duration int) error {
   485→   485→        location, err := time.LoadLocation(uc.cfg.AppTimezone)
   486→   486→        if err != nil {
   487→   487→                return err
   488→   488→        }
   489→   489→        local := scheduledAt.In(location)
   490→   490→        weekday := (int(local.Weekday()) + 1) % 7 // Go Sunday=0; Vodda Saturday=0.
   491→   491→        startMinute := local.Hour()*60 + local.Minute()
   492→   492→        endMinute := startMinute + duration
   493→   493→
   494→   494→        workTimes, err := uc.teachers.WorkTimes(ctx, teacherID)
   495→   495→        if err != nil {
   496→   496→                return err
   497→   497→        }
   498→   498→        for _, slot := range workTimes {
   499→   499→                if !slot.Active || slot.Weekday != weekday {
   500→   500→                        continue
   501→   501→                }
   502→   502→                from, okFrom := parseClock(slot.FromTime)
   503→   503→                to, okTo := parseClock(slot.ToTime)
   504→   504→                if okFrom && okTo && startMinute >= from && endMinute <= to {
   505→   505→                        return nil
   506→   506→                }
   507→   507→        }
   508→   508→        return domain.ErrValidation
   509→   509→}

   511→   511→func (uc *SessionUsecase) ensureNoOverlap(ctx context.Context, learnerID, teacherID uint, start time.Time, duration int) error {
   512→   512→        end := start.Add(time.Duration(duration) * time.Minute)
   513→   513→        statuses := []domain.SessionStatus{domain.SessionPending, domain.SessionConfirmed, domain.SessionAwaitingPayment}
   514→   514→        teacherSessions, err := uc.sessions.ForTeacher(ctx, teacherID, statuses)
   515→   515→        if err != nil {
   516→   516→                return err
   517→   517→        }
   518→   518→        for _, existing := range teacherSessions {
   519→   519→                if overlaps(start, end, existing.ScheduledAt, existing.ScheduledAt.Add(time.Duration(existing.Duration)*time.Minute)) {
   520→   520→                        return domain.ErrConflict
   521→   521→                }
   522→   522→        }
   523→   523→
   524→   524→        learnerSessions, err := uc.sessions.ForLearner(ctx, learnerID, "upcoming")
   525→   525→        if err != nil {
   526→   526→                return err
   527→   527→        }
   528→   528→        for _, existing := range learnerSessions {
   529→   529→                if existing.Status != domain.SessionPending && existing.Status != domain.SessionConfirmed && existing.Status != domain.SessionAwaitingPayment {
   530→   530→                        continue
   531→   531→                }
   532→   532→                if overlaps(start, end, existing.ScheduledAt, existing.ScheduledAt.Add(time.Duration(existing.Duration)*time.Minute)) {
   533→   533→                        return domain.ErrConflict
   534→   534→                }
   535→   535→        }
   536→   536→        return nil
   537→   537→}
