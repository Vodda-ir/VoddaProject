    77→function recommendationScore(teacher: Teacher) {
    78→  return teacher.rating * Math.log10(10 + Math.max(0, teacher.sessions));
    79→}
    80→
    81→function sortTeachers(rows: Teacher[], sort: SortKey): Teacher[] {
    82→  const copy = [...rows];
    83→  switch (sort) {
    84→    case "rating":
    85→      return copy.sort((a, b) => b.rating - a.rating);
    86→    case "sessions":
    87→      return copy.sort((a, b) => b.sessions - a.sessions);
    88→    case "newest":
    89→      return copy.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
    90→    default:
    91→      return copy.sort((a, b) => recommendationScore(b) - recommendationScore(a));
    92→  }
    93→}

    99→function readInitialFilters(): SearchFilters {
   100→  const empty: SearchFilters = {
   101→    query: "",
   102→    field: "",
   103→    city: "",
   104→    female: "all",
   105→    tier: "",
   106→    availableOnly: false,
   107→    sort: "recommended",
   108→  };
   109→  if (typeof window === "undefined") return empty;
   110→  const params = new URLSearchParams(window.location.search);
   111→  const femaleParam = params.get("female");
   112→  const tierParam = params.get("tier");
   113→  const sortParam = params.get("sort");
   114→  return {
   115→    query: params.get("q") ?? "",
   116→    field: params.get("field") ?? "",
   117→    city: params.get("city") ?? "",
   118→    female: femaleParam === "true" ? "female" : femaleParam === "false" ? "male" : "all",
   119→    tier: (["international", "grade2", "grade3"].includes(tierParam ?? "") ? tierParam : "") as TierKey,
   120→    availableOnly: params.get("available") === "true",
   121→    sort: (["recommended", "rating", "sessions", "newest"].includes(sortParam ?? "")
   122→      ? sortParam
   123→      : "recommended") as SortKey,
   124→  };
   125→}

   127→function readRecentSearches(): string[] {
   128→  if (typeof window === "undefined") return [];
   129→  try {
   130→    const raw = window.localStorage.getItem(RECENT_KEY);
   131→    const parsed = raw ? JSON.parse(raw) : [];
   132→    return Array.isArray(parsed)
   133→      ? parsed.filter((item): item is string => typeof item === "string" && item.length > 0).slice(0, RECENT_LIMIT)
   134→      : [];
   135→  } catch {
   136→    return [];
   137→  }

   311→export function LearnerSearchScreen({ config }: { config: UniversalAppConfig }) {
   312→  const runtime = usePlatform();
   313→  const initial = useMemo(readInitialFilters, []);
   314→  const [mode, setMode] = useState<SearchMode>(() => {
   315→    if (typeof window === "undefined") return "teachers";
   316→    return new URLSearchParams(window.location.search).get("mode") === "services" ? "services" : "teachers";
   317→  });
   318→  const [filters, setFilters] = useState<SearchFilters>(initial);
   319→  const [debouncedQuery, setDebouncedQuery] = useState(initial.query);
   320→  const [topicId, setTopicId] = useState<number>(() => {
   321→    if (typeof window === "undefined") return 0;
   322→    return Number(new URLSearchParams(window.location.search).get("topic") ?? 0) || 0;
   323→  });
   324→  const [topicTitle, setTopicTitle] = useState<string>("");
   325→  const [rows, setRows] = useState<Teacher[]>([]);
   326→  const [serverTotal, setServerTotal] = useState(0);
   327→  const [loading, setLoading] = useState(true);
   328→  const [error, setError] = useState("");
   329→  const [reloadKey, setReloadKey] = useState(0);
   330→  const [filtersOpen, setFiltersOpen] = useState(false);
   331→  const [advancedDraft, setAdvancedDraft] = useState<AdvancedDraft>({ city: initial.city, female: initial.female });
   332→  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
   333→  const [cities, setCities] = useState<string[]>([]);
   334→  const [cityQuery, setCityQuery] = useState("");
   335→  const [recent, setRecent] = useState<string[]>(() => readRecentSearches());
   336→  const [fallback, setFallback] = useState<Teacher[]>([]);
   337→  const sentinelRef = useRef<HTMLDivElement | null>(null);
   338→
   339→  // داده‌های تب «خدمات»: کاتالوگ آنلاین + کاتالوگ حضوری، فقط یک بار
   340→  const [onlineServices, setOnlineServices] = useState<OfficialService[]>([]);
   341→  const [presenceServices, setPresenceServices] = useState<OfficialInPersonService[]>([]);
   342→  const [servicesLoading, setServicesLoading] = useState(false);
   343→  const [servicesError, setServicesError] = useState("");
   344→  const [servicesQuery, setServicesQuery] = useState("");
   345→  const [servicesLoaded, setServicesLoaded] = useState(false);
   346→
   347→  useEffect(() => {
   348→    if (mode !== "services" || servicesLoaded) return;
   349→    let active = true;
   350→    setServicesLoading(true);
   351→    setServicesError("");
   352→    Promise.all([getCatalog(runtime), getInPersonCatalog(runtime).catch(() => [] as OfficialInPersonService[])])
   353→      .then(([online, presence]) => {
   354→        if (!active) return;
   355→        setOnlineServices(online);
   356→        setPresenceServices(presence);
   357→        setServicesLoaded(true);
   358→      })
   359→      .catch((reason: unknown) => {
   360→        if (active) setServicesError(messageOf(reason, "دریافت فهرست خدمات ناموفق بود."));
   361→      })
   362→      .finally(() => {
   363→        if (active) setServicesLoading(false);
   364→      });
   365→    return () => {
   366→      active = false;
   367→    };
   368→  }, [mode, servicesLoaded, runtime]);
   369→
   370→  useEffect(() => {
   371→    const timer = window.setTimeout(() => setDebouncedQuery(filters.query.trim()), 350);
   372→    return () => window.clearTimeout(timer);
   373→  }, [filters.query]);
   374→
   375→  // ثبت جستجوهای اخیر
   376→  useEffect(() => {
   377→    const query = debouncedQuery;
   378→    if (!query) return;
   379→    setRecent((current) => {
   380→      const next = [query, ...current.filter((item) => item !== query)].slice(0, RECENT_LIMIT);
   381→      try {
   382→        window.localStorage.setItem(RECENT_KEY, JSON.stringify(next));
   383→      } catch {
   384→        // حافظه محلی در دسترس نیست — بی‌اهمیت
   385→      }
   386→      return next;
   387→    });
   388→  }, [debouncedQuery]);
   389→
   390→  useEffect(() => {
   391→    const params = new URLSearchParams();
   392→    if (filters.query.trim()) params.set("q", filters.query.trim());
   393→    if (filters.field.trim()) params.set("field", filters.field.trim());
   394→    if (filters.city.trim()) params.set("city", filters.city.trim());
   395→    if (filters.female === "female") params.set("female", "true");
   396→    if (filters.female === "male") params.set("female", "false");
   397→    if (filters.tier) params.set("tier", filters.tier);
   398→    if (filters.availableOnly) params.set("available", "true");
   399→    if (filters.sort !== "recommended") params.set("sort", filters.sort);
   400→    const next = window.location.pathname + (params.size ? "?" + params.toString() : "");
   401→    window.history.replaceState({}, "", next);
   402→  }, [filters]);
   403→
   404→  useEffect(() => {
   405→    let active = true;
   406→    setLoading(true);
   407→    setError("");
   408→    setVisibleCount(PAGE_SIZE);
   409→    searchTeachersPage(runtime, {
   410→      query: debouncedQuery,
   411→      field: filters.field,
   412→      city: filters.city,
   413→      female: filters.female === "all" ? undefined : filters.female === "female",
   414→      officialServiceId: topicId,
   415→      limit: 100,
   416→      offset: 0,
   417→    })
   418→      .then((result) => {
   419→        if (!active) return;
   420→        const visible = result.items.filter((teacher) => !MADDAHI_PATTERN.test(teacher.field + " " + teacher.tag));
   421→        setRows(visible);
   422→        setServerTotal(result.total);
   423→      })
   424→      .catch((reason: unknown) => {
   425→        if (!active) return;
   426→        setRows([]);
   427→        setError(messageOf(reason, "دریافت فهرست استادان با خطا روبه‌رو شد."));
   428→      })
   429→      .finally(() => {
   430→        if (active) setLoading(false);
   431→      });
   432→    return () => {
   433→      active = false;
   434→    };
   435→  }, [runtime, debouncedQuery, filters.field, filters.city, filters.female, topicId, reloadKey]);
   436→
   437→  // عنوان موضوع فراخوان (برای نشانِ فیلتر فعال)
   438→  useEffect(() => {
   439→    if (!topicId) {
   440→      setTopicTitle("");
   441→      return;
   442→    }
   443→    let active = true;
   444→    getCatalog(runtime)
   445→      .then((rows) => {
   446→        if (active) setTopicTitle(rows.find((row) => row.id === topicId)?.title ?? "");
   447→      })
   448→      .catch(() => {
   449→        if (active) setTopicTitle("");
   450→      });
   451→    return () => {
   452→      active = false;
   453→    };
   454→  }, [runtime, topicId]);
   455→
   456→  // شهرها فقط وقتی شیت فیلتر باز می‌شود دریافت می‌شوند (کش ۵ دقیقه‌ای)
   457→  useEffect(() => {
   458→    if (!filtersOpen || cities.length > 0) return;
   459→    let active = true;
   460→    getTeacherCities(runtime)
   461→      .then((list) => {
   462→        if (active) setCities(list);
   463→      })
   464→      .catch(() => {});
   465→    return () => {
   466→      active = false;
   467→    };
   468→  }, [filtersOpen, cities.length, runtime]);
   469→
   470→  const filtered = useMemo(() => {
   471→    let out = rows;
   472→    if (filters.tier) out = out.filter((teacher) => teacher.tier === filters.tier);
   473→    if (filters.availableOnly) out = out.filter((teacher) => teacher.available);
   474→    return sortTeachers(out, filters.sort);
   475→  }, [rows, filters.tier, filters.availableOnly, filters.sort]);
   476→
   477→  const visible = useMemo(() => filtered.slice(0, visibleCount), [filtered, visibleCount]);
   478→  const hasMore = visibleCount < filtered.length;
   479→  const hasAdvancedFilter = Boolean(filters.city || filters.female !== "all" || filters.availableOnly || filters.tier);
   480→  const hasAnyFilter = Boolean(filters.query || filters.field || hasAdvancedFilter || topicId);
   481→  const filteredCities = useMemo(() => {
   482→    const needle = cityQuery.trim();
   483→    return needle ? cities.filter((city) => city.includes(needle)) : cities;
   484→  }, [cities, cityQuery]);
   485→
   486→  // پیشنهاد جایگزین برای حالت خالی: پرمخاطب‌ترین استادها
   487→  useEffect(() => {
   488→    if (loading || error || filtered.length > 0 || !hasAnyFilter) {
   489→      setFallback([]);
   490→      return;
   491→    }
   492→    let active = true;
   493→    getAllTeachersCached(runtime)
   494→      .then((items) => {
   495→        if (!active) return;
   496→        setFallback(sortTeachers(items, "rating").slice(0, 4));
   497→      })
   498→      .catch(() => {});
   499→    return () => {
   500→      active = false;
   501→    };
   502→  }, [runtime, loading, error, filtered.length, hasAnyFilter]);
   503→
   504→  // بارگذاری خودکار نتایج بعدی هنگام اسکرول
   505→  useEffect(() => {
   506→    const node = sentinelRef.current;
   507→    if (!node || !hasMore || loading) return;
   508→    const observer = new IntersectionObserver(
   509→      (entries) => {
   510→        if (entries.some((entry) => entry.isIntersecting)) setVisibleCount((current) => current + PAGE_SIZE);
   511→      },
   512→      { rootMargin: "240px" },
   513→    );
   514→    observer.observe(node);
   515→    return () => observer.disconnect();
   516→  }, [hasMore, loading, visibleCount]);
   517→
