    55→function tehranHour() {
    56→  try {
    57→    return Number(
    58→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),
    59→    );
    60→  } catch {
    61→    return new Date().getHours();
    62→  }
    63→}

    69→  if (hour < 20) return "عصر بخیر";
    70→  return "شب بخیر";

    77→function dayOfYearIndex() {
    78→  const now = new Date();
    79→  const start = new Date(now.getFullYear(), 0, 0);

    83→function tehranDayKey(dateIso: string) {
    84→  if (!dateIso) return "";
    85→  try {
    86→    return new Intl.DateTimeFormat("en-US", {
    87→      timeZone: "Asia/Tehran",
    88→      year: "numeric",
    89→      month: "2-digit",
    90→      day: "2-digit",
    91→    }).format(new Date(dateIso));
    92→  } catch {
    93→    return "";
    94→  }
    95→}
