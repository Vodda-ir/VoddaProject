    60→function parseService(value: unknown): TeacherService | null {
    61→  const row = asRecord(value);
    62→  if (!row) return null;
    63→  const id = asInteger(row.id);
    64→  if (id <= 0) return null;
    65→
    66→  return {
    67→    id,
    68→    official_service_id: row.official_service_id == null ? null : asInteger(row.official_service_id) || null,
    69→    title: asString(row.title, "خدمت"),
    70→    duration: asInteger(row.duration, 30),
    71→    price: asInteger(row.price),
    72→    active: row.active === true,
    73→  };
    74→}
    75→
    76→async function getServices(runtime: PlatformRuntime): Promise<TeacherService[]> {
    77→  const response = await runtime.request<ApiEnvelope<UnknownRecord>>("/teacher/me");
    78→  const payload = asRecord(response.data);
    79→  return asArray(payload?.services).map(parseService).filter((item): item is TeacherService => item !== null);
    80→}

   105→const IN_PERSON_CATEGORY = "خدمات حضوری قرآنی";

   107→async function getCatalog(runtime: PlatformRuntime): Promise<CatalogCategory[]> {
   108→  // فقط خدمات آنلاین — دستهٔ «خدمات حضوری قرآنی» حذف می‌شود تا ثبت خدمت
   109→  // حضوری فقط از مسیر مخصوص خودش («خدمات حضوری» /t/presence با موجودیت و
   110→  // قیمتی مستقل) انجام شود. قبلاً همین دسته این‌جا هم نمایش داده می‌شد و
   111→  // استادها را گیج می‌کرد (دو مسیر برای یک کار).
   112→  const response = await runtime.request<ApiEnvelope<UnknownRecord>>("/catalog/services");
   113→  const payload = asRecord(response.data);
   114→  const categories: CatalogCategory[] = [];
   115→  for (const group of asArray(payload?.categories)) {
   116→    const row = asRecord(group);
   117→    if (!row) continue;
   118→    const category = asString(row.category, "خدمات رسمی");
   119→    if (category === IN_PERSON_CATEGORY) continue;
   120→    const services: CatalogEntry[] = [];
   121→    for (const item of asArray(row.services)) {
   122→      const entry = asRecord(item);
   123→      if (!entry) continue;
   124→      const id = asInteger(entry.id);
   125→      if (id <= 0) continue;
   126→      services.push({
   127→        id,
   128→        category,
   129→        title: asString(entry.title, "خدمت رسمی"),
   130→        description: asString(entry.description, ""),
   131→        icon: entry.icon == null ? undefined : asString(entry.icon, undefined as unknown as string),
   132→      });
   133→    }
   134→    if (services.length > 0) categories.push({ category, services });
   135→  }
   136→  return categories;
   137→}

   197→function normalizePersian(value: string): string {
   198→  return value
   199→    .replace(/[\u200c\u064b-\u0652]/g, "")
   200→    .replace(/[أإآ]/g, "ا")
   201→    .replace(/ة/g, "ه")
   202→    .replace(/ى/g, "ی")
   203→    .replace(/ك/g, "ک")
   204→    .trim()
   205→    .toLowerCase();
   206→}

   349→  async function saveService() {
   350→    if (!editing || saving) return;
   351→    const duration = Number.parseInt(draft.duration.trim(), 10);
   352→    const price = Number.parseInt(draft.price.trim(), 10);
   353→
   354→    if (!Number.isFinite(duration) || duration < 10 || duration > 240) {
   355→      setFormError("مدت خدمت باید بین ۱۰ تا ۲۴۰ دقیقه باشد.");

   359→    if (!Number.isFinite(price) || price < 0) {
   360→      setFormError("مبلغ معتبر وارد کنید.");
   361→      return;
   362→    }

   364→    setSaving(true);
   365→    setFormError(null);
   366→    try {
   367→      await updateService(runtime, editing.id, { duration, price });
