    58→  level: string;

    62→  price30: string;

    65→
    66→function parseUser(value: unknown): UserProfile | null {
    67→  const record = asRecord(value);
    68→  if (!record) return null;
    69→  return {
    70→    name: asString(record.name).trim() || "استاد",
    71→    avatarUrl: asString(record.avatar_url) || undefined,
    72→  };
    73→}
    74→
    75→function parseTeacher(value: unknown): TeacherProfile | null {
    76→  const record = asRecord(value);
    77→  if (!record) return null;
    78→  const services = asArray(record.services)
    79→    .map(asRecord)
    80→    .filter((item): item is Record<string, unknown> => item !== null);
    81→  const skills = asArray(record.skills)
    82→    .filter((item): item is string => typeof item === "string")
    83→    .map((item) => item.trim())
    84→    .filter(Boolean);
    85→
    86→  return {
    87→    id: asNumber(record.id),
    88→    field: asString(record.field),
    89→    tag: asString(record.tag),
    90→    level: asString(record.level),
    91→    city: asString(record.city),
    92→    bio: asString(record.bio),
    93→    skills,
    94→    price30: asNumber(record.price30),
    95→    price60: asNumber(record.price60),
    96→    available: record.available === true,
    97→    status: asString(record.status) || "pending",
    98→    verified: record.verified === true,
    99→    tier: asString(record.tier),
   100→    rating: asNumber(record.rating),
   101→    reviews: asNumber(record.reviews),
   102→    sessions: asNumber(record.sessions),
   103→    services: services.map((service) => ({ id: asNumber(service.id), active: service.active === true })),
   104→  };
   105→}
   106→
   107→function toForm(teacher: TeacherProfile): TeacherForm {
   108→  return {
   109→    field: teacher.field,
   110→    tag: teacher.tag,
   111→    level: teacher.level,
   112→    city: teacher.city,
   113→    bio: teacher.bio,
   114→    skills: teacher.skills.join("، "),
   115→    price30: teacher.price30 > 0 ? String(teacher.price30) : "",
   116→    price60: teacher.price60 > 0 ? String(teacher.price60) : "",
   117→  };
   118→}

   121→// آن‌ها را به بند/چیپ مرتب تبدیل می‌کند؛ نتیجه همیشه خوش‌فرم می‌ماند.
   122→const BIO_TEMPLATE = "📖 معرفی کوتاه خود را اینجا بنویسید؛ سابقه، سبک تدریس و دستاورد‌ها…\n🎯 زمینه‌های تدریس:\n• روخوانی و تجوید قرآن کریم\n• حفظ قرآن به روش تکرار هوشمند\n• آموزش مقدماتی تجوید\n📍 مدرک یا اجازه | شهر";

   124→function validate(form: TeacherForm): string | null {
   125→  if (form.field.trim().length > 60) return "حوزه تخصص حداکثر ۶۰ کاراکتر است.";
   126→  if (form.tag.trim().length > 60) return "عنوان استاد حداکثر ۶۰ کاراکتر است.";
   127→  if (form.level.trim().length > 30) return "سطح حداکثر ۳۰ کاراکتر است.";
   128→  if (form.city.trim().length > 80) return "شهر حداکثر ۸۰ کاراکتر است.";
   129→  if (form.bio.trim().length > 1000) return "معرفی استاد حداکثر ۱۰۰۰ کاراکتر است.";
   130→
   131→  const skills = form.skills.split(/[،,]/).map((item) => item.trim()).filter(Boolean);
   132→  if (skills.length > 20) return "حداکثر ۲۰ مهارت قابل ثبت است.";
   133→  if (skills.some((skill) => skill.length > 50)) return "هر مهارت حداکثر ۵۰ کاراکتر است.";
   134→
   135→  for (const [label, raw] of [["تعرفه ۳۰ دقیقه", form.price30], ["تعرفه ۶۰ دقیقه", form.price60]] as const) {
   136→    if (!raw.trim()) continue;
   137→    const value = Number(raw);
   138→    if (!Number.isFinite(value) || value <= 0 || value > 1_000_000_000) {
   139→      return `${label} باید عددی مثبت و حداکثر یک میلیارد تومان باشد.`;
   140→    }
   141→  }
   142→  return null;
   143→}
