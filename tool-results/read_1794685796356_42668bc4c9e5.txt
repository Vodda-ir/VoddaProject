    70→    70→type CatalogEntry = { id: number; category: string; title: string; description: string };

    72→    72→async function fetchCatalog(): Promise<CatalogEntry[]> {
    73→    73→  const response = await fetch("/api/backend/catalog/services", { cache: "no-store" });
    74→    74→  if (!response.ok) throw new Error("catalog failed");
    75→    75→  const payload = (await response.json()) as { data?: unknown };
    76→    76→  const data = asRecord(payload.data);
    77→    77→  const entries: CatalogEntry[] = [];
    78→    78→  for (const group of asArray(data?.categories)) {
    79→    79→    const row = asRecord(group);
    80→    80→    const category = String(row?.category ?? "");
    81→    81→    for (const item of asArray(row?.services)) {
    82→    82→      const entry = asRecord(item);
    83→    83→      const id = Number(entry?.id ?? 0);
    84→    84→      if (id > 0 && entry?.title) {
    85→    85→        entries.push({ id, category, title: String(entry.title), description: String(entry.description ?? "") });
    86→    86→      }
    87→    87→    }
    88→    88→  }
    89→    89→  return entries;
    90→    90→}

   466→   466→async function fetchFreeServices(): Promise<FreeServiceCardData[]> {
   467→   467→  const response = await fetch("/api/backend/catalog/free?limit=12", { cache: "no-store" });
   468→   468→  if (!response.ok) throw new Error("free services failed");
   469→   469→  const payload = (await response.json()) as { data?: unknown };
   470→   470→  const rows: FreeServiceCardData[] = [];
   471→   471→  for (const item of asArray(payload.data)) {
   472→   472→    const row = asRecord(item);
   473→   473→    const serviceId = Number(row?.service_id ?? 0);
   474→   474→    const teacherId = Number(row?.teacher_id ?? 0);
   475→   475→    const title = String(row?.title ?? "");
   476→   476→    const teacherName = String(row?.teacher_name ?? "");
   477→   477→    if (serviceId > 0 && teacherId > 0 && title && teacherName) {
   478→   478→      rows.push({
   479→   479→        service_id: serviceId,
   480→   480→        title,
   481→   481→        duration: Number(row?.duration ?? 0),
   482→   482→        teacher_id: teacherId,
   483→   483→        teacher_name: teacherName,
   484→   484→        avatar_url: String(row?.avatar_url ?? ""),
   485→   485→        field: String(row?.field ?? ""),
   486→   486→        tag: String(row?.tag ?? ""),
   487→   487→        city: String(row?.city ?? ""),
   488→   488→        rating: Number(row?.rating ?? 0),
   489→   489→        reviews: Number(row?.reviews ?? 0),
   490→   490→        sessions: Number(row?.sessions ?? 0),
   491→   491→      });
   492→   492→    }
   493→   493→  }
   494→   494→  return rows;
   495→   495→}
