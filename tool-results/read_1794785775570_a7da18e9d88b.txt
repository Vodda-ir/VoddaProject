   210→   210→export function AmountPicker({
   211→   211→  label,
   212→   212→  hint,
   213→   213→  value,
   214→   214→  onChange,
   215→   215→  presets,
   216→   216→  ariaLabel,
   217→   217→  emptyHint,
   218→   218→}: {
   219→   219→  label: string;
   220→   220→  hint?: string;
   221→   221→  value: string;
   222→   222→  onChange: (next: string) => void;
   223→   223→  presets: readonly number[];
   224→   224→  ariaLabel: string;
   225→   225→  emptyHint?: string;
   226→   226→}) {
   227→   227→  const numeric = Number(digitsOnly(value) || 0);
   228→   228→  const [customMode, setCustomMode] = useState(false);
   229→   229→  const uniquePresets = useMemo(
   230→   230→    () => [...new Set(presets.filter((preset) => preset > 0))].sort((a, b) => a - b),
   231→   231→    [presets],
   232→   232→  );
   233→   233→  const showManual = customMode || (numeric > 0 && !uniquePresets.includes(numeric));
   234→   234→  const selectValue = numeric > 0
   235→   235→    ? uniquePresets.includes(numeric) && !customMode
   236→   236→      ? String(numeric)
   237→   237→      : "__custom__"
   238→   238→    : "";
   239→   239→  return (
   240→   240→    <FieldShell label={label} hint={hint}>
   241→   241→      {uniquePresets.length > 0 ? (
   242→   242→        <SelectField
   243→   243→          ariaLabel={ariaLabel}
   244→   244→          value={selectValue}
   245→   245→          onChange={(next) => {
   246→   246→            if (next === "__custom__") {
   247→   247→              setCustomMode(true);
   248→   248→              return;
   249→   249→            }
   250→   250→            setCustomMode(false);
   251→   251→            onChange(next);
   252→   252→          }}
   253→   253→          options={[
   254→   254→            { value: "", label: "مبلغ توافقی — بدون مبلغ ثابت" },
   255→   255→            ...uniquePresets.map((preset) => ({ value: String(preset), label: formatToman(preset) })),
   256→   256→            { value: "__custom__", label: "مبلغ دلخواه…" },
   257→   257→          ]}
   258→   258→        />
   259→   259→      ) : null}
   260→   260→      {showManual || uniquePresets.length === 0 ? (
   261→   261→        <AmountInput value={value} onChange={onChange} ariaLabel={ariaLabel} />
   262→   262→      ) : null}
   263→   263→      {!value && emptyHint ? (
   264→   264→        <span style={{ display: "block", fontSize: 11, color: colors.inkFaint, lineHeight: 1.8 }}>{emptyHint}</span>
   265→   265→      ) : null}
   266→   266→    </FieldShell>
   267→   267→  );
   268→   268→}

   284→   284→function activeBudgetKey(minValue: number, maxValue: number): string {
   285→   285→  for (const tier of BUDGET_TIERS) {
   286→   286→    if (tier.key === "custom") continue;
   287→   287→    if (tier.min === minValue && tier.max === maxValue) return tier.key;
   288→   288→  }
   289→   289→  return minValue > 0 || maxValue > 0 ? "custom" : "open";
   290→   290→}

   292→   292→/** بازهٔ حق‌الزحمهٔ فراخوان مراسم — فهرست کشویی پیشنهادی + بازهٔ دلخواه. */
   293→   293→export function PriceRangePicker({
   294→   294→  minValue,
   295→   295→  maxValue,
   296→   296→  onChange,
   297→   297→}: {
   298→   298→  minValue: number;
   299→   299→  maxValue: number;
   300→   300→  onChange: (next: { min: string; max: string }) => void;
   301→   301→}) {
   302→   302→  // بازهٔ دلخواه حتی با مقادیر خالی هم باید فعال بماند — پس علاوه بر تطبیق
   303→   303→  // مقادیر، یک وضعیت محلی هم نگه می‌داریم.
   304→   304→  const [customMode, setCustomMode] = useState(false);
   305→   305→  const activeKey = customMode ? "custom" : activeBudgetKey(minValue, maxValue);
   306→   306→  const select = (tier: BudgetTier) => {
   307→   307→    if (tier.key === "custom") {
   308→   308→      setCustomMode(true);
   309→   309→      onChange({ min: minValue > 0 ? String(minValue) : "", max: maxValue > 0 ? String(maxValue) : "" });
   310→   310→      return;
   311→   311→    }
   312→   312→    setCustomMode(false);
   313→   313→    onChange({ min: tier.min > 0 ? String(tier.min) : "", max: tier.max > 0 ? String(tier.max) : "" });
   314→   314→  };
   315→   315→  return (
   316→   316→    <FieldShell label="بودجهٔ پیشنهادی" hint="حق‌الزحمهٔ منصفانه برای قاریان">
   317→   317→      <SelectField
   318→   318→        ariaLabel="بازهٔ بودجهٔ پیشنهادی"
   319→   319→        value={activeKey}
   320→   320→        onChange={(next) => select(BUDGET_TIERS.find((tier) => tier.key === next) ?? BUDGET_TIERS[0])}
   321→   321→        options={BUDGET_TIERS.map((tier) => ({
   322→   322→          value: tier.key,
   323→   323→          label: tier.key === "custom" ? "بازهٔ دلخواه — تعیین دقیق کف و سقف" : `${tier.label} — ${tier.sub}`,
   324→   324→        }))}
   325→   325→      />
   326→   326→
   327→   327→      {activeKey === "custom" ? (
   328→   328→        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: spacing.md, marginTop: spacing.xs }}>
   329→   329→          <AmountInput value={minValue > 0 ? String(minValue) : ""} onChange={(next) => onChange({ min: next, max: String(maxValue || "") })} ariaLabel="حداقل بودجه به تومان" placeholder="حداقل" />
   330→   330→          <AmountInput value={maxValue > 0 ? String(maxValue) : ""} onChange={(next) => onChange({ min: String(minValue || ""), max: next })} ariaLabel="حداکثر بودجه به تومان" placeholder="حداکثر" />
   331→   331→        </div>
   332→   332→      ) : null}
   333→   333→
   334→   334→      <span style={{ display: "block", fontSize: 11.5, color: colors.inkMuted, marginTop: 2 }}>
   335→   335→        بازهٔ اعلام‌شده به قاریان: {formatBudgetRange(minValue, maxValue)}
   336→   336→      </span>
   337→   337→    </FieldShell>
   338→   338→  );
   339→   339→}

   343→   343→export function CountField({
   344→   344→  label,
   345→   345→  hint,
   346→   346→  value,
   347→   347→  onChange,
   348→   348→  presets,
   349→   349→  min,
   350→   350→  max,
   351→   351→  step = 1,
   352→   352→  unit = "نفر",
   353→   353→  ariaLabel,
   354→   354→}: {
   355→   355→  label: string;
   356→   356→  hint?: string;
   357→   357→  value: string;
   358→   358→  onChange: (next: string) => void;
   359→   359→  presets: readonly number[];
   360→   360→  min: number;
   361→   361→  max: number;
   362→   362→  step?: number;
   363→   363→  unit?: string;
   364→   364→  ariaLabel: string;
   365→   365→}) {
   366→   366→  const numeric = Number(digitsOnly(value) || 0);
   367→   367→  const set = (next: number) => onChange(String(Math.min(max, Math.max(min, Math.round(next)))));
   368→   368→  const presetOptions = presets.filter((preset) => preset >= min && preset <= max);
   369→   369→  const isPreset = presetOptions.includes(numeric);
   370→   370→  const stepButtonStyle: CSSProperties = {
   371→   371→    width: 46,
   372→   372→    height: 46,
   373→   373→    display: "grid",
   374→   374→    placeItems: "center",
   375→   375→    flex: "0 0 46px",
   376→   376→    borderRadius: 14,
   377→   377→    border: `1px solid ${colors.sand200}`,
   378→   378→    background: colors.surface,
   379→   379→    color: colors.olive700,
   380→   380→    cursor: "pointer",
   381→   381→    fontFamily: typography.fontFamily,
   382→   382→  };
   383→   383→  return (
   384→   384→    <FieldShell label={label} hint={hint}>
   385→   385→      <SelectField
   386→   386→        ariaLabel={ariaLabel}
   387→   387→        value={numeric > 0 ? String(numeric) : ""}
   388→   388→        onChange={(next) => {
   389→   389→          if (next) set(Number(next));
   390→   390→        }}
   391→   391→        options={[
   392→   392→          ...(numeric > 0 ? [] : [{ value: "", label: "انتخاب کنید…" }]),
   393→   393→          ...presetOptions.map((preset) => ({ value: String(preset), label: `${toFaDigits(preset)} ${unit}` })),
   394→   394→          ...(numeric > 0 && !isPreset ? [{ value: String(numeric), label: `${toFaDigits(numeric)} ${unit} (دلخواه)` }] : []),
   395→   395→        ]}
   396→   396→      />
   397→   397→      <div style={{ display: "flex", alignItems: "center", gap: spacing.md }}>
   398→   398→        <button
   399→   399→          type="button"
   400→   400→          aria-label={`کاهش ${ariaLabel}`}
   401→   401→          onClick={() => set((numeric > 0 ? numeric : min) - step)}
   402→   402→          disabled={numeric <= min}
   403→   403→          style={{ ...stepButtonStyle, opacity: numeric <= min ? 0.45 : 1 }}
   404→   404→        >

   491→   491→function getCurrentPosition(): Promise<GeolocationPosition> {
   492→   492→  return new Promise((resolve, reject) => {
   493→   493→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   494→   494→      reject(new Error("unsupported"));
   495→   495→      return;
   496→   496→    }
   497→   497→    navigator.geolocation.getCurrentPosition(
   498→   498→      resolve,
   499→   499→      reject,
   500→   500→      { enableHighAccuracy: true, timeout: 12000, maximumAge: 60_000 },
   501→   501→    );
   502→   502→  });
   503→   503→}

   505→   505→async function reverseGeocodeNominatim(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   506→   506→  const url = `https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${lat}&lon=${lon}&zoom=17&addressdetails=1&accept-language=fa`;
   507→   507→  const response = await fetch(url, { headers: { Accept: "application/json" } });
   508→   508→  if (!response.ok) return null;
   509→   509→  const data = (await response.json()) as Record<string, unknown>;
   510→   510→  const address = (data.address && typeof data.address === "object" ? data.address : {}) as Record<string, unknown>;
   511→   511→  const pick = (key: string): string => (typeof address[key] === "string" ? (address[key] as string).trim() : "");
   512→   512→  const city = pick("city") || pick("town") || pick("village") || pick("municipality") || pick("county") || pick("state");
   513→   513→  const parts = [pick("amenity"), pick("road"), pick("neighbourhood"), pick("quarter"), pick("suburb")]
   514→   514→    .filter((part) => part.length > 0);
   515→   515→  const unique = [...new Set(parts)];
   516→   516→  if (!city && unique.length === 0) return null;
   517→   517→  return { city, address: unique.join("، ") };
   518→   518→}

   520→   520→async function reverseGeocodeFallback(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   521→   521→  const url = `https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${lat}&longitude=${lon}&localityLanguage=fa`;
   522→   522→  const response = await fetch(url);
   523→   523→  if (!response.ok) return null;
   524→   524→  const data = (await response.json()) as Record<string, unknown>;
   525→   525→  const str = (value: unknown): string => (typeof value === "string" ? value.trim() : "");
   526→   526→  const city = str(data.city) || str(data.locality) || str(data.principalSubdivision);
   527→   527→  if (!city) return null;
   528→   528→  const locality = str(data.locality);
   529→   529→  return { city, address: locality && locality !== city ? locality : "" };
   530→   530→}

   532→   532→function geoErrorText(error: unknown): string {
   533→   533→  const code = typeof (error as GeolocationPositionError)?.code === "number" ? (error as GeolocationPositionError).code : 0;
   534→   534→  if (code === 1) return "دسترسی به موقعیت دستگاه داده نشد؛ مجوز موقعیت را در تنظیمات فعال کنید یا نشانی را دستی وارد کنید.";
   535→   535→  if (code === 3) return "یافتن موقعیت بیش از حد طول کشید؛ لطفاً دوباره تلاش کنید.";
   536→   536→  return "موقعیت دستگاه در دسترس نیست؛ نشانی را دستی وارد کنید.";
   537→   537→}

   807→   807→
   808→   808→export function NoteTemplateChips({ value, onChange }: { value: string; onChange: (next: string) => void }) {
   809→   809→  const toggle = (text: string) => {
   810→   810→    const current = value.trim();
   811→   811→    if (current.includes(text)) {
   812→   812→      const next = current
   813→   813→        .replace(`؛ ${text}`, "")
   814→   814→        .replace(`؛ ${text}؛`, "")
   815→   815→        .replace(`${text}؛ `, "")
   816→   816→        .replace(text, "")
   817→   817→        .replace(/؛\s*؛/g, "؛")
   818→   818→        .replace(/^[؛\s]+|[؛\s]+$/g, "")
   819→   819→        .trim();
   820→   820→      onChange(next);
   821→   821→      return;
   822→   822→    }
   823→   823→    onChange(current ? `${current}؛ ${text}` : text);
   824→   824→  };
   825→   825→  return (
   826→   826→    <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   827→   827→      {NOTE_TEMPLATES.map((template) => (
   828→   828→        <Chip key={template.chip} active={value.includes(template.text)} onClick={() => toggle(template.text)}>
   829→   829→          {template.chip}
   830→   830→        </Chip>
   831→   831→      ))}
   832→   832→    </div>
   833→   833→  );
   834→   834→}

   838→   838→export function DurationPicker({ value, onChange }: { value: string; onChange: (next: string) => void }) {
   839→   839→  const presets: readonly number[] = [30, 45, 60, 90, 120, 180];
   840→   840→  const numeric = Number(digitsOnly(value) || 0);
   841→   841→  const isPreset = presets.includes(numeric);
   842→   842→  return (
   843→   843→    <FieldShell label="مدت اجرا" hint="دقیقه">
   844→   844→      <SelectField
   845→   845→        ariaLabel="مدت اجرا به دقیقه"
   846→   846→        value={numeric > 0 && isPreset ? String(numeric) : numeric > 0 ? "__custom__" : ""}
   847→   847→        onChange={(next) => {
   848→   848→          if (next && next !== "__custom__") onChange(next);
   849→   849→        }}
   850→   850→        options={[
   851→   851→          { value: "", label: "انتخاب مدت اجرا…" },
   852→   852→          ...presets.map((preset) => ({ value: String(preset), label: `${toFaDigits(preset)} دقیقه` })),

   854→   854→        ]}

   858→   858→          type="text"
   859→   859→          inputMode="numeric"
   860→   860→          value={toFaDigits(value)}
   861→   861→          onChange={(event) => onChange(digitsOnly(event.target.value, 4))}
   862→   862→          placeholder="دقیقهٔ دلخواه…"
   863→   863→          style={{ ...inputStyle, minHeight: 46 }}
   864→   864→          aria-label="مدت اجرا به دقیقه"

   867→   867→    </FieldShell>
   868→   868→  );
   869→   869→}

   873→   873→export function CitiesChipsField({

   875→   875→  onChange,
   876→   876→  label = "شهرهای پذیرش مراسم",
