    49→function fa(n: number, lang: "fa" | "en"): string {
    50→  return new Intl.NumberFormat(lang === "en" ? "en-US" : "fa-IR").format(n);
    51→}

    53→function priceLabel(n: number, lang: "fa" | "en", L: (text: string) => string): string {
    54→  return n === 0 ? L("رایگان") : L("{n} تومان").replace("{n}", localizedDigits(formatWalletAmount(Math.max(0, n)), lang));
    55→}
    56→
    57→function timeLabelOf(
    58→  call: { flexible: boolean; scheduled_at?: string | null },
    59→  lang: "fa" | "en",
    60→  L: (text: string) => string,
    61→): string {
    62→  if (call.flexible || !call.scheduled_at) return L("جلسهٔ فوری — همین حالا");
    63→  const at = new Date(call.scheduled_at);
    64→  if (Number.isNaN(at.getTime())) return L("در اسرع وقت");
    65→  const now = new Date();
    66→  const diffMin = Math.round((at.getTime() - now.getTime()) / 60000);
    67→  if (diffMin < 60 && diffMin > -60) return L("حدود {n} دقیقهٔ دیگر").replace("{n}", fa(Math.max(0, diffMin), lang));
    68→  return formatAppDateTime(call.scheduled_at, lang, { withWeekday: true });
    69→}
    70→
    71→function remainingSeconds(expiresAt: string): number {
    72→  const at = new Date(expiresAt).getTime();
    73→  if (Number.isNaN(at)) return 0;
    74→  return Math.max(0, Math.round((at - Date.now()) / 1000));
    75→}

    77→function countdownLabel(total: number, lang: "fa" | "en"): string {
    78→  const m = Math.floor(total / 60);
    79→  const s = total % 60;
    80→  return `${localizedDigits(String(m).padStart(2, "0"), lang)}:${localizedDigits(String(s).padStart(2, "0"), lang)}`;
    81→}

   200→export function LearnerDispatchScreen({ config }: { config: UniversalAppConfig }) {
   201→  const runtime = usePlatform();
   202→  const { L, lang } = useAppearance();
   203→  const [catalog, setCatalog] = useState<OfficialService[]>([]);
   204→  const [catalogLoading, setCatalogLoading] = useState(true);
   205→  const [catalogError, setCatalogError] = useState("");
   206→  const [query, setQuery] = useState("");
   207→  const [selectedTopic, setSelectedTopic] = useState<OfficialService | null>(null);
   208→  const [note, setNote] = useState("");
   209→  const [whenMode, setWhenMode] = useState<"now" | "later">("now");
   210→  const [when, setWhen] = useState<WhenValue>(() => {
   211→    const nowHour = new Date().getHours();
   212→    return { localDate: upcomingJalaliDays(1)[0]?.localDate ?? "", hh: Math.min(Math.max(nowHour + 2, 8), 22), mm: 0 };
   213→  });
   214→  const [creating, setCreating] = useState(false);
   215→  const [actionError, setActionError] = useState("");
   216→  const [view, setView] = useState<DispatchCallView | null>(null);
   217→  const [busyOffer, setBusyOffer] = useState(0);
   218→  const [canceling, setCanceling] = useState(false);
   219→  const [resuming, setResuming] = useState(true);
   220→  const [payMethod, setPayMethod] = useState<"wallet" | "gateway">("gateway");
   221→  const [tick, setTick] = useState(0);
   222→  const pollRef = useRef<number | null>(null);
   223→  /** فراخوانی که کاربر با «فراخوان جدید» از آن عبور کرده — تا کارت موفقیت
   224→   *  دوباره روی آهنگ‌ساز تحمیل نشود و بتوان فراخوان تازه ساخت. */
   225→  const dismissedCallIdRef = useRef(0);
   226→
   227→  const call = view?.call ?? null;
   228→  const mode: ScreenMode = call ? (call.status === "searching" ? "searching" : "matched") : "idle";
   229→
   230→  const loadCatalog = useCallback(async () => {
   231→    setCatalogLoading(true);
   232→    setCatalogError("");
   233→    try {
   234→      setCatalog(await getCatalog(runtime));
   235→    } catch (error) {
   236→      setCatalogError(messageOf(error, "دریافت فهرست موضوعات ناموفق بود."));
   237→    } finally {
   238→      setCatalogLoading(false);
   239→    }
   240→  }, [runtime]);
   241→
   242→  const refresh = useCallback(async () => {
   243→    try {
   244→      const current = await getCurrentDispatchCall(runtime);
   245→      const dismissed = dismissedCallIdRef.current;
   246→      if (current.call && dismissed > 0 && current.call.id === dismissed && current.call.status !== "searching") {
   247→        // کاربر آگاهانه «فراخوان جدید» را زده — آهنگ‌ساز باید باز بماند
   248→        setView({ call: null, candidates: [] });
   249→      } else {
   250→        setView(current);
   251→      }
   252→    } catch {
   253→      // polling errors are non-fatal
   254→    } finally {
   255→      setResuming(false);
   256→    }
   257→  }, [runtime]);
   258→
   259→  useEffect(() => {
   260→    void loadCatalog();
   261→    void refresh();
   262→  }, [loadCatalog, refresh]);
   263→
   264→  // Live polling while a call is active.
   265→  useEffect(() => {
   266→    if (mode === "idle") return;
   267→    pollRef.current = window.setInterval(() => {
   268→      void refresh();
   269→      setTick((value) => value + 1);
   270→    }, 2500);
   271→    return () => {
   272→      if (pollRef.current) window.clearInterval(pollRef.current);
   273→      pollRef.current = null;
   274→    };
   275→  }, [mode, refresh]);
   276→
   277→  const secondsLeft = useMemo(
   278→    () => (call ? remainingSeconds(call.expires_at) : 0),
   279→    // recompute when the poll tick advances so the countdown stays live
   280→    [call, tick],
   281→  );
   282→
   283→  const filteredCatalog = useMemo(() => {
   284→    const q = query.trim();
   285→    if (!q) return catalog;
   286→    return catalog.filter(
   287→      (item) => item.title.includes(q) || item.category.includes(q) || item.description.includes(q),
   288→    );
   289→  }, [catalog, query]);
   290→
   291→  const grouped = useMemo(() => {
   292→    const groups = new Map<string, OfficialService[]>();
   293→    for (const item of filteredCatalog) {
   294→      const rows = groups.get(item.category) ?? [];
   295→      rows.push(item);
   296→      groups.set(item.category, rows);
   297→    }
   298→    return [...groups.entries()];
   299→  }, [filteredCatalog]);
   300→
   301→  /* گزینه‌های شمسی «زمان مشخص» — هم‌سو با فرم مراسم (A3/B1) */
   302→  const dayOptions = useMemo(
   303→    () =>
   304→      upcomingJalaliDays(7).map((day, index) => {
   305→        const [gy, gm, gd] = day.localDate.split("-").map(Number);
   306→        const weekdayIndex = persianWeekdayIndex(new Date(gy, gm - 1, gd).getDay());
   307→        if (lang === "en") {
   308→          const dateLabel = formatAppDate(day.localDate, "en", { withWeekday: weekdayIndex, withYear: false });
   309→          const prefix = index === 0 ? L("امروز") : index === 1 ? L("فردا") : "";
   310→          return { value: day.localDate, label: prefix ? `${prefix} — ${dateLabel}` : dateLabel };
   311→        }
   312→        const prefix = index === 0 ? L("امروز") : index === 1 ? L("فردا") : day.weekday;
   313→        return { value: day.localDate, label: `${prefix} — ${day.dayLabel}` };
   314→      }),
   315→    [lang, L],
   316→  );
   317→
   318→  const timeOptions = useMemo(() => {
   319→    const options: Array<{ value: string; label: string }> = [];
   320→    for (let hh = 8; hh <= 23; hh += 1) {
   321→      for (const mm of [0, 30]) {
   322→        options.push({
   323→          value: `${hh}:${mm}`,
   324→          label: `${localizedDigits(String(hh).padStart(2, "0"), lang)}:${localizedDigits(String(mm).padStart(2, "0"), lang)}`,
   325→        });
   326→      }
   327→    }
   328→    return options;
   329→  }, [lang]);
   330→
   331→  const start = useCallback(async () => {
   332→    if (!selectedTopic || creating) return;
   333→    if (whenMode === "later" && !isWhenInFuture(when)) {
   334→      setActionError(L("زمان انتخاب‌شده باید حداقل ۱۵ دقیقه در آینده باشد."));
   335→      return;
   336→    }
   337→    setCreating(true);
   338→    setActionError("");
   339→    try {
   340→      const scheduledAt = whenMode === "later" ? tehranWallToRfc3339(when) || null : null;
   341→      const created = await createDispatchCall(runtime, {
   342→        officialServiceId: selectedTopic.id,
   343→        note,
   344→        scheduledAt,
   345→        flexible: whenMode === "now",
   346→      });
   347→      showToast(L("فراخوان شما برای اساتید این موضوع ارسال شد"));
   348→      setView(created);
   349→    } catch (error) {
   350→      setActionError(messageOf(error, "ثبت فراخوان انجام نشد."));
   351→    } finally {
   352→      setCreating(false);
   353→    }
   354→  }, [creating, note, runtime, selectedTopic, when, whenMode, L]);
   355→
   356→  const cancel = useCallback(async () => {
   357→    if (!call || canceling) return;
   358→    setCanceling(true);
   359→    setActionError("");
   360→    try {
   361→      await cancelDispatchCall(runtime, call.id);

   363→    } catch (error) {
   364→      setActionError(messageOf(error, "لغو فراخوان انجام نشد."));
   365→    } finally {
   366→      setCanceling(false);
   367→    }
   368→  }, [call, canceling, runtime]);

   381→        if (result.token) {
   382→          // پست فرم توکن به OnlinePG از مرورگر کاربر — دقیقاً مثل منافسه، در همان تب

   384→        }
   385→      } catch (error) {

   388→        } else {
   389→          setActionError(messageOf(error, "ثبت رزرو انجام نشد."));

   391→        await refresh();
   392→      } finally {
   393→        setBusyOffer(0);
   394→      }
   395→    },

   405→        const updated = await declineDispatchOffer(runtime, call.id, offerId);

   410→        setBusyOffer(0);

   422→          <div
   423→            aria-hidden="true"
   424→            style={{
   425→              width: 76, height: 76, display: "grid", placeItems: "center", marginInline: "auto",
   426→              borderRadius: "50%", color: colors.surface, background: gradients.hero, boxShadow: shadows.olive,
   427→            }}
   428→          >
   429→            <Icon name="checkCircle" size={36} stroke={2.2} />
   430→          </div>
   431→          <div>
   432→            <h2 style={{ ...typography.title, margin: 0 }}>{L("جلسه شما رزرو شد 🎉")}</h2>
   433→            <p style={{ ...typography.body, color: colors.inkSecondary, marginBottom: 0, marginTop: 6 }}>
   434→              {L("موضوع «{topic}» با استاد {name} ثبت شد و در انتظار تأیید نهایی استاد است.")
   435→                .replace("{topic}", L(call.topic_title))
   436→                .replace("{name}", L(candidate?.name ?? "انتخاب‌شده"))}
   437→            </p>
   438→          </div>
   439→          {candidate ? (
   440→            <div style={{ display: "flex", alignItems: "center", gap: 12, textAlign: "right", background: colors.sand100, borderRadius: 16, padding: 12 }}>
   441→              <Avatar name={candidate.name} src={candidate.avatar_url || undefined} size={48} ring={false} />
   442→              <div style={{ minWidth: 0, flex: 1 }}>
   443→                <strong style={{ display: "block", fontSize: 14.5, color: colors.ink }}>{candidate.name}</strong>
   444→                <span style={{ fontSize: 12, color: colors.inkMuted }}>
   445→                  {priceLabel(candidate.price, lang, L)} · {timeLabelOf(call, lang, L)}
   446→                </span>
   447→              </div>
   448→            </div>
   449→          ) : null}
   450→          <Button fullWidth onClick={() => navigateApp("/sessions", config)}>{L("پیگیری در جلسات من")}</Button>
   451→          <Button
   452→            variant="secondary"
   453→            fullWidth
   454→            onClick={() => {
   455→              if (call) dismissedCallIdRef.current = call.id;
   456→              setView({ call: null, candidates: [] });
   457→              void refresh();
   458→            }}
   459→          >
   460→            {L("فراخوان جدید")}
   461→          </Button>
   462→        </Card>
   463→      </div>
   464→    );
   465→  }
   466→
   467→  // ── live searching state ──
   468→  if (mode === "searching" && call) {
   469→    return (
   470→      <div className="vodda-screen" style={{ display: "grid", gap: spacing.lg }}>
   471→        <div
   472→          className="vodda-enter"
   473→          style={{
   474→            position: "relative",
   475→            overflow: "hidden",
   476→            borderRadius: 22,
   477→            padding: 20,
   478→            color: colors.surface,
   479→            background: gradients.deep,
   480→            boxShadow: shadows.olive,
   481→          }}
   482→        >
   483→          <span aria-hidden="true" className="vodda-dispatch-pulse" style={{ position: "absolute", insetInlineStart: 18, top: 18 }} />
   484→          <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
   485→            <div style={{ display: "grid", placeItems: "center", width: 46, height: 46, borderRadius: 15, background: "rgba(255,255,255,.14)" }}>
   486→              <TopicIcon icon={call.topic_icon} size={24} />
