   214→   214→function activeBudgetKey(minValue: number, maxValue: number): string {
   215→   215→  for (const tier of BUDGET_TIERS) {

   218→   218→  }

   310→   310→export function CountField({
   311→   311→  label,
   312→   312→  hint,
   313→   313→  value,
   314→   314→  onChange,
   315→   315→  presets,
   316→   316→  min,
   317→   317→  max,
   318→   318→  step = 1,
   319→   319→  unit = "نفر",
   320→   320→  ariaLabel,
   321→   321→}: {
   322→   322→  label: string;
   323→   323→  hint?: string;
   324→   324→  value: string;
   325→   325→  onChange: (next: string) => void;

   327→   327→  min: number;

   334→   334→  const set = (next: number) => onChange(String(Math.min(max, Math.max(min, Math.round(next)))));
   335→   335→  const stepButtonStyle: CSSProperties = {
   336→   336→    width: 46,
   337→   337→    height: 46,
   338→   338→    display: "grid",

   349→   349→    <FieldShell label={label} hint={hint}>
   350→   350→      <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   351→   351→        {presets.map((preset) => (
   352→   352→          <Chip key={preset} active={numeric === preset} onClick={() => set(preset)}>
   353→   353→            {toFaDigits(preset)}
   354→   354→            {unit ? ` ${unit}` : ""}
   355→   355→          </Chip>
   356→   356→        ))}
   357→   357→      </div>
   358→   358→      <div style={{ display: "flex", alignItems: "center", gap: spacing.md }}>
   359→   359→        <button
   360→   360→          type="button"
   361→   361→          aria-label={`کاهش ${ariaLabel}`}
   362→   362→          onClick={() => set((numeric > 0 ? numeric : min) - step)}
   363→   363→          disabled={numeric <= min}
   364→   364→          style={{ ...stepButtonStyle, opacity: numeric <= min ? 0.45 : 1 }}
   365→   365→        >
   366→   366→          <Icon name="x" size={18} stroke={2.4} />
   367→   367→        </button>
   368→   368→        <strong

   376→   376→            fontVariantNumeric: "tabular-nums",
   377→   377→          }}
   378→   378→        >
   379→   379→          {numeric > 0 ? toFaDigits(numeric) : "—"}
   380→   380→          {numeric > 0 && unit ? <span style={{ fontSize: 12, fontWeight: 700, color: colors.inkMuted }}> {unit}</span> : null}
   381→   381→        </strong>
   382→   382→        <button
   383→   383→          type="button"
   384→   384→          aria-label={`افزایش ${ariaLabel}`}
   385→   385→          onClick={() => set((numeric > 0 ? numeric : min - step) + step)}
   386→   386→          disabled={numeric >= max}
   387→   387→          style={{ ...stepButtonStyle, opacity: numeric >= max ? 0.45 : 1 }}
   388→   388→        >
   389→   389→          <Icon name="plus" size={18} stroke={2.4} />
   390→   390→        </button>
   391→   391→      </div>
   392→   392→    </FieldShell>
   393→   393→  );
   394→   394→}

   445→   445→function getCurrentPosition(): Promise<GeolocationPosition> {
   446→   446→  return new Promise((resolve, reject) => {
   447→   447→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   448→   448→      reject(new Error("unsupported"));
   449→   449→      return;
   450→   450→    }
   451→   451→    navigator.geolocation.getCurrentPosition(
   452→   452→      resolve,
   453→   453→      reject,
   454→   454→      { enableHighAccuracy: true, timeout: 12000, maximumAge: 60_000 },
   455→   455→    );
   456→   456→  });
   457→   457→}

   459→   459→async function reverseGeocodeNominatim(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   460→   460→  const url = `https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${lat}&lon=${lon}&zoom=17&addressdetails=1&accept-language=fa`;
   461→   461→  const response = await fetch(url, { headers: { Accept: "application/json" } });
   462→   462→  if (!response.ok) return null;
   463→   463→  const data = (await response.json()) as Record<string, unknown>;
   464→   464→  const address = (data.address && typeof data.address === "object" ? data.address : {}) as Record<string, unknown>;
   465→   465→  const pick = (key: string): string => (typeof address[key] === "string" ? (address[key] as string).trim() : "");
   466→   466→  const city = pick("city") || pick("town") || pick("village") || pick("municipality") || pick("county") || pick("state");
   467→   467→  const parts = [pick("amenity"), pick("road"), pick("neighbourhood"), pick("quarter"), pick("suburb")]
   468→   468→    .filter((part) => part.length > 0);
   469→   469→  const unique = [...new Set(parts)];
   470→   470→  if (!city && unique.length === 0) return null;
   471→   471→  return { city, address: unique.join("، ") };
   472→   472→}

   474→   474→async function reverseGeocodeFallback(lat: number, lon: number): Promise<{ city: string; address: string } | null> {
   475→   475→  const url = `https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${lat}&longitude=${lon}&localityLanguage=fa`;
   476→   476→  const response = await fetch(url);
   477→   477→  if (!response.ok) return null;
   478→   478→  const data = (await response.json()) as Record<string, unknown>;
   479→   479→  const str = (value: unknown): string => (typeof value === "string" ? value.trim() : "");
   480→   480→  const city = str(data.city) || str(data.locality) || str(data.principalSubdivision);
   481→   481→  if (!city) return null;
   482→   482→  const locality = str(data.locality);
   483→   483→  return { city, address: locality && locality !== city ? locality : "" };
   484→   484→}

   486→   486→function geoErrorText(error: unknown): string {

   811→   811→
   812→   812→export function CitiesChipsField({
   813→   813→  value,
   814→   814→  onChange,
   815→   815→  label = "شهرهای پذیرش مراسم",
   816→   816→}: {
   817→   817→  value: string;
   818→   818→  onChange: (next: string) => void;
   819→   819→  label?: string;
   820→   820→}) {
   821→   821→  const selected = useMemo(
   822→   822→    () => value.split(/[،,]/).map((part) => part.trim()).filter(Boolean),
   823→   823→    [value],
   824→   824→  );
   825→   825→  const toggle = (cityOption: string) => {
   826→   826→    if (selected.includes(cityOption)) {
   827→   827→      onChange(selected.filter((city) => city !== cityOption).join("، "));
   828→   828→    } else {
   829→   829→      onChange([...selected, cityOption].join("، "));
   830→   830→    }
   831→   831→  };
   832→   832→  return (
   833→   833→    <FieldShell label={label} hint="با تراشه انتخاب کنید">
   834→   834→      <div style={{ display: "flex", flexWrap: "wrap", gap: spacing.sm }}>
   835→   835→        <Chip accent="gold" active={selected.length === 0} onClick={() => onChange("")}>
   836→   836→          همهٔ شهرها
   837→   837→        </Chip>
   838→   838→        {POPULAR_CITIES.map((cityOption) => (
   839→   839→          <Chip key={cityOption} active={selected.includes(cityOption)} onClick={() => toggle(cityOption)}>
   840→   840→            {cityOption}
   841→   841→          </Chip>
   842→   842→        ))}
   843→   843→      </div>
   844→   844→      <input
   845→   845→        type="text"
   846→   846→        value={value}
   847→   847→        onChange={(event) => onChange(event.target.value.slice(0, 200))}
   848→   848→        placeholder="شهرهای دیگر را با ویرگول اضافه کنید…"
   849→   849→        style={{ ...inputStyle, minHeight: 46 }}
   850→   850→        aria-label={label}
   851→   851→      />
   852→   852→    </FieldShell>
   853→   853→  );
   854→   854→}
