   448→function FeaturedTeacher({ teacher, config }: { teacher: Teacher; config: UniversalAppConfig }) {
   449→  const sample = teacher.samples.find((item) => item.url) ?? null;
   450→  const audioRef = useRef<HTMLAudioElement | null>(null);
   451→  const [playing, setPlaying] = useState(false);
   452→
   453→  useEffect(() => () => {
   454→    audioRef.current?.pause();
   455→  }, []);
   456→
   457→  const toggleSample = (event: { stopPropagation(): void }) => {
   458→    event.stopPropagation();
   459→    if (!sample) return;
   460→    if (!audioRef.current) {
   461→      audioRef.current = new Audio(resolveAssetUrl(sample.url) ?? sample.url);
   462→      audioRef.current.addEventListener("ended", () => setPlaying(false));
   463→    }
   464→    if (playing) {
   465→      audioRef.current.pause();
   466→      setPlaying(false);
   467→    } else {
   468→      void audioRef.current.play().catch(() => setPlaying(false));
   469→      setPlaying(true);
   470→    }
   471→  };
   472→
   473→  return (
   474→    <div
   475→      onClick={() => navigateApp(`/teacher/${teacher.id}`, config)}
   476→      className="vodda-enter"
   477→      style={{
   478→        width: "100%",
   479→        position: "relative",
   480→        overflow: "hidden",
