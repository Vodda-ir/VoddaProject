    73→function recommendationScore(teacher: Teacher) {
    74→  return teacher.rating * Math.log10(10 + Math.max(0, teacher.sessions));
    75→}

    77→function sortTeachers(rows: Teacher[], sort: SortKey): Teacher[] {
    78→  const copy = [...rows];
    79→  switch (sort) {
    80→    case "rating":
    81→      return copy.sort((a, b) => b.rating - a.rating);
    82→    case "sessions":
    83→      return copy.sort((a, b) => b.sessions - a.sessions);
    84→    case "newest":
    85→      return copy.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
    86→    default:
    87→      return copy.sort((a, b) => recommendationScore(b) - recommendationScore(a));
    88→  }
    89→}

    95→function readInitialFilters(): SearchFilters {
    96→  const empty: SearchFilters = {
    97→    query: "",
    98→    field: "",
    99→    city: "",
   100→    female: "all",
   101→    tier: "",
   102→    availableOnly: false,
   103→    sort: "recommended",
   104→  };
   105→  if (typeof window === "undefined") return empty;
   106→  const params = new URLSearchParams(window.location.search);
   107→  const femaleParam = params.get("female");
   108→  const tierParam = params.get("tier");
   109→  const sortParam = params.get("sort");
   110→  return {
   111→    query: params.get("q") ?? "",
   112→    field: params.get("field") ?? "",
   113→    city: params.get("city") ?? "",
   114→    female: femaleParam === "true" ? "female" : femaleParam === "false" ? "male" : "all",
   115→    tier: (["international", "grade2", "grade3"].includes(tierParam ?? "") ? tierParam : "") as TierKey,
   116→    availableOnly: params.get("available") === "true",
   117→    sort: (["recommended", "rating", "sessions", "newest"].includes(sortParam ?? "")
   118→      ? sortParam
   119→      : "recommended") as SortKey,
   120→  };
   121→}

   123→function readRecentSearches(): string[] {
   124→  if (typeof window === "undefined") return [];
   125→  try {
   126→    const raw = window.localStorage.getItem(RECENT_KEY);
   127→    const parsed = raw ? JSON.parse(raw) : [];
   128→    return Array.isArray(parsed)
   129→      ? parsed.filter((item): item is string => typeof item === "string" && item.length > 0).slice(0, RECENT_LIMIT)
   130→      : [];
   131→  } catch {
   132→    return [];
   133→  }
   134→}
