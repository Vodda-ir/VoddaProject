    56→function tehranHour() {
    57→  try {
    58→    return Number(
    59→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),

    78→function dayOfYearIndex() {
    79→  const now = new Date();
    80→  const start = new Date(now.getFullYear(), 0, 0);
    81→  return Math.floor((now.getTime() - start.getTime()) / 86_400_000);

    84→function tehranDayKey(dateIso: string) {
    85→  if (!dateIso) return "";
    86→  try {
    87→    return new Intl.DateTimeFormat("en-US", {
    88→      timeZone: "Asia/Tehran",
    89→      year: "numeric",
    90→      month: "2-digit",
    91→      day: "2-digit",
    92→    }).format(new Date(dateIso));
    93→  } catch {
    94→    return "";
    95→  }
    96→}
