    58→function tehranHour() {
    59→  try {
    60→    return Number(
    61→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),
    62→    );
    63→  } catch {
    64→    return new Date().getHours();
    65→  }

    80→function dayOfYearIndex() {
    81→  const now = new Date();
    82→  const start = new Date(now.getFullYear(), 0, 0);
    83→  return Math.floor((now.getTime() - start.getTime()) / 86_400_000);

    86→function tehranDayKey(dateIso: string) {
    87→  if (!dateIso) return "";
    88→  try {
    89→    return new Intl.DateTimeFormat("en-US", {
    90→      timeZone: "Asia/Tehran",
    91→      year: "numeric",
    92→      month: "2-digit",
    93→      day: "2-digit",
    94→    }).format(new Date(dateIso));
    95→  } catch {
    96→    return "";
    97→  }
    98→}
