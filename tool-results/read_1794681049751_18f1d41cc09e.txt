   178→      try {
   179→        stream = await navigator.mediaDevices.getUserMedia({
   180→          video: camId ? { deviceId: { exact: camId } } : { width: { ideal: 1280 } },
   181→          audio: micId ? { deviceId: { exact: micId } } : { echoCancellation: true, noiseSuppression: true },

   194→          const buffer = new Uint8Array(analyser.frequencyBinCount);

   218→  /* ---------------- event handling ---------------- */
   219→  const pushReaction = useCallback((emoji: string) => {
   220→    const reaction: Reaction = { id: uid(), emoji, at: Date.now() };

   225→  const handleEvent = useCallback((event: LiveEvent) => {
   226→    if (event.from === myIdRef.current && event.type !== 'system') {
   227→      // own frames echoed by WS path are ignored (chat already optimistic)
   228→      if (event.type !== 'reading') return;
   229→    }
   230→    switch (event.type) {
   231→      case 'chat': {
   232→        const payload = event.payload as { text?: string } | undefined;
   233→        const text = String(payload?.text ?? '').slice(0, 1000);
   234→        if (!text) return;
   235→        setChat((items) => [...items, { id: uid(), from: event.from, role: String(event.role), text, at: event.at, mine: false }]);
   236→        setUnread((count) => count + 1);
   237→        break;
   238→      }
   239→      case 'wb': {
   240→        const payload = event.payload as { action: string; stroke?: Stroke } | undefined;
   241→        if (payload?.action === 'clear') setStrokes([]);
   242→        else if (payload?.action === 'undo') setStrokes((items) => items.slice(0, -1));
   243→        else if (payload?.action === 'draw' && payload.stroke) setStrokes((items) => [...items, payload.stroke!]);
   244→        break;
   245→      }
   246→      case 'reading': {
   247→        const payload = event.payload as { text?: string } | undefined;
   248→        setReadingText(String(payload?.text ?? ''));
   249→        break;
   250→      }
   251→      case 'control': {
   252→        const payload = event.payload as { action: string; value?: boolean | string } | undefined;
   253→        if (payload?.action === 'hand') {
   254→          setNotice(payload.value ? '✋ دانش‌آموز دستش را بالا برده است.' : 'دست پایین آمد.');
   255→          setTimeout(() => setNotice(''), 4000);
   256→        }
   257→        if (payload?.action === 'reaction' && typeof payload.value === 'string') pushReaction(payload.value);
   258→        if (payload?.action === 'bye') {
   259→          setNotice('طرف مقابل کلاس را ترک کرد.');
   260→          setTimeout(() => setNotice(''), 4000);
   261→        }
   262→        if (payload?.action === 'end') {
   263→          setPhase('ended');
   264→        }
   265→        break;
   266→      }
   267→      case 'offer':
   268→      case 'answer':
   269→      case 'ice':
   270→        void engineRef.current?.handleSignal(event.type, event.payload);
   271→        break;
   272→      case 'system': {
   273→        const payload = event.payload as { action?: string } | undefined;
   274→        if (payload?.action === 'join') {
   275→          setPeerOnline(true);
   276→          setChat((items) => [...items, { id: uid(), from: 0, role: 'system', text: 'طرف مقابل وارد کلاس شد.', at: event.at, mine: false }]);
   277→          if (isTeacherSide) void engineRef.current?.initiate();
   278→        }
   279→        if (payload?.action === 'leave') {
   280→          setPeerOnline(false);
   281→          setChat((items) => [...items, { id: uid(), from: 0, role: 'system', text: 'طرف مقابل خارج شد.', at: event.at, mine: false }]);
   282→        }
   283→        break;
   284→      }
   285→    }
   286→  }, [isTeacherSide, pushReaction]);
   287→
   288→  /* ---------------- classroom countdown ---------------- */
   289→  useEffect(() => {
   290→    if (!session?.scheduled_at) return;
   291→    const endAt = new Date(session.scheduled_at).getTime() + (session.duration || 60) * 60_000;
   292→    const tick = () => {
   293→      const diff = endAt - Date.now();
   294→      if (diff <= 0) { setCountdown('به پایان زمان جلسه رسیده است'); return; }
   295→      const minutes = Math.floor(diff / 60_000);
   296→      const seconds = Math.floor((diff % 60_000) / 1000);
   297→      setCountdown(`${faDigits(minutes)}:${faDigits(String(seconds).padStart(2, '0'))} باقی‌مانده`);
   298→    };
   299→    tick();
   300→    const timer = setInterval(tick, 1000);
   301→    return () => clearInterval(timer);
   302→  }, [session]);
   303→
   304→  /* ---------------- board rendering ---------------- */
   305→  useEffect(() => {
   306→    const canvas = boardRef.current;
   307→    const wrapper = boardWrapperRef.current;
   308→    if (!canvas || !wrapper || sideTab !== 'board') return;
   309→    const resize = () => {
   310→      const rect = wrapper.getBoundingClientRect();
   311→      if (canvas.width !== rect.width || canvas.height !== rect.height) {
   312→        canvas.width = rect.width;
   313→        canvas.height = rect.height;
   314→      }
   315→      const ctx = canvas.getContext('2d');
   316→      if (!ctx) return;
   317→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   318→      ctx.lineCap = 'round';
   319→      ctx.lineJoin = 'round';
   320→      for (const stroke of strokes) {
   321→        ctx.strokeStyle = stroke.color;
   322→        ctx.lineWidth = stroke.width;
   323→        ctx.beginPath();
   324→        stroke.points.forEach((point, index) => {
   325→          const x = point.x * canvas.width;
   326→          const y = point.y * canvas.height;
   327→          if (index === 0) ctx.moveTo(x, y);
   328→          else ctx.lineTo(x, y);
   329→        });
   330→        ctx.stroke();
   331→      }
   332→    };
   333→    resize();
   334→    const observer = new ResizeObserver(resize);
   335→    observer.observe(wrapper);
   336→    return () => observer.disconnect();
   337→  }, [strokes, sideTab]);
   338→
   339→  /* ---------------- lifecycle ---------------- */
   340→  useEffect(() => () => {
   341→    clientRef.current?.close();
   342→    engineRef.current?.close();

   350→  };

   352→  /* ---------------- actions ---------------- */
   353→  async function joinClassroom(withMedia: boolean) {
   354→    if (!session) return;
   355→    stopPreview();
   356→    void audioCtxRef.current?.close();
   357→    audioCtxRef.current = null;
   358→
   359→    let stream: MediaStream | null = null;
   360→    if (withMedia) {
   361→      try {
   362→        stream = await navigator.mediaDevices.getUserMedia({
   363→          video: camId ? { deviceId: { exact: camId } } : true,
   364→          audio: micId ? { deviceId: { exact: micId } } : true,
   365→        });
   366→      } catch {
   367→        setMediaError('ورود با دوربین/میکروفون ناموفق بود؛ بدون رسانه وارد می‌شوید.');
   368→      }
   369→    }
   370→    liveStreamRef.current = stream;
   371→    setPhase('live');
   372→
   373→    const client = new LiveClient({
   374→      sessionId: session.id,
   375→      onEvent: handleEvent,
   376→      onStatus: (value) => setStatus(value),
   377→    });
   378→    clientRef.current = client;
   379→
   380→    const engine = new LiveEngine({
   381→      role,
   382→      onRemoteStream: (remote) => {
   383→        setRemoteActive(remote.getTracks().length > 0);
   384→        if (remoteVideoRef.current) remoteVideoRef.current.srcObject = remote;
   385→      },
   386→      onState: (state) => setConnState(`${state.connection}/${state.ice}`),
   387→      send: (type, payload) => client.send(type, payload),
   388→    });
   389→    engineRef.current = engine;
   390→    engine.setCameraTrack(stream?.getVideoTracks()[0] ?? null);
   391→    await engine.ensureConnection();
   392→    if (stream) await engine.setLocalMedia(stream);
   393→    if (localVideoRef.current && stream) localVideoRef.current.srcObject = stream;
   394→    if (engine.isInitiator && peerOnline) void engine.initiate();
   395→
   396→    await client.connect();
   397→  }

   399→  function sendChat() {
   400→    const text = chatDraft.trim();
   401→    if (!text || !clientRef.current) return;
   402→    setChat((items) => [...items, { id: uid(), from: myIdRef.current, role, text, at: Date.now(), mine: true }]);
   403→    clientRef.current.send('chat', { text: text.slice(0, 1000) });
   404→    setChatDraft('');
   405→  }
   406→
   407→  function sendReading(text: string) {
   408→    setReadingText(text);

   410→  }

   412→  function sendReaction(emoji: string) {
   413→    pushReaction(emoji);
   414→    clientRef.current?.send('control', { action: 'reaction', value: emoji });
   415→  }
   416→
   417→  function toggleHand() {
   418→    const next = !handRaised;
   419→    setHandRaised(next);
   420→    clientRef.current?.send('control', { action: 'hand', value: next });
   421→  }
   422→
   423→  async function toggleShare() {
   424→    const engine = engineRef.current;
   425→    if (!engine) return;
   426→    if (sharing) {
   427→      screenRef.current?.getTracks().forEach((track) => track.stop());
   428→      screenRef.current = null;
   429→      await engine.stopScreenShare();
   430→      if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;
   431→      setSharing(false);
   432→      return;
   433→    }
   434→    try {
   435→      const screen = await engine.startScreenShare();
   436→      screenRef.current = screen;
   437→      if (localVideoRef.current) localVideoRef.current.srcObject = screen;
   438→      setSharing(true);
   439→      screen.getVideoTracks()[0]?.addEventListener('ended', () => {
   440→        setSharing(false);
   441→        if (localVideoRef.current && liveStreamRef.current) localVideoRef.current.srcObject = liveStreamRef.current;
   442→      });
   443→    } catch (cause) {
   444→      setNotice(cause instanceof Error ? cause.message : 'اشتراک صفحه انجام نشد.');
