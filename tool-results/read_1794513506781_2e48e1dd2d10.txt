    54→function tehranHour() {
    55→  try {
    56→    return Number(
    57→      new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tehran", hour: "numeric", hour12: false }).format(new Date()),
    58→    );
    59→  } catch {
    60→    return new Date().getHours();
    61→  }
    62→}

    68→  if (hour < 20) return "عصر بخیر";
    69→  return "شب بخیر";

    76→function dayOfYearIndex() {
    77→  const now = new Date();
    78→  const start = new Date(now.getFullYear(), 0, 0);
    79→  return Math.floor((now.getTime() - start.getTime()) / 86_400_000);
    80→}

    82→function tehranDayKey(dateIso: string) {
    83→  if (!dateIso) return "";
    84→  try {
    85→    return new Intl.DateTimeFormat("en-US", {
    86→      timeZone: "Asia/Tehran",
    87→      year: "numeric",
    88→      month: "2-digit",
    89→      day: "2-digit",
    90→    }).format(new Date(dateIso));
    91→  } catch {
    92→    return "";
    93→  }
    94→}

    96→function isTodayInTehran(dateIso: string) {
